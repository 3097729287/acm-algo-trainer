## [USACO 2010 Feb G]Slowing down

### 题意

牛依次从根走向各自牧场，统计经过已到达牛的次数。

### 思路

已停在节点u的牛，会使后来目标位于u子树的牛减速。因此每到达一头牛，就给u的整个DFS子树区间加1；下一头牛的减速次数是其目标点当前位置值。用差分树状数组做区间加、单点查询，先查询再加入。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct EulerTree {
    int n;
    vector<vector<int>> edge;
    vector<int> parent, depth, in, out, order;
    EulerTree(int n) : n(n), edge(n + 1), parent(n + 1), depth(n + 1), in(n + 1), out(n + 1) {}
    void add(int u, int v) {
        edge[u].push_back(v);
        edge[v].push_back(u);
    }
    void root(int start) {
        vector<int> stack = {start};
        parent[start] = 0;
        order.clear();
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            in[u] = order.size() + 1;
            order.push_back(u);
            for (int v : edge[u])
                if (v != parent[u]) {
                    parent[v] = u;
                    depth[v] = depth[u] + 1;
                    stack.push_back(v);
                }
        }
        vector<int> size(n + 1, 1);
        for (int i = (int)order.size() - 1; i >= 0; i--) {
            int u = order[i];
            out[u] = in[u] + size[u] - 1;
            if (parent[u])
                size[parent[u]] += size[u];
        }
    }
};

struct Fenwick {
    int n;
    vector<long long> tree;
    Fenwick(int n) : n(n), tree(n + 1) {}
    void add(int p, long long v) {
        for (; p <= n; p += p & -p)
            tree[p] += v;
    }
    long long sum(int p) {
        long long result = 0;
        for (; p; p -= p & -p)
            result += tree[p];
        return result;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    EulerTree tree(n);
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        tree.add(u, v);
    }
    tree.root(1);
    Fenwick bit(n);
    for (int i = 0; i < n; i++) {
        int u;
        cin >> u;
        cout << bit.sum(tree.in[u]) << '\n';
        bit.add(tree.in[u], 1);
        if (tree.out[u] < n)
            bit.add(tree.out[u] + 1, -1);
    }
    return 0;
}
```

### 复杂度

时间 $O(n\log n)$，空间 $O(n)$。

### 易错点

根节点也可能是某头牛的牧场，会影响之后所有牛。当前牛尚未到达，不能先插入再查询。
