## Cross on a Plane

### 题意

两条互相垂直且同宽的无限条带并成十字，求覆盖所有点的最小宽度。

### 思路

固定十字方向θ，将点投影成(u,v)。若第一条带覆盖按u排序的一段点，其他点必须能放进宽度w的v条带。枚举第一条带左端为某个点的u，用双指针扩展所有u≤左端+w的点；外部点的v最小最大值可由前缀、后缀极值得到，差≤w就可行。这一步O(n log n)。

还需完整搜索连续方向，不能三分一个可能多峰的函数。固定候选宽度w后，方向变化时，任意点对能否同时放进u条带或v条带，只在投影差的绝对值等于w时改变。点对方向φ、长度d若d≥w，则事件为φ±acos(w/d)；方向对π/2取模自动同时包含v条带的事件，因为交换两条带不改变十字。

将所有O(n²)个事件角加0和π/2排序，在每个相邻事件间取中点检查。区间内全部点对兼容关系不变，所以可行性不变；检查θ=0也覆盖端点情形。二分w至充分精度；若最优只在孤立角度取得，任意略大于最优的宽度都会产生可行角区间，因此仍可逼近正确答案。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

constexpr long double EPS = 1e-12L;
struct Point {
    long double x = 0, y = 0;
    Point operator+(Point b) const { return {x + b.x, y + b.y}; }
    Point operator-(Point b) const { return {x - b.x, y - b.y}; }
    Point operator*(long double t) const { return {x * t, y * t}; }
    Point operator/(long double t) const { return {x / t, y / t}; }
};
long double cross(Point a, Point b) { return a.x * b.y - a.y * b.x; }
long double dot(Point a, Point b) { return a.x * b.x + a.y * b.y; }
long double norm(Point a) { return sqrtl(dot(a, a)); }
long double distance(Point a, Point b) { return norm(a - b); }
Point rotate90(Point a) { return {-a.y, a.x}; }

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<Point> p(n);
    for (auto &q : p)
        cin >> q.x >> q.y;
    const long double period = acosl(-1.0L) / 2;
    auto directionWorks = [&](long double angle, long double width) {
        vector<pair<long double, long double>> q;
        long double c = cosl(angle), s = sinl(angle);
        for (auto a : p)
            q.push_back({a.x * c + a.y * s, -a.x * s + a.y * c});
        sort(q.begin(), q.end());
        vector<long double> low(n + 1, 1e100L), high(n + 1, -1e100L), suffixLow = low,
                                                                      suffixHigh = high;
        for (int i = 0; i < n; i++) {
            low[i + 1] = min(low[i], q[i].second);
            high[i + 1] = max(high[i], q[i].second);
        }
        for (int i = n - 1; i >= 0; i--) {
            suffixLow[i] = min(suffixLow[i + 1], q[i].second);
            suffixHigh[i] = max(suffixHigh[i + 1], q[i].second);
        }
        int r = 0;
        for (int l = 0; l < n; l++) {
            r = max(r, l);
            while (r < n && q[r].first - q[l].first <= width + 1e-12L)
                r++;
            long double bottom = min(low[l], suffixLow[r]), top = max(high[l], suffixHigh[r]);
            if (top - bottom <= width + 1e-12L)
                return true;
        }
        return false;
    };
    auto feasible = [&](long double width) {
        vector<long double> event = {0, period};
        auto normalized = [&](long double angle) {
            angle = fmodl(angle, period);
            if (angle < 0)
                angle += period;
            return angle;
        };
        for (int i = 0; i < n; i++)
            for (int j = 0; j < i; j++) {
                Point d = p[i] - p[j];
                long double length = norm(d);
                if (length < width)
                    continue;
                long double base = atan2l(d.y, d.x),
                            delta = acosl(clamp(width / length, -1.0L, 1.0L));
                event.push_back(normalized(base + delta));
                event.push_back(normalized(base - delta));
            }
        sort(event.begin(), event.end());
        if (directionWorks(0, width))
            return true;
        for (int i = 1; i < (int)event.size(); i++)
            if (directionWorks((event[i - 1] + event[i]) / 2, width))
                return true;
        return false;
    };
    long double l = 0, r = 30000;
    for (int i = 0; i < 55; i++) {
        long double mid = (l + r) / 2;
        if (feasible(mid))
            r = mid;
        else
            l = mid;
    }
    cout << fixed << setprecision(12) << r;
    return 0;
}
```

### 复杂度

每次判定 $O(n^3\log n)$，共 $O(n^3\log n\log(C/\varepsilon))$ 时间，空间 $O(n^2)$。

### 易错点

条带可以任意平移，不要求十字中心在某个输入点上。固定角度不能只用全体u、v跨度；应允许把点分给两条带。
