## [网络流24题]运输问题

### 题意

供需平衡运输，求最低与最高总费用。

### 思路

源→仓库容量供应，仓库→商店容量总货量、费用c，商店→汇容量需求。满流恰为合法全部运输方案，最小费用流给最小费用。再用独立网络把所有运输费用取负，求最小费用后取负得到最大费用。每次找残量最短路并增广瓶颈量，按流量乘路径费用累积。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct CostFlow {
    struct Edge {
        int to, reverse;
        long long capacity, cost;
    };
    vector<vector<Edge>> graph;
    explicit CostFlow(int n) : graph(n) {}
    void add(int u, int v, long long capacity, long long cost) {
        int a = graph[u].size(), b = graph[v].size();
        graph[u].push_back({v, b, capacity, cost});
        graph[v].push_back({u, a, 0, -cost});
    }
    pair<long long, long long> minimum(int source, int sink) {
        const long long inf = LLONG_MAX / 4;
        long long flow = 0, cost = 0;
        int n = graph.size();
        while (true) {
            vector<long long> distance(n, inf);
            vector<int> parent(n, -1), index(n);
            vector<bool> inQueue(n);
            queue<int> queue;
            queue.push(source);
            inQueue[source] = true;
            distance[source] = 0;
            while (!queue.empty()) {
                int u = queue.front();
                queue.pop();
                inQueue[u] = false;
                for (int i = 0; i < (int)graph[u].size(); i++) {
                    auto e = graph[u][i];
                    if (e.capacity > 0 && distance[e.to] > distance[u] + e.cost) {
                        distance[e.to] = distance[u] + e.cost;
                        parent[e.to] = u;
                        index[e.to] = i;
                        if (!inQueue[e.to]) {
                            inQueue[e.to] = true;
                            queue.push(e.to);
                        }
                    }
                }
            }
            if (parent[sink] < 0)
                break;
            long long amount = inf;
            for (int v = sink; v != source; v = parent[v])
                amount = min(amount, graph[parent[v]][index[v]].capacity);
            for (int v = sink; v != source; v = parent[v]) {
                auto &e = graph[parent[v]][index[v]];
                e.capacity -= amount;
                graph[v][e.reverse].capacity += amount;
            }
            flow += amount;
            cost += amount * distance[sink];
        }
        return {flow, cost};
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int m, n;
    cin >> m >> n;
    vector<long long> supply(m), demand(n);
    long long total = 0;
    for (auto &x : supply) {
        cin >> x;
        total += x;
    }
    for (auto &x : demand)
        cin >> x;
    vector<vector<long long>> cost(m, vector<long long>(n));
    for (auto &row : cost)
        for (auto &x : row)
            cin >> x;
    auto solve = [&](int sign) {
        int source = m + n, sink = source + 1;
        CostFlow flow(sink + 1);
        for (int i = 0; i < m; i++)
            flow.add(source, i, supply[i], 0);
        for (int j = 0; j < n; j++)
            flow.add(m + j, sink, demand[j], 0);
        for (int i = 0; i < m; i++)
            for (int j = 0; j < n; j++)
                flow.add(i, m + j, total, sign * cost[i][j]);
        return sign * flow.minimum(source, sink).second;
    };
    cout << solve(1) << '\n' << solve(-1);
    return 0;
}
```

### 复杂度

每次SPFA最坏 $O(VE)$，若增广A次总 $O(AVE)$，空间 $O(V+E)$。

### 易错点

两次计算必须新建网络，不能直接在已流满残量图上改费用。容量和总费用使用64位。
