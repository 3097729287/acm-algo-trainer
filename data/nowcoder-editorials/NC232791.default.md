## 最小矩形覆盖面积

### 题意

求覆盖全部点的最小矩形面积。

### 思路

最优矩形至少一边与凸包边平行，枚举凸包边方向e。沿e投影的最大最小差为长，沿垂线投影差为宽。凸包都在有向边左侧，垂线最小投影为该边所在直线，只需最远点；另外维护沿e投影最大、最小的两个支撑点。

边方向绕凸包旋转时三支撑点都单调前进，用旋转卡壳求投影差，面积=(maxDot−minDot)*maxCross/|e|²。无需归一化向量，也不输出此页未要求的矩形顶点。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

constexpr long double EPS = 1e-12L;
struct Point {
    long double x = 0, y = 0;
    Point operator+(Point b) const { return {x + b.x, y + b.y}; }
    Point operator-(Point b) const { return {x - b.x, y - b.y}; }
    Point operator*(long double t) const { return {x * t, y * t}; }
    Point operator/(long double t) const { return {x / t, y / t}; }
};
long double cross(Point a, Point b) { return a.x * b.y - a.y * b.x; }
long double dot(Point a, Point b) { return a.x * b.x + a.y * b.y; }
long double norm(Point a) { return sqrtl(dot(a, a)); }
long double distance(Point a, Point b) { return norm(a - b); }
Point rotate90(Point a) { return {-a.y, a.x}; }

vector<Point> convexHull(vector<Point> point) {
    sort(point.begin(), point.end(),
         [](Point a, Point b) { return a.x != b.x ? a.x < b.x : a.y < b.y; });
    point.erase(unique(point.begin(), point.end(),
                       [](Point a, Point b) { return a.x == b.x && a.y == b.y; }),
                point.end());
    if (point.size() < 2)
        return point;
    vector<Point> hull;
    for (Point p : point) {
        while (hull.size() >= 2 && cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    size_t lower = hull.size();
    for (int i = (int)point.size() - 2; i >= 0; i--) {
        Point p = point[i];
        while (hull.size() > lower &&
               cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    hull.pop_back();
    return hull;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<Point> p(n);
    for (auto &a : p)
        cin >> a.x >> a.y;
    auto h = convexHull(p);
    int m = h.size();
    if (m < 3) {
        cout << "0.000000000000";
        return 0;
    }
    int high = 0, maximum = 0, minimum = 0;
    Point first = h[1] - h[0];
    for (int j = 1; j < m; j++) {
        if (cross(first, h[j] - h[0]) > cross(first, h[high] - h[0]))
            high = j;
        if (dot(first, h[j]) > dot(first, h[maximum]))
            maximum = j;
        if (dot(first, h[j]) < dot(first, h[minimum]))
            minimum = j;
    }
    long double answer = 1e100L;
    for (int i = 0; i < m; i++) {
        Point e = h[(i + 1) % m] - h[i];
        auto area = [&](int j) { return cross(e, h[j] - h[i]); };
        while (area((high + 1) % m) > area(high) + EPS)
            high = (high + 1) % m;
        while (dot(e, h[(maximum + 1) % m]) > dot(e, h[maximum]) + EPS)
            maximum = (maximum + 1) % m;
        while (dot(e, h[(minimum + 1) % m]) < dot(e, h[minimum]) - EPS)
            minimum = (minimum + 1) % m;
        long double value = (dot(e, h[maximum]) - dot(e, h[minimum])) * area(high) / dot(e, e);
        answer = min(answer, value);
    }
    cout << fixed << setprecision(12) << answer;
    return 0;
}
```

### 复杂度

时间 $O(n\log n)$，卡壳 $O(n)$，空间 $O(n)$。

### 易错点

矩形可任意旋转，不是轴平行包围盒。全共线时面积为0。
