## 牛可乐和公平点

### 题意

查询到x与y等距的树节点数。

### 思路

x=y时全树等距；路径长度为奇数时没有整数节点中点，答案0。长度为偶数时，等距点恰是去掉中点通向x、y的两条边后，包含中点的连通块。

若x、y同深度，中点为LCA，设其两侧儿子为a、b，答案n−size[a]−size[b]。否则令x较深，中点z在x一侧，令a为z通向x的儿子，答案size[z]−size[a]。用倍增跳指定步数得到这些点。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct EulerTree {
    int n;
    vector<vector<int>> edge;
    vector<int> parent, depth, in, out, order;
    EulerTree(int n) : n(n), edge(n + 1), parent(n + 1), depth(n + 1), in(n + 1), out(n + 1) {}
    void add(int u, int v) {
        edge[u].push_back(v);
        edge[v].push_back(u);
    }
    void root(int start) {
        vector<int> stack = {start};
        parent[start] = 0;
        order.clear();
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            in[u] = order.size() + 1;
            order.push_back(u);
            for (int v : edge[u])
                if (v != parent[u]) {
                    parent[v] = u;
                    depth[v] = depth[u] + 1;
                    stack.push_back(v);
                }
        }
        vector<int> size(n + 1, 1);
        for (int i = (int)order.size() - 1; i >= 0; i--) {
            int u = order[i];
            out[u] = in[u] + size[u] - 1;
            if (parent[u])
                size[parent[u]] += size[u];
        }
    }
};

struct Ancestors {
    const EulerTree &tree;
    vector<vector<int>> up;
    Ancestors(const EulerTree &tree) : tree(tree) {
        int levels = 1;
        while ((1LL << levels) <= tree.n)
            levels++;
        up.assign(levels, vector<int>(tree.n + 1));
        up[0] = tree.parent;
        for (int b = 1; b < levels; b++)
            for (int u = 1; u <= tree.n; u++)
                up[b][u] = up[b - 1][up[b - 1][u]];
    }
    int lca(int u, int v) const {
        if (tree.depth[u] < tree.depth[v])
            swap(u, v);
        int gap = tree.depth[u] - tree.depth[v];
        for (int b = 0; b < (int)up.size(); b++)
            if (gap >> b & 1)
                u = up[b][u];
        if (u == v)
            return u;
        for (int b = (int)up.size() - 1; b >= 0; b--)
            if (up[b][u] != up[b][v]) {
                u = up[b][u];
                v = up[b][v];
            }
        return up[0][u];
    }
    int distance(int u, int v) const {
        return tree.depth[u] + tree.depth[v] - 2 * tree.depth[lca(u, v)];
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    EulerTree tree(n);
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        tree.add(u, v);
    }
    tree.root(1);
    Ancestors ancestor(tree);
    vector<int> size(n + 1, 1);
    for (int i = n - 1; i > 0; i--) {
        int u = tree.order[i];
        size[tree.parent[u]] += size[u];
    }
    auto jump = [&](int u, int steps) {
        for (int b = 0; steps; b++, steps >>= 1)
            if (steps & 1)
                u = ancestor.up[b][u];
        return u;
    };
    int q;
    cin >> q;
    while (q--) {
        int x, y;
        cin >> x >> y;
        if (x == y) {
            cout << n << '\n';
            continue;
        }
        int distance = ancestor.distance(x, y);
        if (distance % 2) {
            cout << 0 << '\n';
            continue;
        }
        int half = distance / 2;
        if (tree.depth[x] == tree.depth[y])
            cout << n - size[jump(x, half - 1)] - size[jump(y, half - 1)] << '\n';
        else {
            if (tree.depth[x] < tree.depth[y])
                swap(x, y);
            int z = jump(x, half), a = jump(x, half - 1);
            cout << size[z] - size[a] << '\n';
        }
    }
    return 0;
}
```

### 复杂度

预处理 $O(n\log n)$，每问 $O(\log n)$，空间 $O(n\log n)$。

### 易错点

等距点不是只算路径中点，还包含中点旁支。半长至少1，x=y已单独处理。
