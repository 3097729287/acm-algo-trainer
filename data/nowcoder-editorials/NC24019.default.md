## Max Flow

### 题意

树上批量路径计数。

### 思路

每条点路径u→v对所有经过节点加1。设a=LCA(u,v)，差分标记diff[u]++、diff[v]++、diff[a]−−、diff[parent(a)]−−。逆序累加子树到父节点后，a以下两条链各得到1，a本身也得到1，a以上抵消为0。全部路径累加后取最大节点值。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct EulerTree {
    int n;
    vector<vector<int>> edge;
    vector<int> parent, depth, in, out, order;
    EulerTree(int n) : n(n), edge(n + 1), parent(n + 1), depth(n + 1), in(n + 1), out(n + 1) {}
    void add(int u, int v) {
        edge[u].push_back(v);
        edge[v].push_back(u);
    }
    void root(int start) {
        vector<int> stack = {start};
        parent[start] = 0;
        order.clear();
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            in[u] = order.size() + 1;
            order.push_back(u);
            for (int v : edge[u])
                if (v != parent[u]) {
                    parent[v] = u;
                    depth[v] = depth[u] + 1;
                    stack.push_back(v);
                }
        }
        vector<int> size(n + 1, 1);
        for (int i = (int)order.size() - 1; i >= 0; i--) {
            int u = order[i];
            out[u] = in[u] + size[u] - 1;
            if (parent[u])
                size[parent[u]] += size[u];
        }
    }
};

struct Ancestors {
    const EulerTree &tree;
    vector<vector<int>> up;
    Ancestors(const EulerTree &tree) : tree(tree) {
        int levels = 1;
        while ((1LL << levels) <= tree.n)
            levels++;
        up.assign(levels, vector<int>(tree.n + 1));
        up[0] = tree.parent;
        for (int b = 1; b < levels; b++)
            for (int u = 1; u <= tree.n; u++)
                up[b][u] = up[b - 1][up[b - 1][u]];
    }
    int lca(int u, int v) const {
        if (tree.depth[u] < tree.depth[v])
            swap(u, v);
        int gap = tree.depth[u] - tree.depth[v];
        for (int b = 0; b < (int)up.size(); b++)
            if (gap >> b & 1)
                u = up[b][u];
        if (u == v)
            return u;
        for (int b = (int)up.size() - 1; b >= 0; b--)
            if (up[b][u] != up[b][v]) {
                u = up[b][u];
                v = up[b][v];
            }
        return up[0][u];
    }
    int distance(int u, int v) const {
        return tree.depth[u] + tree.depth[v] - 2 * tree.depth[lca(u, v)];
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    int k;
    cin >> n >> k;
    vector<int> visit;
    EulerTree tree(n);
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        tree.add(u, v);
    }
    tree.root(1);
    Ancestors ancestor(tree);
    vector<long long> difference(n + 1);
    auto addPath = [&](int u, int v) {
        int a = ancestor.lca(u, v);
        difference[u]++;
        difference[v]++;
        difference[a]--;
        difference[tree.parent[a]]--;
    };
    while (k--) {
        int u, v;
        cin >> u >> v;
        addPath(u, v);
    }
    for (int i = n - 1; i > 0; i--) {
        int u = tree.order[i];
        difference[tree.parent[u]] += difference[u];
    }
    cout << *max_element(difference.begin() + 1, difference.end());
    return 0;
}
```

### 复杂度

时间 $O((n+q)\log n)$，空间 $O(n\log n)$。

### 易错点

这里统计点而非边，LCA本身必须保留一次；根的父亲用0哨兵，不向它继续累加。
