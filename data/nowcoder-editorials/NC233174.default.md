## Abstract Art

### 题意

多个可凹简单多边形涂满后，求颜料总面积与画布覆盖面积。

### 思路

颜料总量是各多边形面积之和。覆盖面积使用Green公式：并集边界上有向线段AB的积分贡献是cross(A,B)/2。把每个多边形统一为逆时针，逐条边找没有被其他多边形覆盖的部分，只对这些外露边积分。

固定一条边AB，将它与另一个多边形所有边的交点参数，以及共线重叠段端点，加入0、1并排序。每两个相邻参数间不会再跨越边界，用中点绕数判断是否在另一多边形内；若在内部，该区间被覆盖。若正好落在重合边界上，同向边只保留编号较小多边形的一份，反向边两侧都是内部，应两份都去掉。

把所有其他多边形产生的覆盖参数区间求并，外露比例为1减覆盖总长。整条AB的积分乘这个比例，最后将所有边相加。即使一条边被切成多段，它们的积分仍按参数长度线性缩放。此方法直接支持凹多边形、包含以及并集中的洞。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

constexpr long double EPS = 1e-12L;
struct Point {
    long double x = 0, y = 0;
    Point operator+(Point b) const { return {x + b.x, y + b.y}; }
    Point operator-(Point b) const { return {x - b.x, y - b.y}; }
    Point operator*(long double t) const { return {x * t, y * t}; }
    Point operator/(long double t) const { return {x / t, y / t}; }
};
long double cross(Point a, Point b) { return a.x * b.y - a.y * b.x; }
long double dot(Point a, Point b) { return a.x * b.x + a.y * b.y; }
long double norm(Point a) { return sqrtl(dot(a, a)); }
long double distance(Point a, Point b) { return norm(a - b); }
Point rotate90(Point a) { return {-a.y, a.x}; }

int location(const vector<Point> &polygon, Point p, int &edge) {
    int winding = 0, n = polygon.size();
    for (int i = 0; i < n; i++) {
        Point a = polygon[i], b = polygon[(i + 1) % n];
        long double side = cross(b - a, p - a);
        if (fabsl(side) < 1e-9L && dot(p - a, p - b) <= 1e-9L) {
            edge = i;
            return 2;
        }
        if (a.y <= p.y) {
            if (b.y > p.y && side > 0)
                winding++;
        } else if (b.y <= p.y && side < 0)
            winding--;
    }
    return winding != 0;
}
long double signedArea(const vector<Point> &p) {
    long double result = 0;
    for (int i = 0; i < (int)p.size(); i++)
        result += cross(p[i], p[(i + 1) % p.size()]);
    return result / 2;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<vector<Point>> polygon(n);
    long double paint = 0;
    for (auto &p : polygon) {
        int m;
        cin >> m;
        p.resize(m);
        for (auto &q : p)
            cin >> q.x >> q.y;
        long double a = signedArea(p);
        paint += fabsl(a);
        if (a < 0)
            reverse(p.begin(), p.end());
    }
    long double coveredArea = 0;
    for (int id = 0; id < n; id++)
        for (int at = 0; at < (int)polygon[id].size(); at++) {
            Point a = polygon[id][at], b = polygon[id][(at + 1) % polygon[id].size()], e = b - a;
            vector<pair<long double, long double>> covered;
            for (int other = 0; other < n; other++)
                if (other != id) {
                    const auto &p = polygon[other];
                    vector<long double> cut = {0, 1};
                    for (int j = 0; j < (int)p.size(); j++) {
                        Point u = p[j], v = p[(j + 1) % p.size()], f = v - u;
                        long double den = cross(e, f);
                        if (fabsl(den) > 1e-12L) {
                            long double t = cross(u - a, f) / den, s = cross(u - a, e) / den;
                            if (t > 0 && t < 1 && s >= -1e-12L && s <= 1 + 1e-12L)
                                cut.push_back(t);
                        } else if (fabsl(cross(e, u - a)) < 1e-9L) {
                            for (Point q : {u, v}) {
                                long double t = dot(q - a, e) / dot(e, e);
                                if (t > 0 && t < 1)
                                    cut.push_back(t);
                            }
                        }
                    }
                    sort(cut.begin(), cut.end());
                    for (int j = 1; j < (int)cut.size(); j++) {
                        long double l = cut[j - 1], r = cut[j];
                        if (r - l < 1e-15L)
                            continue;
                        int edge = -1, where = location(p, a + e * ((l + r) / 2), edge);
                        bool hide = where == 1;
                        if (where == 2) {
                            Point f = p[(edge + 1) % p.size()] - p[edge];
                            hide = dot(e, f) < 0 || other < id;
                        }
                        if (hide)
                            covered.push_back({l, r});
                    }
                }
            sort(covered.begin(), covered.end());
            long double length = 0, reach = 0;
            for (auto [l, r] : covered) {
                if (r > reach) {
                    length += r - max(l, reach);
                    reach = r;
                }
            }
            long double exposed = clamp(1 - length, 0.0L, 1.0L);
            coveredArea += cross(a, b) * exposed / 2;
        }
    cout << fixed << setprecision(12) << paint << ' ' << fabsl(coveredArea);
    return 0;
}
```

### 复杂度

设总边数E、每个多边形最多M=20条边，时间 $O(E^2M+E^2\log E)$，空间 $O(E)$。

### 易错点

不能把每两个多边形交面积直接做一次容斥，多重覆盖需要更高阶项。只有最终有向边积分求和后才取绝对值。重合边要按方向和编号处理。
