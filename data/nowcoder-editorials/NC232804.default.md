## How Many Triangles

### 题意

多组统计锐角三角形，允许三点共线。

### 思路

所有三点组C(n,3)，减去直角、钝角及退化三角形。非退化三角形最多一个≥90°角，所以按顶点统计角度在[90°,180°)的向量对，每个坏三角形恰减一次。共线三点则只在中间点有180°角，额外统计相反方向向量对一次。

极角排序并复制一圈，用三个单调指针分出<90°、<180°、=180°区间。叉积确定是否在正半周，点积≤0确定≥90°，相反向量叉积0且点积负。180°会按两个方向计算，累计除2。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

using Wide = __int128_t;
struct IPoint {
    long long x = 0, y = 0;
    IPoint operator+(IPoint b) const { return {x + b.x, y + b.y}; }
    IPoint operator-(IPoint b) const { return {x - b.x, y - b.y}; }
};
Wide cross(IPoint a, IPoint b) { return (Wide)a.x * b.y - (Wide)a.y * b.x; }
Wide dot(IPoint a, IPoint b) { return (Wide)a.x * b.x + (Wide)a.y * b.y; }
Wide absolute(Wide x) { return x < 0 ? -x : x; }
void printWide(Wide value) {
    if (value < 0) {
        cout << '-';
        value = -value;
    }
    string text;
    do {
        text.push_back('0' + value % 10);
        value /= 10;
    } while (value);
    reverse(text.begin(), text.end());
    cout << text;
}
bool angleLess(IPoint a, IPoint b) {
    auto half = [](IPoint p) { return p.y < 0 || (p.y == 0 && p.x < 0); };
    if (half(a) != half(b))
        return half(a) < half(b);
    return cross(a, b) > 0;
}
vector<IPoint> integerHull(vector<IPoint> point) {
    sort(point.begin(), point.end(),
         [](auto a, auto b) { return a.x != b.x ? a.x < b.x : a.y < b.y; });
    point.erase(
        unique(point.begin(), point.end(), [](auto a, auto b) { return a.x == b.x && a.y == b.y; }),
        point.end());
    if (point.size() < 2)
        return point;
    vector<IPoint> hull;
    for (auto p : point) {
        while (hull.size() >= 2 && cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    size_t lower = hull.size();
    for (int i = (int)point.size() - 2; i >= 0; i--) {
        auto p = point[i];
        while (hull.size() > lower &&
               cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    hull.pop_back();
    return hull;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    while (cin >> n) {
        vector<IPoint> p(n);
        for (auto &a : p)
            cin >> a.x >> a.y;
        long long bad = 0, opposite = 0;
        for (int origin = 0; origin < n; origin++) {
            vector<IPoint> v;
            for (int i = 0; i < n; i++)
                if (i != origin)
                    v.push_back(p[i] - p[origin]);
            sort(v.begin(), v.end(), angleLess);
            int m = v.size();
            for (int i = 0; i < m; i++)
                v.push_back(v[i]);
            int ninety = 1, straight = 1, after = 1;
            for (int i = 0; i < m; i++) {
                straight = max(straight, i + 1);
                while (straight < i + m && (cross(v[i], v[straight]) > 0 ||
                                            (straight < m && cross(v[i], v[straight]) == 0 &&
                                             dot(v[i], v[straight]) > 0)))
                    straight++;
                ninety = max(ninety, i + 1);
                while (ninety < straight && dot(v[i], v[ninety]) > 0)
                    ninety++;
                bad += straight - ninety;
                after = max(after, straight);
                while (after < i + m && cross(v[i], v[after]) == 0 && dot(v[i], v[after]) < 0)
                    after++;
                opposite += after - straight;
            }
        }
        cout << 1LL * n * (n - 1) * (n - 2) / 6 - bad - opposite / 2 << '\n';
    }
    return 0;
}
```

### 复杂度

每组时间 $O(n²\log n)$，空间 $O(n)$。

### 易错点

直角也不算锐角，点积为0要减去。共线退化三角形不能漏减或按两个方向减两次。
