## [SCOI2011]糖果

### 题意

每人至少一个糖果，满足相等及严格、非严格比较关系，最小化总数。

### 思路

统一为lower[v]≥lower[u]+w，w只能0或1。相等连两条0边，A<B连A→B权1，A≥B连B→A权0，A>B连B→A权1，A≤B连A→B权0。非负边图中，矛盾等价于某个有向环包含1边；缩SCC后，只要内部有1边就输出−1。

其余SCC内全为0，所有人可取同一个最小值。缩点DAG上所有分量初值1，拓扑求最长下界d[v]=max(d[v],d[u]+w)。这些值同时达到全部约束的逐点最小值，因此按每个分量人数累加就是最小总糖果数。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

pair<int, vector<int>> components(const vector<vector<int>> &graph,
                                  const vector<vector<int>> &reverse) {
    int n = graph.size();
    vector<bool> seen(n);
    vector<int> finish;
    for (int start = 0; start < n; start++)
        if (!seen[start]) {
            vector<pair<int, int>> stack{{start, 0}};
            seen[start] = true;
            while (!stack.empty()) {
                int u = stack.back().first, &position = stack.back().second;
                if (position == (int)graph[u].size()) {
                    finish.push_back(u);
                    stack.pop_back();
                } else {
                    int v = graph[u][position++];
                    if (!seen[v]) {
                        seen[v] = true;
                        stack.push_back({v, 0});
                    }
                }
            }
        }
    vector<int> component(n, -1);
    int count = 0;
    for (int i = n - 1; i >= 0; i--) {
        int start = finish[i];
        if (component[start] != -1)
            continue;
        vector<int> stack{start};
        component[start] = count;
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            for (int v : reverse[u])
                if (component[v] == -1) {
                    component[v] = count;
                    stack.push_back(v);
                }
        }
        count++;
    }
    return {count, component};
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, k;
    cin >> n >> k;
    struct Edge {
        int a, b, w;
    };
    vector<Edge> edges;
    vector<vector<int>> graph(n), reverse(n);
    auto addEdge = [&](int a, int b, int w) {
        edges.push_back({a, b, w});
        graph[a].push_back(b);
        reverse[b].push_back(a);
    };
    while (k--) {
        int type, a, b;
        cin >> type >> a >> b;
        --a;
        --b;
        if (type == 1) {
            addEdge(a, b, 0);
            addEdge(b, a, 0);
        } else if (type == 2)
            addEdge(a, b, 1);
        else if (type == 3)
            addEdge(b, a, 0);
        else if (type == 4)
            addEdge(b, a, 1);
        else
            addEdge(a, b, 0);
    }
    auto [count, component] = components(graph, reverse);
    vector<vector<pair<int, int>>> dag(count);
    vector<int> degree(count);
    for (auto e : edges) {
        int a = component[e.a], b = component[e.b];
        if (a == b) {
            if (e.w) {
                cout << -1;
                return 0;
            }
        } else {
            dag[a].push_back({b, e.w});
            degree[b]++;
        }
    }
    vector<int> queue;
    vector<long long> value(count, 1);
    for (int c = 0; c < count; c++)
        if (!degree[c])
            queue.push_back(c);
    for (size_t h = 0; h < queue.size(); h++) {
        int u = queue[h];
        for (auto [v, w] : dag[u]) {
            value[v] = max(value[v], value[u] + w);
            if (--degree[v] == 0)
                queue.push_back(v);
        }
    }
    long long answer = 0;
    for (int u = 0; u < n; u++)
        answer += value[component[u]];
    cout << answer;
    return 0;
}
```

### 复杂度

时间、空间均为 $O(N+K)$。

### 易错点

比较是整数，严格少于至少差1。相同编号的严格比较会形成1权自环，必须判无解。总数可能超过32位。
