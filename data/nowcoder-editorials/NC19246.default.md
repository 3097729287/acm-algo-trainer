## 数据结构

### 题意

支持区间乘、区间加，以及区间和、平方和查询。

### 思路

统一把修改写成 x→ax+b。一个长 len 的节点维护 S=Σx、Q=Σx²，应用修改后 S′=aS+b len，Q′=a²Q+2abS+b²len。懒标记也按先后顺序复合：原标记 (m,c) 后再执行 (a,b)，变成 (am,ac+b)。求平方和时必须使用修改前的 S。

题目只保证查询结果落在有符号64位，未保证长时间未查询的中间量与懒乘积不溢出。代码把全部维护量放在模 2^64 的环中：unsigned long long 的加乘溢出有明确定义，所有上述多项式等式仍成立。查询结果既然保证在有符号64位范围，其模 2^64 余数就唯一决定真实答案；最高位为一时按负数的绝对值输出。这也处理先多次放大、再乘零、最后查询等情况。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct AffineTree {
    using U = unsigned long long;
    struct Node {
        U sum = 0, squares = 0, multiply = 1, add = 0;
    };
    int n;
    U modulus;
    vector<Node> tree;
    AffineTree(const vector<U> &a, U mod) : n((int)a.size() - 1), modulus(mod), tree(4 * n + 4) {
        build(1, 1, n, a);
    }
    U plus(U a, U b) const { return modulus ? (a + b) % modulus : a + b; }
    U times(U a, U b) const { return modulus ? a * b % modulus : a * b; }
    U length(int len) const { return modulus ? (U)len % modulus : (U)len; }
    void pull(int p) {
        tree[p].sum = plus(tree[p * 2].sum, tree[p * 2 + 1].sum);
        tree[p].squares = plus(tree[p * 2].squares, tree[p * 2 + 1].squares);
    }
    void build(int p, int l, int r, const vector<U> &a) {
        if (l == r) {
            tree[p].sum = a[l];
            tree[p].squares = times(a[l], a[l]);
            return;
        }
        int mid = (l + r) / 2;
        build(p * 2, l, mid, a);
        build(p * 2 + 1, mid + 1, r, a);
        pull(p);
    }
    void apply(int p, int len, U a, U b) {
        U old = tree[p].sum;
        tree[p].squares =
            plus(plus(times(times(a, a), tree[p].squares), times(times(times(2, a), b), old)),
                 times(times(b, b), length(len)));
        tree[p].sum = plus(times(a, old), times(b, length(len)));
        tree[p].multiply = times(tree[p].multiply, a);
        tree[p].add = plus(times(tree[p].add, a), b);
    }
    void push(int p, int l, int r) {
        if (tree[p].multiply == 1 && tree[p].add == 0)
            return;
        int mid = (l + r) / 2;
        apply(p * 2, mid - l + 1, tree[p].multiply, tree[p].add);
        apply(p * 2 + 1, r - mid, tree[p].multiply, tree[p].add);
        tree[p].multiply = 1;
        tree[p].add = 0;
    }
    void change(int p, int l, int r, int ql, int qr, U a, U b) {
        if (ql <= l && r <= qr) {
            apply(p, r - l + 1, a, b);
            return;
        }
        push(p, l, r);
        int mid = (l + r) / 2;
        if (ql <= mid)
            change(p * 2, l, mid, ql, qr, a, b);
        if (qr > mid)
            change(p * 2 + 1, mid + 1, r, ql, qr, a, b);
        pull(p);
    }
    pair<U, U> query(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr)
            return {tree[p].sum, tree[p].squares};
        push(p, l, r);
        int mid = (l + r) / 2;
        pair<U, U> result{0, 0};
        if (ql <= mid)
            result = query(p * 2, l, mid, ql, qr);
        if (qr > mid) {
            auto other = query(p * 2 + 1, mid + 1, r, ql, qr);
            result.first = plus(result.first, other.first);
            result.second = plus(result.second, other.second);
        }
        return result;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    using U = unsigned long long;
    int n, m;
    cin >> n >> m;
    vector<U> a(n + 1);
    for (int i = 1; i <= n; i++) {
        long long x;
        cin >> x;
        a[i] = (U)x;
    }
    AffineTree tree(a, 0);
    while (m--) {
        int op, l, r;
        cin >> op >> l >> r;
        if (op <= 2) {
            auto result = tree.query(1, 1, n, l, r);
            U value = op == 1 ? result.first : result.second;
            if (value <= (U)LLONG_MAX)
                cout << value << '\n';
            else
                cout << '-' << (~value + 1ULL) << '\n';
        } else {
            long long x;
            cin >> x;
            if (op == 3)
                tree.change(1, 1, n, l, r, (U)x, 0);
            else
                tree.change(1, 1, n, l, r, 1, (U)x);
        }
    }
    return 0;
}
```

### 复杂度

每个操作 $O(\log n)$，空间 $O(n)$。

### 易错点

有符号整数溢出在 C++ 中不可依赖；这里刻意使用无符号环，最终再根据题目保证还原有符号值。此技巧只因查询输出范围已知且运算全是整数多项式才成立。
