## 小A的最短路

### 题意

树上另有一条免费缆车边，批量求两点最少步行距离。

### 思路

最短路线若不用缆车，代价为树距离dist(x,y)；若使用，因为树边正权，免费缆车最多有必要用一次，分别考虑x→U、V→y以及x→V、U→y。取三个代价的最小值。树距离由深度和LCA得到，倍增预处理后每次查询对数时间。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct EulerTree {
    int n;
    vector<vector<int>> edge;
    vector<int> parent, depth, in, out, order;
    EulerTree(int n) : n(n), edge(n + 1), parent(n + 1), depth(n + 1), in(n + 1), out(n + 1) {}
    void add(int u, int v) {
        edge[u].push_back(v);
        edge[v].push_back(u);
    }
    void root(int start) {
        vector<int> stack = {start};
        parent[start] = 0;
        order.clear();
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            in[u] = order.size() + 1;
            order.push_back(u);
            for (int v : edge[u])
                if (v != parent[u]) {
                    parent[v] = u;
                    depth[v] = depth[u] + 1;
                    stack.push_back(v);
                }
        }
        vector<int> size(n + 1, 1);
        for (int i = (int)order.size() - 1; i >= 0; i--) {
            int u = order[i];
            out[u] = in[u] + size[u] - 1;
            if (parent[u])
                size[parent[u]] += size[u];
        }
    }
};

struct Ancestors {
    const EulerTree &tree;
    vector<vector<int>> up;
    Ancestors(const EulerTree &tree) : tree(tree) {
        int levels = 1;
        while ((1LL << levels) <= tree.n)
            levels++;
        up.assign(levels, vector<int>(tree.n + 1));
        up[0] = tree.parent;
        for (int b = 1; b < levels; b++)
            for (int u = 1; u <= tree.n; u++)
                up[b][u] = up[b - 1][up[b - 1][u]];
    }
    int lca(int u, int v) const {
        if (tree.depth[u] < tree.depth[v])
            swap(u, v);
        int gap = tree.depth[u] - tree.depth[v];
        for (int b = 0; b < (int)up.size(); b++)
            if (gap >> b & 1)
                u = up[b][u];
        if (u == v)
            return u;
        for (int b = (int)up.size() - 1; b >= 0; b--)
            if (up[b][u] != up[b][v]) {
                u = up[b][u];
                v = up[b][v];
            }
        return up[0][u];
    }
    int distance(int u, int v) const {
        return tree.depth[u] + tree.depth[v] - 2 * tree.depth[lca(u, v)];
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    EulerTree tree(n);
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        tree.add(u, v);
    }
    tree.root(1);
    Ancestors ancestor(tree);
    int u, v, q;
    cin >> u >> v >> q;
    while (q--) {
        int x, y;
        cin >> x >> y;
        cout << min({ancestor.distance(x, y), ancestor.distance(x, u) + ancestor.distance(v, y),
                     ancestor.distance(x, v) + ancestor.distance(u, y)})
             << '\n';
    }
    return 0;
}
```

### 复杂度

预处理 $O(n\log n)$，每问 $O(\log n)$，空间 $O(n\log n)$。

### 易错点

缆车双向且代价为0，不要额外加1。迭代遍历避免长链递归栈溢出。
