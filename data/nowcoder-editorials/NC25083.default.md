## Dining

### 题意

每牛选一份喜欢的食物与饮料，最大满足牛数。

### 思路

建源→食物→牛入点→牛出点→饮料→汇。食物、饮料各只能被用一次，源边及汇边容量1；牛拆点边容量1保证最多领一份完整餐；喜好边连通相应食物与饮料。每条单位流都对应一头得到完整餐的牛，最大流即答案。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Dinic {
    struct Edge {
        int to, reverse;
        long long capacity;
    };
    vector<vector<Edge>> graph;
    vector<int> level, current;
    explicit Dinic(int n) : graph(n), level(n), current(n) {}
    void resize(int n) {
        graph.resize(n);
        level.resize(n);
        current.resize(n);
    }
    int add(int u, int v, long long capacity) {
        int index = graph[u].size(), reverse = graph[v].size();
        graph[u].push_back({v, reverse, capacity});
        graph[v].push_back({u, index, 0});
        return index;
    }
    bool bfs(int source, int sink) {
        fill(level.begin(), level.end(), -1);
        vector<int> queue = {source};
        level[source] = 0;
        for (size_t i = 0; i < queue.size(); i++)
            for (auto e : graph[queue[i]])
                if (e.capacity > 0 && level[e.to] < 0) {
                    level[e.to] = level[queue[i]] + 1;
                    queue.push_back(e.to);
                }
        return level[sink] >= 0;
    }
    long long send(int u, int sink, long long amount) {
        if (u == sink)
            return amount;
        for (int &i = current[u]; i < (int)graph[u].size(); i++) {
            auto &e = graph[u][i];
            if (e.capacity > 0 && level[e.to] == level[u] + 1) {
                long long value = send(e.to, sink, min(amount, e.capacity));
                if (value) {
                    e.capacity -= value;
                    graph[e.to][e.reverse].capacity += value;
                    return value;
                }
            }
        }
        return 0;
    }
    long long maximum(int source, int sink, long long limit = LLONG_MAX / 4) {
        long long total = 0;
        while (total < limit && bfs(source, sink)) {
            fill(current.begin(), current.end(), 0);
            while (total < limit) {
                long long value = send(source, sink, limit - total);
                if (!value)
                    break;
                total += value;
            }
        }
        return total;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, f, d;
    cin >> n >> f >> d;
    int cowIn = f, cowOut = f + n, drink = f + 2 * n, source = drink + d, sink = source + 1;
    Dinic flow(sink + 1);
    for (int i = 0; i < f; i++)
        flow.add(source, i, 1);
    for (int j = 0; j < d; j++)
        flow.add(drink + j, sink, 1);
    for (int i = 0; i < n; i++) {
        int a, b;
        cin >> a >> b;
        flow.add(cowIn + i, cowOut + i, 1);
        while (a--) {
            int x;
            cin >> x;
            flow.add(x - 1, cowIn + i, 1);
        }
        while (b--) {
            int x;
            cin >> x;
            flow.add(cowOut + i, drink + x - 1, 1);
        }
    }
    cout << flow.maximum(source, sink);
    return 0;
}
```

### 复杂度

单位容量网络Dinic为 $O(E\sqrt V)$ 级界，空间 $O(V+E)$。

### 易错点

不能把食物、饮料分别独立最大匹配，因为两边可能满足不同的牛。拆点把两种选择绑定。
