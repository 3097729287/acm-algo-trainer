## Typewriter

### 题意

付p键入一字符或付q复制已经输出的子串，最小打印费用。

### 思路

后缀自动机把结束位置集合相同的子串合并为一个状态。状态u代表的长度恰为(len[link[u]],len[u]]，这些子串出现次数相同。每加入一个字符，建立当前完整前缀的状态并令初始次数为1；必要时克隆旧状态，分离较短的结束位置类，克隆初始次数为0。按len降序把次数加到link，得到每个状态的真实出现次数；link长度严格减小，所以可以用计数排序得到传播顺序。

先对整串建SAM，记录每个状态的最小结束位置minEnd：每个前缀状态加入自身位置，克隆初始无穷，再按len降序向link取最小。对于以i结束、长L的后缀，能从已经输出的前缀复制，当且仅当其最早出现结束点≤i−L。

沿prefix[i]的link祖先链，越短越容易满足。用倍增跳过“minEnd[u]+len[u]>i”的状态，停在第一个临界状态u；它的父亲长度整段可复制，而u自身区间能复制到min(len[u],i−minEnd[u])，二者最大值就是最长不重叠可复制后缀L。

设dp[i]为前i字最小费用，dp[i]=min(dp[i−1]+p,dp[i−L]+q)。只用最长L足够，因为dp不减：把某个打印方案截短，最后一次复制也可截成更短的已出现子串，费用不会增加。因此同费用复制越长不会更差。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct SuffixAutomaton {
    struct Node {
        array<int, 26> next = {};
        int link = -1, length = 0, count = 0;
    };
    vector<Node> node;
    vector<int> prefix, order;
    int last = 0;
    explicit SuffixAutomaton(const string &s) {
        node.reserve(2 * s.size() + 1);
        node.emplace_back();
        prefix.reserve(s.size() + 1);
        prefix.push_back(0);
        for (char ch : s) {
            int c = ch - 'a', current = node.size();
            node.emplace_back();
            node[current].length = node[last].length + 1;
            node[current].count = 1;
            int p = last;
            while (p != -1 && !node[p].next[c]) {
                node[p].next[c] = current;
                p = node[p].link;
            }
            if (p == -1)
                node[current].link = 0;
            else {
                int q = node[p].next[c];
                if (node[p].length + 1 == node[q].length)
                    node[current].link = q;
                else {
                    int clone = node.size();
                    node.push_back(node[q]);
                    node[clone].length = node[p].length + 1;
                    node[clone].count = 0;
                    while (p != -1 && node[p].next[c] == q) {
                        node[p].next[c] = clone;
                        p = node[p].link;
                    }
                    node[q].link = node[current].link = clone;
                }
            }
            last = current;
            prefix.push_back(last);
        }
        vector<int> count(s.size() + 1);
        for (auto &v : node)
            count[v.length]++;
        for (size_t i = 1; i < count.size(); i++)
            count[i] += count[i - 1];
        order.resize(node.size());
        for (int u = (int)node.size() - 1; u >= 0; u--)
            order[--count[node[u].length]] = u;
        for (int i = (int)order.size() - 1; i > 0; i--) {
            int u = order[i];
            node[node[u].link].count += node[u].count;
        }
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string s;
    long long p, q;
    cin >> s >> p >> q;
    int n = s.size();
    SuffixAutomaton sam(s);
    int size = sam.node.size(), levels = 1;
    while ((1LL << levels) <= size)
        levels++;
    vector<int> minimum(size, n + 1);
    for (int i = 1; i <= n; i++)
        minimum[sam.prefix[i]] = min(minimum[sam.prefix[i]], i);
    for (int i = (int)sam.order.size() - 1; i > 0; i--) {
        int u = sam.order[i];
        minimum[sam.node[u].link] = min(minimum[sam.node[u].link], minimum[u]);
    }
    vector<vector<int>> up(levels, vector<int>(size));
    for (int u = 1; u < size; u++)
        up[0][u] = sam.node[u].link;
    for (int b = 1; b < levels; b++)
        for (int u = 0; u < size; u++)
            up[b][u] = up[b - 1][up[b - 1][u]];
    vector<long long> dp(n + 1);
    for (int i = 1; i <= n; i++) {
        int u = sam.prefix[i];
        for (int b = levels - 1; b >= 0; b--) {
            int v = up[b][u];
            if (v && minimum[v] + sam.node[v].length > i)
                u = v;
        }
        int parent = sam.node[u].link,
            length = max(sam.node[parent].length, min(sam.node[u].length, i - minimum[u]));
        dp[i] = dp[i - 1] + p;
        if (length > 0)
            dp[i] = min(dp[i], dp[i - length] + q);
    }
    cout << dp[n];
    return 0;
}
```

### 复杂度

时间 $O(n\log n)$，空间 $O(26n+n\log n)$。

### 易错点

复制源必须完整位于此次粘贴之前，不允许一边粘贴一边借用新输出字符。用minEnd条件排除重叠。
