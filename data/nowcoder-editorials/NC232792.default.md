## 模拟人生2077

### 题意

求任取三点的不信任度最小值和最大值，即二倍三角形面积。

### 思路

不信任度是两条边向量行列式的绝对值，等于二倍三角形面积，最后不再除2。最大值只需在凸包上找；固定第一个顶点，第二个顶点前进时，第三个顶点的最大面积位置也单调前进，得到O(h²)。

最小值不能只看一个原点极角相邻的点。考虑旋转一条投影法线，维护所有点的投影顺序：两个点A、B投影相同时，扫描方向平行AB，两点交换次序。此时距离AB直线最近的第三点，只可能是这对相邻点在顺序外的两个邻居。把所有点对方向归一到[0,π)，按叉积排序，依次检查两个外邻居并交换两端；所有三角形的某条底边都会作为事件出现，因此覆盖最小值。

初始顺序是y升序、同y时x升序，对应角度0稍前。若事件端点不相邻，说明两点之间有第三点投影相等，即三点共线，最小值立即为0；邻居叉积为0同理。平行但不共线的点对交换互不影响。重复点也直接得到最小值0。点对事件用两个16位编号减少内存，方向用32位，面积用128位。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

using Wide = __int128_t;
struct IPoint {
    long long x = 0, y = 0;
    IPoint operator+(IPoint b) const { return {x + b.x, y + b.y}; }
    IPoint operator-(IPoint b) const { return {x - b.x, y - b.y}; }
};
Wide cross(IPoint a, IPoint b) { return (Wide)a.x * b.y - (Wide)a.y * b.x; }
Wide dot(IPoint a, IPoint b) { return (Wide)a.x * b.x + (Wide)a.y * b.y; }
Wide absolute(Wide x) { return x < 0 ? -x : x; }
void printWide(Wide value) {
    if (value < 0) {
        cout << '-';
        value = -value;
    }
    string text;
    do {
        text.push_back('0' + value % 10);
        value /= 10;
    } while (value);
    reverse(text.begin(), text.end());
    cout << text;
}
bool angleLess(IPoint a, IPoint b) {
    auto half = [](IPoint p) { return p.y < 0 || (p.y == 0 && p.x < 0); };
    if (half(a) != half(b))
        return half(a) < half(b);
    return cross(a, b) > 0;
}
vector<IPoint> integerHull(vector<IPoint> point) {
    sort(point.begin(), point.end(),
         [](auto a, auto b) { return a.x != b.x ? a.x < b.x : a.y < b.y; });
    point.erase(
        unique(point.begin(), point.end(), [](auto a, auto b) { return a.x == b.x && a.y == b.y; }),
        point.end());
    if (point.size() < 2)
        return point;
    vector<IPoint> hull;
    for (auto p : point) {
        while (hull.size() >= 2 && cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    size_t lower = hull.size();
    for (int i = (int)point.size() - 2; i >= 0; i--) {
        auto p = point[i];
        while (hull.size() > lower &&
               cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    hull.pop_back();
    return hull;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<IPoint> p(n);
    for (auto &q : p)
        cin >> q.x;
    for (auto &q : p)
        cin >> q.y;
    auto hull = integerHull(p);
    Wide maximum = 0;
    int h = hull.size();
    for (int i = 0; i < h; i++) {
        int k = i + 2;
        for (int j = i + 1; j + 1 < h; j++) {
            k = max(k, j + 1);
            auto area = [&](int z) { return cross(hull[j] - hull[i], hull[z] - hull[i]); };
            while (k + 1 < h && area(k + 1) > area(k))
                k++;
            maximum = max(maximum, area(k));
        }
    }
    vector<int> order(n), position(n);
    iota(order.begin(), order.end(), 0);
    sort(order.begin(), order.end(),
         [&](int a, int b) { return p[a].y != p[b].y ? p[a].y < p[b].y : p[a].x < p[b].x; });
    bool duplicate = false;
    for (int i = 0; i < n; i++) {
        position[order[i]] = i;
        if (i && p[order[i]].x == p[order[i - 1]].x && p[order[i]].y == p[order[i - 1]].y)
            duplicate = true;
    }
    Wide minimum = maximum;
    if (!duplicate && minimum) {
        struct Event {
            int dx, dy;
            uint16_t u, v;
        };
        vector<Event> events;
        events.reserve((long long)n * (n - 1) / 2);
        for (int i = 0; i < n; i++)
            for (int j = i + 1; j < n; j++) {
                IPoint d = p[j] - p[i];
                if (d.y < 0 || (d.y == 0 && d.x < 0)) {
                    d.x = -d.x;
                    d.y = -d.y;
                }
                events.push_back({(int)d.x, (int)d.y, (uint16_t)i, (uint16_t)j});
            }
        sort(events.begin(), events.end(), [](const Event &a, const Event &b) {
            long long c = (long long)a.dx * b.dy - (long long)a.dy * b.dx;
            if (c)
                return c > 0;
            return a.u != b.u ? a.u < b.u : a.v < b.v;
        });
        for (auto e : events) {
            int u = position[e.u], v = position[e.v];
            if (u > v)
                swap(u, v);
            if (v - u != 1) {
                minimum = 0;
                break;
            }
            for (int at : {u - 1, v + 1})
                if (at >= 0 && at < n)
                    minimum = min(minimum, absolute(cross(p[e.v] - p[e.u], p[order[at]] - p[e.u])));
            if (!minimum)
                break;
            swap(order[u], order[v]);
            position[order[u]] = u;
            position[order[v]] = v;
        }
    } else if (duplicate)
        minimum = 0;
    printWide(minimum);
    cout << '\n';
    printWide(maximum);
    return 0;
}
```

### 复杂度

时间 $O(n^2\log n)$，空间 $O(n^2)$；事件12字节，n=3000时约54MB。

### 易错点

点对方向归一后都位于同一个半平面，叉积排序才有严格顺序。不能用atan2相等判三点共线，也不能用随机抽样代替最小值证明。
