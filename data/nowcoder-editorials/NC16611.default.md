## [NOIP2009]最优贸易

### 题意

从1走到n，可重复经过城市，至多买卖一次，最大化卖价减买价。

### 思路

将有向图缩成强连通分量DAG。每个分量内可以任意来回，记录最低、最高商品价。拓扑传播buy[c]：从1到该分量的所有路线中，能遇到的最低买价，进入分量时与该分量最低价取小。

再在缩点反图从n的分量搜索，标记能继续到终点的分量。对既可由1到达又能到n的每个分量，尝试在此分量最高价处卖出，收益maxPrice[c]−buy[c]，全局至少0。SCC内能先到最低价点再到最高价点，所以使用分量极值是可实现的，重复经过城市的情况也被包含。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

pair<int, vector<int>> components(const vector<vector<int>> &graph,
                                  const vector<vector<int>> &reverse) {
    int n = graph.size();
    vector<bool> seen(n);
    vector<int> finish;
    for (int start = 0; start < n; start++)
        if (!seen[start]) {
            vector<pair<int, int>> stack{{start, 0}};
            seen[start] = true;
            while (!stack.empty()) {
                int u = stack.back().first, &position = stack.back().second;
                if (position == (int)graph[u].size()) {
                    finish.push_back(u);
                    stack.pop_back();
                } else {
                    int v = graph[u][position++];
                    if (!seen[v]) {
                        seen[v] = true;
                        stack.push_back({v, 0});
                    }
                }
            }
        }
    vector<int> component(n, -1);
    int count = 0;
    for (int i = n - 1; i >= 0; i--) {
        int start = finish[i];
        if (component[start] != -1)
            continue;
        vector<int> stack{start};
        component[start] = count;
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            for (int v : reverse[u])
                if (component[v] == -1) {
                    component[v] = count;
                    stack.push_back(v);
                }
        }
        count++;
    }
    return {count, component};
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    vector<int> price(n);
    for (int &x : price)
        cin >> x;
    vector<vector<int>> graph(n), reverse(n);
    auto addEdge = [&](int a, int b) {
        graph[a].push_back(b);
        reverse[b].push_back(a);
    };
    while (m--) {
        int a, b, type;
        cin >> a >> b >> type;
        --a;
        --b;
        addEdge(a, b);
        if (type == 2)
            addEdge(b, a);
    }
    auto [count, component] = components(graph, reverse);
    vector<int> low(count, 101), high(count), degree(count);
    vector<vector<int>> dag(count), back(count);
    for (int u = 0; u < n; u++) {
        int a = component[u];
        low[a] = min(low[a], price[u]);
        high[a] = max(high[a], price[u]);
        for (int v : graph[u])
            if (a != component[v]) {
                dag[a].push_back(component[v]);
                back[component[v]].push_back(a);
                degree[component[v]]++;
            }
    }
    vector<bool> reachesTarget(count);
    vector<int> queue{component[n - 1]};
    reachesTarget[queue[0]] = true;
    for (size_t h = 0; h < queue.size(); h++)
        for (int v : back[queue[h]])
            if (!reachesTarget[v]) {
                reachesTarget[v] = true;
                queue.push_back(v);
            }
    queue.clear();
    for (int c = 0; c < count; c++)
        if (!degree[c])
            queue.push_back(c);
    vector<int> buy(count, 101);
    buy[component[0]] = low[component[0]];
    int answer = 0;
    for (size_t h = 0; h < queue.size(); h++) {
        int u = queue[h];
        if (buy[u] != 101 && reachesTarget[u])
            answer = max(answer, high[u] - buy[u]);
        for (int v : dag[u]) {
            if (buy[u] != 101)
                buy[v] = min(buy[v], min(buy[u], low[v]));
            if (--degree[v] == 0)
                queue.push_back(v);
        }
    }
    cout << answer;
    return 0;
}
```

### 复杂度

时间、空间均为 $O(n+m)$。

### 易错点

买卖点必须按可行路线先后到达，不能只在所有可达城市里取全局最低最高价。卖出后还必须能到n。强连通搜索用显式栈处理长链。
