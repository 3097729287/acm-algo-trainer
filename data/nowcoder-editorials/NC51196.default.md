## Cut the Sequence

### 题意

切成若干连续段，每段和≤M，最小化每段最大值之和。

### 思路

设dp[i]为前i项最优值。末段在j后开始，转移为dp[j]+max(a[j+1..i])，并要求末段和≤M。非负数使合法j形成一段后缀，可双指针维护最早末段起点L。

在段树的叶子j维护这个转移候选。追加a[i]时，先将新叶i−1置为dp[i−1]+a[i]。旧候选的最大值如何变化？用值递减的单调栈保存当前后缀最大值的分段。弹出位置p时，弹后栈顶q（无则0），原来以a[p]为最大值的分界j恰为[q,p−1]，这些叶统一增加a[i]−a[p]。弹完后压i，在允许分界区间[L−1,i−1]查询最小，即得dp[i]。每个下标只弹一次，总更新线性次，每次段树对数时间。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct RangeMinimum {
    int n;
    const long long inf = 1LL << 60;
    vector<long long> value, lazy;
    explicit RangeMinimum(int n) : n(n), value(4 * n, 1LL << 60), lazy(4 * n) {}
    void apply(int p, long long delta) {
        value[p] += delta;
        lazy[p] += delta;
    }
    void push(int p) {
        if (lazy[p]) {
            apply(p * 2, lazy[p]);
            apply(p * 2 + 1, lazy[p]);
            lazy[p] = 0;
        }
    }
    void assign(int p, int l, int r, int position, long long x) {
        if (l == r) {
            value[p] = x;
            lazy[p] = 0;
            return;
        }
        push(p);
        int mid = (l + r) / 2;
        if (position <= mid)
            assign(p * 2, l, mid, position, x);
        else
            assign(p * 2 + 1, mid + 1, r, position, x);
        value[p] = min(value[p * 2], value[p * 2 + 1]);
    }
    void add(int p, int l, int r, int ql, int qr, long long delta) {
        if (ql <= l && r <= qr) {
            apply(p, delta);
            return;
        }
        push(p);
        int mid = (l + r) / 2;
        if (ql <= mid)
            add(p * 2, l, mid, ql, qr, delta);
        if (qr > mid)
            add(p * 2 + 1, mid + 1, r, ql, qr, delta);
        value[p] = min(value[p * 2], value[p * 2 + 1]);
    }
    long long query(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr)
            return value[p];
        push(p);
        int mid = (l + r) / 2;
        long long answer = inf;
        if (ql <= mid)
            answer = min(answer, query(p * 2, l, mid, ql, qr));
        if (qr > mid)
            answer = min(answer, query(p * 2 + 1, mid + 1, r, ql, qr));
        return answer;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    long long limit;
    cin >> n >> limit;
    vector<long long> a(n + 1), prefix(n + 1), dp(n + 1);
    bool possible = true;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        prefix[i] = prefix[i - 1] + a[i];
        if (a[i] > limit)
            possible = false;
    }
    if (!possible) {
        cout << -1;
        return 0;
    }
    RangeMinimum tree(n);
    vector<int> stack;
    int left = 1;
    for (int i = 1; i <= n; i++) {
        while (!stack.empty() && a[stack.back()] <= a[i]) {
            int p = stack.back();
            stack.pop_back();
            int q = stack.empty() ? 0 : stack.back();
            tree.add(1, 0, n - 1, q, p - 1, a[i] - a[p]);
        }
        stack.push_back(i);
        tree.assign(1, 0, n - 1, i - 1, dp[i - 1] + a[i]);
        while (prefix[i] - prefix[left - 1] > limit)
            left++;
        dp[i] = tree.query(1, 0, n - 1, left - 1, i - 1);
    }
    cout << dp[n];
    return 0;
}
```

### 复杂度

时间 $O(n\log n)$，空间 $O(n)$。

### 易错点

单个元素>M立即无解。M、前缀和、DP用64位；数组含0仍保持窗口合法性和单调栈分段。栈更新的是所有分界候选，不能只维护一个当前最大值。
