## 区间加区间sin和

### 题意

区间整数角度加，查询正弦之和。

### 思路

同时维护S=Σsin(a[i])和C=Σcos(a[i])。加角v后，由和角公式S新=S*cos(v)+C*sin(v)，C新=C*cos(v)−S*sin(v)。这是一组整体旋转，因此区间加可以累积成整数角度懒标记。标记保留整数，向子节点应用时再计算三角函数，减少逐次角度浮点累加误差。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<long double> sine(4 * n + 4), cosine(4 * n + 4);
    vector<long long> tag(4 * n + 4);
    auto build = [&](auto &&self, int u, int l, int r) -> void {
        if (l == r) {
            long long x;
            cin >> x;
            sine[u] = sinl(x);
            cosine[u] = cosl(x);
            return;
        }
        int mid = (l + r) / 2;
        self(self, u * 2, l, mid);
        self(self, u * 2 + 1, mid + 1, r);
        sine[u] = sine[u * 2] + sine[u * 2 + 1];
        cosine[u] = cosine[u * 2] + cosine[u * 2 + 1];
    };
    build(build, 1, 1, n);
    auto apply = [&](int u, long long v) {
        long double s = sinl(v), c = cosl(v), old = sine[u];
        sine[u] = old * c + cosine[u] * s;
        cosine[u] = cosine[u] * c - old * s;
        tag[u] += v;
    };
    auto push = [&](int u) {
        if (tag[u]) {
            apply(u * 2, tag[u]);
            apply(u * 2 + 1, tag[u]);
            tag[u] = 0;
        }
    };
    auto update = [&](auto &&self, int u, int l, int r, int ql, int qr, long long v) -> void {
        if (ql <= l && r <= qr) {
            apply(u, v);
            return;
        }
        push(u);
        int mid = (l + r) / 2;
        if (ql <= mid)
            self(self, u * 2, l, mid, ql, qr, v);
        if (qr > mid)
            self(self, u * 2 + 1, mid + 1, r, ql, qr, v);
        sine[u] = sine[u * 2] + sine[u * 2 + 1];
        cosine[u] = cosine[u * 2] + cosine[u * 2 + 1];
    };
    auto query = [&](auto &&self, int u, int l, int r, int ql, int qr) -> long double {
        if (ql <= l && r <= qr)
            return sine[u];
        push(u);
        int mid = (l + r) / 2;
        long double result = 0;
        if (ql <= mid)
            result += self(self, u * 2, l, mid, ql, qr);
        if (qr > mid)
            result += self(self, u * 2 + 1, mid + 1, r, ql, qr);
        return result;
    };
    int m;
    cin >> m;
    cout << fixed << setprecision(10);
    while (m--) {
        int op, l, r;
        cin >> op >> l >> r;
        if (op == 1) {
            long long v;
            cin >> v;
            update(update, 1, 1, n, l, r, v);
        } else
            cout << query(query, 1, 1, n, l, r) << '\n';
    }
    return 0;
}
```

### 复杂度

构建 $O(n)$，每次操作 $O(\log n)$，空间 $O(n)$。

### 易错点

题目角度按弧度代入sin，不转换为度。更新两个统计量时保存旧S，避免使用已更新值。
