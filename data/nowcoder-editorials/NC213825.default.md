## [网络流24题]星际转移问题

### 题意

周期飞船运输k人从地球到月球，最短完成时间。

### 思路

先把每艘正容量飞船所停站点用并查集合并。允许任意等待，若地球与月球不在同一连通分量就永远无法转移，输出0；否则等待合适周期总能依次中转。

时间t建立每站副本，上一层站点到本层同站连容量k等待边；每艘飞船按其(t−1)模周期到t模周期的实际站点连容量H运输边。源只连t=0的地球，所有时刻月球连汇。逐天扩图，在已有残量网络上继续增广，首次总流达到k的天数即答案。等待边和运输边精确表示每人每分钟的合法选择。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct DSU {
    vector<int> parent, size;
    DSU(int n) : parent(n + 1), size(n + 1, 1) { iota(parent.begin(), parent.end(), 0); }
    int find(int x) {
        while (parent[x] != x) {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        return x;
    }
    bool unite(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b)
            return false;
        if (size[a] < size[b])
            swap(a, b);
        parent[b] = a;
        size[a] += size[b];
        return true;
    }
};

struct Dinic {
    struct Edge {
        int to, reverse;
        long long capacity;
    };
    vector<vector<Edge>> graph;
    vector<int> level, current;
    explicit Dinic(int n) : graph(n), level(n), current(n) {}
    void resize(int n) {
        graph.resize(n);
        level.resize(n);
        current.resize(n);
    }
    int add(int u, int v, long long capacity) {
        int index = graph[u].size(), reverse = graph[v].size();
        graph[u].push_back({v, reverse, capacity});
        graph[v].push_back({u, index, 0});
        return index;
    }
    bool bfs(int source, int sink) {
        fill(level.begin(), level.end(), -1);
        vector<int> queue = {source};
        level[source] = 0;
        for (size_t i = 0; i < queue.size(); i++)
            for (auto e : graph[queue[i]])
                if (e.capacity > 0 && level[e.to] < 0) {
                    level[e.to] = level[queue[i]] + 1;
                    queue.push_back(e.to);
                }
        return level[sink] >= 0;
    }
    long long send(int u, int sink, long long amount) {
        if (u == sink)
            return amount;
        for (int &i = current[u]; i < (int)graph[u].size(); i++) {
            auto &e = graph[u][i];
            if (e.capacity > 0 && level[e.to] == level[u] + 1) {
                long long value = send(e.to, sink, min(amount, e.capacity));
                if (value) {
                    e.capacity -= value;
                    graph[e.to][e.reverse].capacity += value;
                    return value;
                }
            }
        }
        return 0;
    }
    long long maximum(int source, int sink, long long limit = LLONG_MAX / 4) {
        long long total = 0;
        while (total < limit && bfs(source, sink)) {
            fill(current.begin(), current.end(), 0);
            while (total < limit) {
                long long value = send(source, sink, limit - total);
                if (!value)
                    break;
                total += value;
            }
        }
        return total;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m, k;
    cin >> n >> m >> k;
    int stations = n + 2, moon = n + 1;
    struct Ship {
        int capacity;
        vector<int> route;
    };
    vector<Ship> ship(m);
    DSU dsu(stations);
    for (auto &s : ship) {
        int r;
        cin >> s.capacity >> r;
        s.route.resize(r);
        for (int &x : s.route) {
            cin >> x;
            if (x == -1)
                x = moon;
        }
        if (s.capacity > 0)
            for (int x : s.route)
                dsu.unite(s.route[0], x);
    }
    if (dsu.find(0) != dsu.find(moon)) {
        cout << 0;
        return 0;
    }
    auto vertex = [&](int time, int station) { return 2 + time * stations + station; };
    int source = 0, sink = 1;
    Dinic flow(2 + stations);
    flow.add(source, vertex(0, 0), k);
    flow.add(vertex(0, moon), sink, k);
    long long moved = 0;
    for (int day = 1;; day++) {
        flow.resize(2 + (day + 1) * stations);
        for (int s = 0; s < stations; s++)
            flow.add(vertex(day - 1, s), vertex(day, s), k);
        for (auto &s : ship) {
            int r = s.route.size();
            flow.add(vertex(day - 1, s.route[(day - 1) % r]), vertex(day, s.route[day % r]),
                     s.capacity);
        }
        flow.add(vertex(day, moon), sink, k);
        moved += flow.maximum(source, sink, k - moved);
        if (moved == k) {
            cout << day;
            break;
        }
    }
    return 0;
}
```

### 复杂度

若答案T，V=O(T(n+2))、E=O(T(n+m))；一般增量Dinic保守上界 $O(TV²E)$，实际复用既有流且总运量k≤50；空间 $O(V+E)$。

### 易错点

时刻0飞船在路线第一个站，第一分钟边是route[0]→route[1%r]。增量流要保留反向残量，允许重新分配早先乘客路线。
