## [WF2019]First of Her Name

### 题意

每位女士姓名为新字母接母亲姓名，批量统计给定姓名前缀。

### 思路

把姓名和查询都反转：反转姓名是从家族根到该女士的边字母序列，查询前缀变为这条根路径的后缀。将全部反转查询建AC自动机。母亲编号小于女儿，按编号计算state[i]=go(state[parent[i]],c[i])，给该状态计数加1。所有女士处理后按fail逆BFS汇总，查询终点计数即有该反转后缀的根路径数，也就是原问题答案。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Aho {
    struct Node {
        array<int, 26> next = {};
        int fail = 0, parent = 0, letter = 0, longest = 0;
    };
    vector<Node> node;
    vector<int> order;
    int alphabet;
    char base;
    Aho(int alphabet, char base, size_t capacity) : alphabet(alphabet), base(base) {
        node.reserve(capacity);
        node.emplace_back();
    }
    int insert(const string &s) {
        int u = 0;
        for (char ch : s) {
            int c = ch - base;
            if (!node[u].next[c]) {
                int v = node.size();
                node[u].next[c] = v;
                node.emplace_back();
                node[v].parent = u;
                node[v].letter = c;
            }
            u = node[u].next[c];
        }
        node[u].longest = max(node[u].longest, (int)s.size());
        return u;
    }
    void build() {
        order = {0};
        for (int c = 0; c < alphabet; c++)
            if (node[0].next[c])
                order.push_back(node[0].next[c]);
        for (size_t i = 1; i < order.size(); i++) {
            int u = order[i], f = node[u].fail;
            node[u].longest = max(node[u].longest, node[f].longest);
            for (int c = 0; c < alphabet; c++) {
                int v = node[u].next[c];
                if (v) {
                    node[v].fail = node[f].next[c];
                    order.push_back(v);
                } else
                    node[u].next[c] = node[f].next[c];
            }
        }
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, k;
    cin >> n >> k;
    vector<int> parent(n + 1);
    string letter(n + 1, 'A');
    for (int i = 1; i <= n; i++)
        cin >> letter[i] >> parent[i];
    vector<string> query(k);
    size_t total = 1;
    for (auto &s : query) {
        cin >> s;
        reverse(s.begin(), s.end());
        total += s.size();
    }
    Aho ac(26, 'A', total);
    vector<int> end(k);
    for (int i = 0; i < k; i++)
        end[i] = ac.insert(query[i]);
    ac.build();
    vector<int> state(n + 1);
    vector<long long> count(ac.node.size());
    for (int i = 1; i <= n; i++) {
        state[i] = ac.node[state[parent[i]]].next[letter[i] - 'A'];
        count[state[i]]++;
    }
    for (int i = (int)ac.order.size() - 1; i > 0; i--) {
        int u = ac.order[i];
        count[ac.node[u].fail] += count[u];
    }
    for (int u : end)
        cout << count[u] << '\n';
    return 0;
}
```

### 复杂度

查询总长L，时间 $O(26L+n+k)$，空间 $O(26L+n+k)$。

### 易错点

不要显式拼接每个完整姓名，链状家族会产生平方总长度。每位女士只贡献其整条根路径的结尾一次。
