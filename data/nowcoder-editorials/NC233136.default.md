## Beauty Contest

### 题意

求最远点对距离平方，坐标可达2^31。

### 思路

最远点对都在凸包上。对每条凸包边，沿凸包移动对踵点j，直到它到该边的有向面积不再增大；j全程单调绕行，每条边只检查两端与j的距离，再在面积相等时检查j+1。两点退化直接求平方。

坐标差可能达到2^32，两个平方相加可达2^65，连无符号64位也不够。叉积、平方距离及输出全部使用128位整数。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

using Wide = __int128_t;
struct IPoint {
    long long x = 0, y = 0;
    IPoint operator+(IPoint b) const { return {x + b.x, y + b.y}; }
    IPoint operator-(IPoint b) const { return {x - b.x, y - b.y}; }
};
Wide cross(IPoint a, IPoint b) { return (Wide)a.x * b.y - (Wide)a.y * b.x; }
Wide dot(IPoint a, IPoint b) { return (Wide)a.x * b.x + (Wide)a.y * b.y; }
Wide absolute(Wide x) { return x < 0 ? -x : x; }
void printWide(Wide value) {
    if (value < 0) {
        cout << '-';
        value = -value;
    }
    string text;
    do {
        text.push_back('0' + value % 10);
        value /= 10;
    } while (value);
    reverse(text.begin(), text.end());
    cout << text;
}
bool angleLess(IPoint a, IPoint b) {
    auto half = [](IPoint p) { return p.y < 0 || (p.y == 0 && p.x < 0); };
    if (half(a) != half(b))
        return half(a) < half(b);
    return cross(a, b) > 0;
}
vector<IPoint> integerHull(vector<IPoint> point) {
    sort(point.begin(), point.end(),
         [](auto a, auto b) { return a.x != b.x ? a.x < b.x : a.y < b.y; });
    point.erase(
        unique(point.begin(), point.end(), [](auto a, auto b) { return a.x == b.x && a.y == b.y; }),
        point.end());
    if (point.size() < 2)
        return point;
    vector<IPoint> hull;
    for (auto p : point) {
        while (hull.size() >= 2 && cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    size_t lower = hull.size();
    for (int i = (int)point.size() - 2; i >= 0; i--) {
        auto p = point[i];
        while (hull.size() > lower &&
               cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    hull.pop_back();
    return hull;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<IPoint> p(n);
    for (auto &a : p)
        cin >> a.x >> a.y;
    auto h = integerHull(p);
    int m = h.size();
    Wide answer = 0;
    auto squared = [](IPoint a, IPoint b) { return dot(a - b, a - b); };
    if (m == 2)
        answer = squared(h[0], h[1]);
    else if (m > 2) {
        int j = 1;
        for (int i = 0; i < m; i++) {
            int next = (i + 1) % m;
            auto area = [&](int k) { return absolute(cross(h[next] - h[i], h[k] - h[i])); };
            while (area((j + 1) % m) > area(j))
                j = (j + 1) % m;
            answer = max({answer, squared(h[i], h[j]), squared(h[next], h[j])});
            if (area((j + 1) % m) == area(j))
                answer =
                    max({answer, squared(h[i], h[(j + 1) % m]), squared(h[next], h[(j + 1) % m])});
        }
    }
    printWide(answer);
    return 0;
}
```

### 复杂度

时间 $O(n\log n)$，卡壳 $O(n)$，空间 $O(n)$。

### 易错点

输出平方，无需开方。求凸包时也使用128位叉积，否则极大坐标的边界就可能先算错。
