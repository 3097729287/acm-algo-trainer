## 智乃酱的双塔问题·改

### 题意

两塔斜楼梯方向可改，查询从指定层与侧到更高层指定侧的上楼方案数。

### 思路

每层对应一个二乘二计数转移矩阵，同塔两条边的转移数为一；若斜楼梯为 /，再加左到右的一；若为反斜线，则加右到左的一。两段路径方案数按普通矩阵乘法组合：枚举中间侧，前后方案数相乘再相加。线段树维护有序矩阵积，修改只换一个叶子。查询 hs 到 ht 使用楼层 hs 到 ht-1 的矩阵，取起点侧、终点侧对应元素。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

const long long Empty = 0;
struct Matrix {
    long long value[2][2];
    Matrix() {
        for (auto &row : value)
            for (auto &x : row)
                x = Empty;
    }
};
Matrix identity() {
    Matrix a;
    a.value[0][0] = a.value[1][1] = 1;
    return a;
}
Matrix product(const Matrix &a, const Matrix &b) {
    Matrix result;
    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
            for (int k = 0; k < 2; k++) {
                result.value[i][j] =
                    (result.value[i][j] + a.value[i][k] * b.value[k][j]) % 1000000007;
            }
    return result;
}
struct MatrixTree {
    int base = 1;
    vector<Matrix> tree;
    explicit MatrixTree(const vector<Matrix> &a) {
        while (base < (int)a.size())
            base *= 2;
        tree.assign(base * 2, identity());
        for (int i = 0; i < (int)a.size(); i++)
            tree[base + i] = a[i];
        for (int i = base - 1; i > 0; i--)
            tree[i] = product(tree[i * 2], tree[i * 2 + 1]);
    }
    void change(int p, Matrix a) {
        p += base;
        tree[p] = a;
        while (p > 1) {
            p /= 2;
            tree[p] = product(tree[p * 2], tree[p * 2 + 1]);
        }
    }
    Matrix query(int l, int r) {
        Matrix left = identity(), right = identity();
        for (l += base, r += base; l < r; l /= 2, r /= 2) {
            if (l & 1)
                left = product(left, tree[l++]);
            if (r & 1)
                right = product(tree[--r], right);
        }
        return product(left, right);
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    string direction;
    cin >> n >> m >> direction;
    auto leaf = [&](char c) {
        Matrix a;
        a.value[0][0] = a.value[1][1] = 1;
        if (c == '/')
            a.value[0][1] = 1;
        else
            a.value[1][0] = 1;
        return a;
    };
    vector<Matrix> initial;
    initial.reserve(n - 1);
    for (char c : direction)
        initial.push_back(leaf(c));
    MatrixTree tree(initial);
    while (m--) {
        int op;
        cin >> op;
        if (op == 0) {
            int h;
            char c;
            cin >> h >> c;
            tree.change(h - 1, leaf(c));
        } else {
            int hs, ht, ps, pt;
            cin >> hs >> ht >> ps >> pt;
            cout << tree.query(hs - 1, ht - 1).value[ps][pt] << '\n';
        }
    }
    return 0;
}
```

### 复杂度

每次操作 $O(2^3\log n)$，空间 $O(n)$。

### 易错点

矩阵乘法有方向，不能把查询中的分段任意交换。不可达时方案数自然为零，单位矩阵表示不跨任何楼层时留在原侧的一种空路径。
