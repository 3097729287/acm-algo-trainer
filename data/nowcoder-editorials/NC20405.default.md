## [SHOI2007]VOTE 善意的投票

### 题意

改变个人投票偏好与朋友分歧各付1，最少冲突。

### 思路

把源侧看作投1、汇侧看作投0。原偏好1者连源→人容量1，偏好0者连人→汇容量1，割掉时恰罚一次改票。朋友间连两个方向容量1，分在两侧时只有一个方向从源侧跨到汇侧，恰罚一次朋友分歧。总割容量与冲突数一一对应，求最大流。

Dinic先BFS建立剩余容量正边的层次图，再用当前弧DFS只沿层数加1的边增广，更新正反向残量；一轮阻塞流结束后重新分层。最大流等于最小割容量，反向残量允许撤回之前的分配。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Dinic {
    struct Edge {
        int to, reverse;
        long long capacity;
    };
    vector<vector<Edge>> graph;
    vector<int> level, current;
    explicit Dinic(int n) : graph(n), level(n), current(n) {}
    void resize(int n) {
        graph.resize(n);
        level.resize(n);
        current.resize(n);
    }
    int add(int u, int v, long long capacity) {
        int index = graph[u].size(), reverse = graph[v].size();
        graph[u].push_back({v, reverse, capacity});
        graph[v].push_back({u, index, 0});
        return index;
    }
    bool bfs(int source, int sink) {
        fill(level.begin(), level.end(), -1);
        vector<int> queue = {source};
        level[source] = 0;
        for (size_t i = 0; i < queue.size(); i++)
            for (auto e : graph[queue[i]])
                if (e.capacity > 0 && level[e.to] < 0) {
                    level[e.to] = level[queue[i]] + 1;
                    queue.push_back(e.to);
                }
        return level[sink] >= 0;
    }
    long long send(int u, int sink, long long amount) {
        if (u == sink)
            return amount;
        for (int &i = current[u]; i < (int)graph[u].size(); i++) {
            auto &e = graph[u][i];
            if (e.capacity > 0 && level[e.to] == level[u] + 1) {
                long long value = send(e.to, sink, min(amount, e.capacity));
                if (value) {
                    e.capacity -= value;
                    graph[e.to][e.reverse].capacity += value;
                    return value;
                }
            }
        }
        return 0;
    }
    long long maximum(int source, int sink, long long limit = LLONG_MAX / 4) {
        long long total = 0;
        while (total < limit && bfs(source, sink)) {
            fill(current.begin(), current.end(), 0);
            while (total < limit) {
                long long value = send(source, sink, limit - total);
                if (!value)
                    break;
                total += value;
            }
        }
        return total;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    int source = n, sink = n + 1;
    Dinic flow(n + 2);
    for (int i = 0; i < n; i++) {
        int preference;
        cin >> preference;
        if (preference)
            flow.add(source, i, 1);
        else
            flow.add(i, sink, 1);
    }
    while (m--) {
        int u, v;
        cin >> u >> v;
        u--;
        v--;
        flow.add(u, v, 1);
        flow.add(v, u, 1);
    }
    cout << flow.maximum(source, sink);
    return 0;
}
```

### 复杂度

一般Dinic最坏 $O(V^2E)$，空间 $O(V+E)$。

### 易错点

朋友关系无向，两个方向各1；不是容量2。两侧含义必须与个人偏好边方向一致。
