## 树上路径

### 题意

子树加、路径加，查询路径所有不同节点对的权值乘积和。

### 思路

重链剖分使每条重链在DFS序连续，整棵子树也连续。路径被拆成O(log n)个重链区间，子树是一个区间，因此两种加法都交给区间加线段树。

节点维护S=Σa、Q=Σa²；长len区间加v时Q+=2vS+len*v²，随后S+=len*v。路径查询分别累加各段S、Q，所求无序不同节点对乘积和=(S²−Q)/2，模P下乘500000004。若把查询拆段，必须先合并全部S再平方，这样才包含跨段节点对。

父子关系、重儿子、重链编号均由显式栈和逆序计算，避免十万长链递归遍历。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct EulerTree {
    int n;
    vector<vector<int>> edge;
    vector<int> parent, depth, in, out, order;
    EulerTree(int n) : n(n), edge(n + 1), parent(n + 1), depth(n + 1), in(n + 1), out(n + 1) {}
    void add(int u, int v) {
        edge[u].push_back(v);
        edge[v].push_back(u);
    }
    void root(int start) {
        vector<int> stack = {start};
        parent[start] = 0;
        order.clear();
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            in[u] = order.size() + 1;
            order.push_back(u);
            for (int v : edge[u])
                if (v != parent[u]) {
                    parent[v] = u;
                    depth[v] = depth[u] + 1;
                    stack.push_back(v);
                }
        }
        vector<int> size(n + 1, 1);
        for (int i = (int)order.size() - 1; i >= 0; i--) {
            int u = order[i];
            out[u] = in[u] + size[u] - 1;
            if (parent[u])
                size[parent[u]] += size[u];
        }
    }
};

struct SquareTree {
    static constexpr long long mod = 1000000007;
    int n;
    vector<long long> sum, square, tag;
    SquareTree(const vector<long long> &a)
        : n((int)a.size() - 1), sum(4 * n + 4), square(4 * n + 4), tag(4 * n + 4) {
        build(1, 1, n, a);
    }
    void pull(int u) {
        sum[u] = (sum[u * 2] + sum[u * 2 + 1]) % mod;
        square[u] = (square[u * 2] + square[u * 2 + 1]) % mod;
    }
    void build(int u, int l, int r, const vector<long long> &a) {
        if (l == r) {
            sum[u] = (a[l] % mod + mod) % mod;
            square[u] = sum[u] * sum[u] % mod;
            return;
        }
        int mid = (l + r) / 2;
        build(u * 2, l, mid, a);
        build(u * 2 + 1, mid + 1, r, a);
        pull(u);
    }
    void apply(int u, int l, int r, long long v) {
        v = (v % mod + mod) % mod;
        square[u] = (square[u] + 2 * v * sum[u] + (r - l + 1) * v % mod * v) % mod;
        sum[u] = (sum[u] + (r - l + 1) * v) % mod;
        tag[u] = (tag[u] + v) % mod;
    }
    void push(int u, int l, int r) {
        if (!tag[u] || l == r)
            return;
        int mid = (l + r) / 2;
        apply(u * 2, l, mid, tag[u]);
        apply(u * 2 + 1, mid + 1, r, tag[u]);
        tag[u] = 0;
    }
    void change(int u, int l, int r, int ql, int qr, long long v) {
        if (ql <= l && r <= qr) {
            apply(u, l, r, v);
            return;
        }
        push(u, l, r);
        int mid = (l + r) / 2;
        if (ql <= mid)
            change(u * 2, l, mid, ql, qr, v);
        if (qr > mid)
            change(u * 2 + 1, mid + 1, r, ql, qr, v);
        pull(u);
    }
    long long query(int u, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr)
            return square[u];
        push(u, l, r);
        int mid = (l + r) / 2;
        long long value = 0;
        if (ql <= mid)
            value += query(u * 2, l, mid, ql, qr);
        if (qr > mid)
            value += query(u * 2 + 1, mid + 1, r, ql, qr);
        return value % mod;
    }
    long long querySum(int u, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr)
            return sum[u];
        push(u, l, r);
        int mid = (l + r) / 2;
        long long value = 0;
        if (ql <= mid)
            value += querySum(u * 2, l, mid, ql, qr);
        if (qr > mid)
            value += querySum(u * 2 + 1, mid + 1, r, ql, qr);
        return value % mod;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    vector<long long> weight(n + 1);
    for (int i = 1; i <= n; i++)
        cin >> weight[i];
    EulerTree tree(n);
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        tree.add(u, v);
    }
    tree.root(1);
    vector<int> size(n + 1, 1), heavy(n + 1), head(n + 1);
    size[0] = 0;
    for (int i = n - 1; i >= 0; i--) {
        int u = tree.order[i];
        for (int v : tree.edge[u])
            if (tree.parent[v] == u) {
                size[u] += size[v];
                if (size[v] > size[heavy[u]])
                    heavy[u] = v;
            }
    }
    vector<pair<int, int>> stack = {{1, 1}};
    int timer = 0;
    vector<long long> a(n + 1);
    while (!stack.empty()) {
        auto [start, h] = stack.back();
        stack.pop_back();
        for (int u = start; u; u = heavy[u]) {
            head[u] = h;
            tree.in[u] = ++timer;
            a[timer] = weight[u];
            for (int v : tree.edge[u])
                if (tree.parent[v] == u && v != heavy[u])
                    stack.push_back({v, v});
        }
    }
    SquareTree segment(a);
    const long long mod = 1000000007;
    auto path = [&](int u, int v, auto process) {
        while (head[u] != head[v]) {
            if (tree.depth[head[u]] < tree.depth[head[v]])
                swap(u, v);
            process(tree.in[head[u]], tree.in[u]);
            u = tree.parent[head[u]];
        }
        if (tree.depth[u] > tree.depth[v])
            swap(u, v);
        process(tree.in[u], tree.in[v]);
    };
    while (m--) {
        int op, u;
        cin >> op >> u;
        if (op == 1) {
            long long value;
            cin >> value;
            segment.change(1, 1, n, tree.in[u], tree.in[u] + size[u] - 1, value);
        } else if (op == 2) {
            int v;
            long long value;
            cin >> v >> value;
            path(u, v, [&](int l, int r) { segment.change(1, 1, n, l, r, value); });
        } else {
            int v;
            cin >> v;
            long long sum = 0, square = 0;
            path(u, v, [&](int l, int r) {
                sum = (sum + segment.querySum(1, 1, n, l, r)) % mod;
                square = (square + segment.query(1, 1, n, l, r)) % mod;
            });
            cout << (sum * sum % mod - square + mod) % mod * 500000004 % mod << '\n';
        }
    }
    return 0;
}
```

### 复杂度

子树修改 $O(\log n)$，路径操作 $O(\log² n)$，空间 $O(n)$。

### 易错点

重儿子比较前size[0]应为0；这里空儿子0初始化后需显式清零。单点路径的不同点对数为0。
