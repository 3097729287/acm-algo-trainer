## 区区区间

### 题意

区间覆盖为k,k+1,...并查询区间和。

### 思路

对本次左端l，目标值可写为a[i]=i+b，其中b=k−l。因此懒标记只保存全局偏移b，不必给每个子区间另算首项。覆盖节点[L,R]后，区间和=(L+R+2b)*(R−L+1)/2。新覆盖直接替换旧标记；部分访问前把标记下传，子节点继续使用相同b。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct SegmentTree {
    int n;
    vector<long long> sum, tag;
    vector<bool> marked;
    SegmentTree(const vector<long long> &a)
        : n((int)a.size() - 1), sum(4 * n + 4), tag(4 * n + 4), marked(4 * n + 4) {
        build(1, 1, n, a);
    }
    void build(int u, int l, int r, const vector<long long> &a) {
        if (l == r) {
            sum[u] = a[l];
            return;
        }
        int mid = (l + r) / 2;
        build(u * 2, l, mid, a);
        build(u * 2 + 1, mid + 1, r, a);
        sum[u] = sum[u * 2] + sum[u * 2 + 1];
    }
    void apply(int u, int l, int r, long long b) {
        sum[u] = (1LL * (l + r) + 2 * b) * (r - l + 1) / 2;
        marked[u] = true;
        tag[u] = b;
    }
    void push(int u, int l, int r) {
        if (!marked[u] || l == r)
            return;
        int mid = (l + r) / 2;
        apply(u * 2, l, mid, tag[u]);
        apply(u * 2 + 1, mid + 1, r, tag[u]);
        marked[u] = false;
    }
    void change(int u, int l, int r, int ql, int qr, long long b) {
        if (ql <= l && r <= qr) {
            apply(u, l, r, b);
            return;
        }
        push(u, l, r);
        int mid = (l + r) / 2;
        if (ql <= mid)
            change(u * 2, l, mid, ql, qr, b);
        if (qr > mid)
            change(u * 2 + 1, mid + 1, r, ql, qr, b);
        sum[u] = sum[u * 2] + sum[u * 2 + 1];
    }
    long long query(int u, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr)
            return sum[u];
        push(u, l, r);
        int mid = (l + r) / 2;
        long long result = 0;
        if (ql <= mid)
            result += query(u * 2, l, mid, ql, qr);
        if (qr > mid)
            result += query(u * 2 + 1, mid + 1, r, ql, qr);
        return result;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    vector<long long> a(n + 1);
    for (int i = 1; i <= n; i++)
        cin >> a[i];
    SegmentTree tree(a);
    while (m--) {
        int op, l, r;
        cin >> op >> l >> r;
        if (op == 1) {
            long long k;
            cin >> k;
            tree.change(1, 1, n, l, r, k - l);
        } else
            cout << tree.query(1, 1, n, l, r) << '\n';
    }
    return 0;
}
```

### 复杂度

构建 $O(n)$，每次操作 $O(\log n)$，空间 $O(n)$。

### 易错点

偏移b可能为0或负数，必须独立保存“是否存在覆盖”的布尔标记，不能用tag==0判断。
