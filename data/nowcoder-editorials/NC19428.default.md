## 树上求和

### 题意

子树加，查询子树权值平方和模23333。

### 思路

DFS序把子树化为区间。线段树同时维护总和S与平方和Q；长度len的一段加v后，Q新=Q+2vS+len*v²，S新=S+len*v。区间加懒标记累加，下传后再做部分修改或查询，始终模23333。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct EulerTree {
    int n;
    vector<vector<int>> edge;
    vector<int> parent, depth, in, out, order;
    EulerTree(int n) : n(n), edge(n + 1), parent(n + 1), depth(n + 1), in(n + 1), out(n + 1) {}
    void add(int u, int v) {
        edge[u].push_back(v);
        edge[v].push_back(u);
    }
    void root(int start) {
        vector<int> stack = {start};
        parent[start] = 0;
        order.clear();
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            in[u] = order.size() + 1;
            order.push_back(u);
            for (int v : edge[u])
                if (v != parent[u]) {
                    parent[v] = u;
                    depth[v] = depth[u] + 1;
                    stack.push_back(v);
                }
        }
        vector<int> size(n + 1, 1);
        for (int i = (int)order.size() - 1; i >= 0; i--) {
            int u = order[i];
            out[u] = in[u] + size[u] - 1;
            if (parent[u])
                size[parent[u]] += size[u];
        }
    }
};

struct SquareTree {
    static constexpr long long mod = 23333;
    int n;
    vector<long long> sum, square, tag;
    SquareTree(const vector<long long> &a)
        : n((int)a.size() - 1), sum(4 * n + 4), square(4 * n + 4), tag(4 * n + 4) {
        build(1, 1, n, a);
    }
    void pull(int u) {
        sum[u] = (sum[u * 2] + sum[u * 2 + 1]) % mod;
        square[u] = (square[u * 2] + square[u * 2 + 1]) % mod;
    }
    void build(int u, int l, int r, const vector<long long> &a) {
        if (l == r) {
            sum[u] = (a[l] % mod + mod) % mod;
            square[u] = sum[u] * sum[u] % mod;
            return;
        }
        int mid = (l + r) / 2;
        build(u * 2, l, mid, a);
        build(u * 2 + 1, mid + 1, r, a);
        pull(u);
    }
    void apply(int u, int l, int r, long long v) {
        v = (v % mod + mod) % mod;
        square[u] = (square[u] + 2 * v * sum[u] + (r - l + 1) * v % mod * v) % mod;
        sum[u] = (sum[u] + (r - l + 1) * v) % mod;
        tag[u] = (tag[u] + v) % mod;
    }
    void push(int u, int l, int r) {
        if (!tag[u] || l == r)
            return;
        int mid = (l + r) / 2;
        apply(u * 2, l, mid, tag[u]);
        apply(u * 2 + 1, mid + 1, r, tag[u]);
        tag[u] = 0;
    }
    void change(int u, int l, int r, int ql, int qr, long long v) {
        if (ql <= l && r <= qr) {
            apply(u, l, r, v);
            return;
        }
        push(u, l, r);
        int mid = (l + r) / 2;
        if (ql <= mid)
            change(u * 2, l, mid, ql, qr, v);
        if (qr > mid)
            change(u * 2 + 1, mid + 1, r, ql, qr, v);
        pull(u);
    }
    long long query(int u, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr)
            return square[u];
        push(u, l, r);
        int mid = (l + r) / 2;
        long long value = 0;
        if (ql <= mid)
            value += query(u * 2, l, mid, ql, qr);
        if (qr > mid)
            value += query(u * 2 + 1, mid + 1, r, ql, qr);
        return value % mod;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, q;
    cin >> n >> q;
    vector<long long> value(n + 1);
    for (int i = 1; i <= n; i++)
        cin >> value[i];
    EulerTree tree(n);
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        tree.add(u, v);
    }
    tree.root(1);
    vector<long long> a(n + 1);
    for (int u = 1; u <= n; u++)
        a[tree.in[u]] = value[u];
    SquareTree segment(a);
    while (q--) {
        int op, u;
        cin >> op >> u;
        if (op == 1) {
            long long v;
            cin >> v;
            segment.change(1, 1, n, tree.in[u], tree.out[u], v);
        } else
            cout << segment.query(1, 1, n, tree.in[u], tree.out[u]) << '\n';
    }
    return 0;
}
```

### 复杂度

构建 $O(n)$，每次操作 $O(\log n)$，空间 $O(n)$。

### 易错点

Q的更新使用修改前的S。平方和与和的平方不同，不能只维护总和。
