## [POI2008]Tro

### 题意

求所有三点三角形面积总和。

### 思路

固定顶点P，将其余向量极角排序。对每个向量v，用双指针找到逆时针半周内的全部向量w，使cross(v,w)≥0（同方向零叉积无影响，反方向端点不计）。这些叉积之和等于cross(v,Σw)，用向量前缀和常数计算。

每个非共线无序向量对恰按正叉积方向计一次。每个三角形又会在三个顶点各贡献一次“二倍面积”，所以全部累计除3得到二倍面积总和，再按整数或半整数输出一位小数。128位累加防止三角形数量放大总量。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

using Wide = __int128_t;
struct IPoint {
    long long x = 0, y = 0;
    IPoint operator+(IPoint b) const { return {x + b.x, y + b.y}; }
    IPoint operator-(IPoint b) const { return {x - b.x, y - b.y}; }
};
Wide cross(IPoint a, IPoint b) { return (Wide)a.x * b.y - (Wide)a.y * b.x; }
Wide dot(IPoint a, IPoint b) { return (Wide)a.x * b.x + (Wide)a.y * b.y; }
Wide absolute(Wide x) { return x < 0 ? -x : x; }
void printWide(Wide value) {
    if (value < 0) {
        cout << '-';
        value = -value;
    }
    string text;
    do {
        text.push_back('0' + value % 10);
        value /= 10;
    } while (value);
    reverse(text.begin(), text.end());
    cout << text;
}
bool angleLess(IPoint a, IPoint b) {
    auto half = [](IPoint p) { return p.y < 0 || (p.y == 0 && p.x < 0); };
    if (half(a) != half(b))
        return half(a) < half(b);
    return cross(a, b) > 0;
}
vector<IPoint> integerHull(vector<IPoint> point) {
    sort(point.begin(), point.end(),
         [](auto a, auto b) { return a.x != b.x ? a.x < b.x : a.y < b.y; });
    point.erase(
        unique(point.begin(), point.end(), [](auto a, auto b) { return a.x == b.x && a.y == b.y; }),
        point.end());
    if (point.size() < 2)
        return point;
    vector<IPoint> hull;
    for (auto p : point) {
        while (hull.size() >= 2 && cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    size_t lower = hull.size();
    for (int i = (int)point.size() - 2; i >= 0; i--) {
        auto p = point[i];
        while (hull.size() > lower &&
               cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    hull.pop_back();
    return hull;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<IPoint> p(n);
    for (auto &a : p)
        cin >> a.x >> a.y;
    Wide total = 0;
    for (int origin = 0; origin < n; origin++) {
        vector<IPoint> v;
        for (int i = 0; i < n; i++)
            if (i != origin && (p[i].x != p[origin].x || p[i].y != p[origin].y))
                v.push_back(p[i] - p[origin]);
        sort(v.begin(), v.end(), angleLess);
        int m = v.size();
        for (int i = 0; i < m; i++)
            v.push_back(v[i]);
        vector<IPoint> prefix(v.size() + 1);
        for (int i = 0; i < (int)v.size(); i++)
            prefix[i + 1] = prefix[i] + v[i];
        int right = 1;
        for (int i = 0; i < m; i++) {
            right = max(right, i + 1);
            while (right < i + m &&
                   (cross(v[i], v[right]) > 0 ||
                    (right < m && cross(v[i], v[right]) == 0 && dot(v[i], v[right]) > 0)))
                right++;
            total += cross(v[i], prefix[right] - prefix[i + 1]);
        }
    }
    Wide twice = total / 3;
    printWide(twice / 2);
    cout << (twice % 2 ? ".5" : ".0");
    return 0;
}
```

### 复杂度

时间 $O(n²\log n)$，空间 $O(n)$。

### 易错点

不能直接把正有向面积当全体面积，必须覆盖每个顶点的全极角并去三重计数。共线三点贡献0。
