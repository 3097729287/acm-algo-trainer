## [网络流24题]圆桌问题

### 题意

同单位代表不能同桌，按容量分配桌号。

### 思路

源到单位容量为人数，单位到每桌容量1，桌到汇容量为桌容量。流满总人数即存在方案；单位→桌边满1表示该单位一名代表坐那桌，逐单位输出这些桌号。容量1精确保证同单位在同桌最多一人。

Dinic先BFS建立剩余容量正边的层次图，再用当前弧DFS只沿层数加1的边增广，更新正反向残量；一轮阻塞流结束后重新分层。最大流等于最小割容量，反向残量允许撤回之前的分配。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Dinic {
    struct Edge {
        int to, reverse;
        long long capacity;
    };
    vector<vector<Edge>> graph;
    vector<int> level, current;
    explicit Dinic(int n) : graph(n), level(n), current(n) {}
    void resize(int n) {
        graph.resize(n);
        level.resize(n);
        current.resize(n);
    }
    int add(int u, int v, long long capacity) {
        int index = graph[u].size(), reverse = graph[v].size();
        graph[u].push_back({v, reverse, capacity});
        graph[v].push_back({u, index, 0});
        return index;
    }
    bool bfs(int source, int sink) {
        fill(level.begin(), level.end(), -1);
        vector<int> queue = {source};
        level[source] = 0;
        for (size_t i = 0; i < queue.size(); i++)
            for (auto e : graph[queue[i]])
                if (e.capacity > 0 && level[e.to] < 0) {
                    level[e.to] = level[queue[i]] + 1;
                    queue.push_back(e.to);
                }
        return level[sink] >= 0;
    }
    long long send(int u, int sink, long long amount) {
        if (u == sink)
            return amount;
        for (int &i = current[u]; i < (int)graph[u].size(); i++) {
            auto &e = graph[u][i];
            if (e.capacity > 0 && level[e.to] == level[u] + 1) {
                long long value = send(e.to, sink, min(amount, e.capacity));
                if (value) {
                    e.capacity -= value;
                    graph[e.to][e.reverse].capacity += value;
                    return value;
                }
            }
        }
        return 0;
    }
    long long maximum(int source, int sink, long long limit = LLONG_MAX / 4) {
        long long total = 0;
        while (total < limit && bfs(source, sink)) {
            fill(current.begin(), current.end(), 0);
            while (total < limit) {
                long long value = send(source, sink, limit - total);
                if (!value)
                    break;
                total += value;
            }
        }
        return total;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int groups, tables;
    cin >> groups >> tables;
    int source = groups + tables, sink = source + 1;
    Dinic flow(sink + 1);
    long long required = 0;
    for (int i = 0; i < groups; i++) {
        int count;
        cin >> count;
        required += count;
        flow.add(source, i, count);
    }
    for (int j = 0; j < tables; j++) {
        int capacity;
        cin >> capacity;
        flow.add(groups + j, sink, capacity);
    }
    vector<vector<int>> index(groups, vector<int>(tables));
    for (int i = 0; i < groups; i++)
        for (int j = 0; j < tables; j++)
            index[i][j] = flow.add(i, groups + j, 1);
    if (flow.maximum(source, sink) != required) {
        cout << 0;
        return 0;
    }
    cout << 1 << '\n';
    for (int i = 0; i < groups; i++) {
        bool space = false;
        for (int j = 0; j < tables; j++)
            if (flow.graph[i][index[i][j]].capacity == 0) {
                if (space)
                    cout << ' ';
                space = true;
                cout << j + 1;
            }
        cout << '\n';
    }
    return 0;
}
```

### 复杂度

一般Dinic最坏 $O(V^2E)$，V=m+n+2、E=O(mn)，空间 $O(E)$。

### 易错点

第一行输入m是单位数、n是桌数，变量按含义命名避免题面前后字母变化。总容量足够并不保证可行，还受每组每桌至多一人限制。
