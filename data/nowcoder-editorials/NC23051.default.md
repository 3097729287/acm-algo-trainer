## 华华和月月种树

### 题意

动态添加零权儿子，子树加，单点查询。

### 思路

先读完操作建最终树，得到DFS子树区间，再按原时间重放。用差分树状数组做最终子树区间加、单点查询。最终子树中包含未来才出生的节点，会被早期更新计入；因此新节点出生时，记下当前该位置的累计值base。其真实权值始终是当前累计值−base，恰好去掉出生之前的全部更新。根从一开始存在，base为0。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct EulerTree {
    int n;
    vector<vector<int>> edge;
    vector<int> parent, depth, in, out, order;
    EulerTree(int n) : n(n), edge(n + 1), parent(n + 1), depth(n + 1), in(n + 1), out(n + 1) {}
    void add(int u, int v) {
        edge[u].push_back(v);
        edge[v].push_back(u);
    }
    void root(int start) {
        vector<int> stack = {start};
        parent[start] = 0;
        order.clear();
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            in[u] = order.size() + 1;
            order.push_back(u);
            for (int v : edge[u])
                if (v != parent[u]) {
                    parent[v] = u;
                    depth[v] = depth[u] + 1;
                    stack.push_back(v);
                }
        }
        vector<int> size(n + 1, 1);
        for (int i = (int)order.size() - 1; i >= 0; i--) {
            int u = order[i];
            out[u] = in[u] + size[u] - 1;
            if (parent[u])
                size[parent[u]] += size[u];
        }
    }
};

struct Fenwick {
    int n;
    vector<long long> tree;
    Fenwick(int n) : n(n), tree(n + 1) {}
    void add(int p, long long v) {
        for (; p <= n; p += p & -p)
            tree[p] += v;
    }
    long long sum(int p) {
        long long result = 0;
        for (; p; p -= p & -p)
            result += tree[p];
        return result;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int m;
    cin >> m;
    struct Operation {
        int type, node, created;
        long long value;
    };
    vector<Operation> operation;
    EulerTree tree(m + 1);
    int nodes = 1;
    for (int i = 0; i < m; i++) {
        int op, u;
        cin >> op >> u;
        u++;
        if (op == 1) {
            int v = ++nodes;
            tree.add(u, v);
            operation.push_back({op, u, v, 0});
        } else {
            long long x = 0;
            if (op == 2)
                cin >> x;
            operation.push_back({op, u, 0, x});
        }
    }
    tree.root(1);
    Fenwick bit(nodes);
    vector<long long> base(nodes + 1);
    for (auto op : operation) {
        if (op.type == 1)
            base[op.created] = bit.sum(tree.in[op.created]);
        else if (op.type == 2) {
            bit.add(tree.in[op.node], op.value);
            if (tree.out[op.node] < nodes)
                bit.add(tree.out[op.node] + 1, -op.value);
        } else
            cout << bit.sum(tree.in[op.node]) - base[op.node] << '\n';
    }
    return 0;
}
```

### 复杂度

时间 $O(M\log M)$，空间 $O(M)$。

### 易错点

内部编号整体加1以兼容Fenwick下标。新节点初始必须为0，不能继承父节点历史增量。
