## King's Quest

### 题意

给定完美婚配，求每个男孩可在某个完美匹配中迎娶的女孩。

### 思路

设owner[g]是给定匹配中女孩g的男孩。每条喜欢边u→g转成男孩图边u→owner[g]。把u改配g会挤占另一个男孩，若可沿此图回到u，便能沿交替环轮换所有婚配。故喜欢边可选当且仅当u和owner[g]在同一个SCC；原匹配对应自环自然可选。每人筛选后按女孩编号排序。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

pair<int, vector<int>> components(const vector<vector<int>> &graph,
                                  const vector<vector<int>> &reverse) {
    int n = graph.size();
    vector<bool> seen(n);
    vector<int> finish;
    for (int start = 0; start < n; start++)
        if (!seen[start]) {
            vector<pair<int, int>> stack{{start, 0}};
            seen[start] = true;
            while (!stack.empty()) {
                int u = stack.back().first, &position = stack.back().second;
                if (position == (int)graph[u].size()) {
                    finish.push_back(u);
                    stack.pop_back();
                } else {
                    int v = graph[u][position++];
                    if (!seen[v]) {
                        seen[v] = true;
                        stack.push_back({v, 0});
                    }
                }
            }
        }
    vector<int> component(n, -1);
    int count = 0;
    for (int i = n - 1; i >= 0; i--) {
        int start = finish[i];
        if (component[start] != -1)
            continue;
        vector<int> stack{start};
        component[start] = count;
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            for (int v : reverse[u])
                if (component[v] == -1) {
                    component[v] = count;
                    stack.push_back(v);
                }
        }
        count++;
    }
    return {count, component};
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<vector<int>> likes(n);
    for (auto &row : likes) {
        int k;
        cin >> k;
        row.resize(k);
        for (int &g : row) {
            cin >> g;
            g--;
        }
    }
    vector<int> owner(n);
    for (int u = 0; u < n; u++) {
        int g;
        cin >> g;
        owner[g - 1] = u;
    }
    vector<vector<int>> graph(n), reverse(n);
    for (int u = 0; u < n; u++)
        for (int g : likes[u]) {
            int v = owner[g];
            graph[u].push_back(v);
            reverse[v].push_back(u);
        }
    auto [count, component] = components(graph, reverse);
    for (int u = 0; u < n; u++) {
        vector<int> answer;
        for (int g : likes[u])
            if (component[u] == component[owner[g]])
                answer.push_back(g + 1);
        sort(answer.begin(), answer.end());
        cout << answer.size();
        for (int g : answer)
            cout << ' ' << g;
        cout << '\n';
    }
    return 0;
}
```

### 复杂度

时间 $O(n+E+E\log n)$（含输出排序），空间 $O(n+E)$。

### 易错点

两完美匹配的对称差是交替环，所以条件既必要又充分。只在原喜欢列表中筛，不输出不喜欢的女孩。
