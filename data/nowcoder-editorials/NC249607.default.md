## 平面最近点对

### 题意

多组整点，求最近两点距离的平方。

### 思路

按x排序后分治，左右各求最近距离平方d。归并使区间按y排序，只保留与分界竖线横距平方<d的点。按y遍历条带，倒着比较纵距平方<d的前驱；平面装箱性质保证每个点只需比较常数个候选。递归进入前记住分界x，不能在左右已改为y序后再读取。所有比较都用整数平方，避免开根号误差。

题面把n下界写成1，而“两个不同点”的任务仅在n≥2时有定义，原题P7883下界为2。代码对n=1保护输出0；有效点对按上述算法处理。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

using Wide = __int128_t;
struct IPoint {
    long long x = 0, y = 0;
    IPoint operator+(IPoint b) const { return {x + b.x, y + b.y}; }
    IPoint operator-(IPoint b) const { return {x - b.x, y - b.y}; }
};
Wide cross(IPoint a, IPoint b) { return (Wide)a.x * b.y - (Wide)a.y * b.x; }
Wide dot(IPoint a, IPoint b) { return (Wide)a.x * b.x + (Wide)a.y * b.y; }
Wide absolute(Wide x) { return x < 0 ? -x : x; }
void printWide(Wide value) {
    if (value < 0) {
        cout << '-';
        value = -value;
    }
    string text;
    do {
        text.push_back('0' + value % 10);
        value /= 10;
    } while (value);
    reverse(text.begin(), text.end());
    cout << text;
}
bool angleLess(IPoint a, IPoint b) {
    auto half = [](IPoint p) { return p.y < 0 || (p.y == 0 && p.x < 0); };
    if (half(a) != half(b))
        return half(a) < half(b);
    return cross(a, b) > 0;
}
vector<IPoint> integerHull(vector<IPoint> point) {
    sort(point.begin(), point.end(),
         [](auto a, auto b) { return a.x != b.x ? a.x < b.x : a.y < b.y; });
    point.erase(
        unique(point.begin(), point.end(), [](auto a, auto b) { return a.x == b.x && a.y == b.y; }),
        point.end());
    if (point.size() < 2)
        return point;
    vector<IPoint> hull;
    for (auto p : point) {
        while (hull.size() >= 2 && cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    size_t lower = hull.size();
    for (int i = (int)point.size() - 2; i >= 0; i--) {
        auto p = point[i];
        while (hull.size() > lower &&
               cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    hull.pop_back();
    return hull;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int tests;
    cin >> tests;
    while (tests--) {
        int n;
        cin >> n;
        vector<IPoint> p(n), buffer(n);
        for (auto &q : p)
            cin >> q.x >> q.y;
        sort(p.begin(), p.end(), [](auto a, auto b) { return a.x < b.x; });
        auto squared = [](IPoint a, IPoint b) {
            auto d = a - b;
            return d.x * d.x + d.y * d.y;
        };
        function<long long(int, int)> solve = [&](int l, int r) -> long long {
            if (r - l <= 3) {
                long long best = LLONG_MAX;
                for (int i = l; i < r; i++)
                    for (int j = i + 1; j < r; j++)
                        best = min(best, squared(p[i], p[j]));
                sort(p.begin() + l, p.begin() + r, [](auto a, auto b) { return a.y < b.y; });
                return best;
            }
            int mid = (l + r) / 2;
            long long split = p[mid].x, d = min(solve(l, mid), solve(mid, r));
            merge(p.begin() + l, p.begin() + mid, p.begin() + mid, p.begin() + r,
                  buffer.begin() + l, [](auto a, auto b) { return a.y < b.y; });
            copy(buffer.begin() + l, buffer.begin() + r, p.begin() + l);
            vector<IPoint> strip;
            for (int i = l; i < r; i++)
                if ((p[i].x - split) * (p[i].x - split) < d) {
                    for (int j = (int)strip.size() - 1;
                         j >= 0 && (p[i].y - strip[j].y) * (p[i].y - strip[j].y) < d; j--)
                        d = min(d, squared(p[i], strip[j]));
                    strip.push_back(p[i]);
                }
            return d;
        };
        cout << (n < 2 ? 0 : solve(0, n)) << '\n';
    }
    return 0;
}
```

### 复杂度

时间 $O(n\log n)$，空间 $O(n)$。

### 易错点

输出距离平方而非距离。坐标差2×10^7，平方和可达8×10^14，用64位。归并保证总复杂度，不在每层重新排序整个条带。
