## Bridge Across Islands

### 题意

两个分离凸岛，求边界最短距离。

### 思路

两集合距离=min|a−b|，所有差向量组成凸多边形C=A+(−B)，即闵可夫斯基和。将两多边形统一成逆时针凸包，从最低顶点开始，按边方向合并；同向边相加，线性构造差多边形。若原点在C内则距离0，否则取原点到C各边线段的最短距离。

点到线段用投影参数t=−a·(b−a)/|b−a|²，截到[0,1]后求|a+t(b−a)|。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

constexpr long double EPS = 1e-12L;
struct Point {
    long double x = 0, y = 0;
    Point operator+(Point b) const { return {x + b.x, y + b.y}; }
    Point operator-(Point b) const { return {x - b.x, y - b.y}; }
    Point operator*(long double t) const { return {x * t, y * t}; }
    Point operator/(long double t) const { return {x / t, y / t}; }
};
long double cross(Point a, Point b) { return a.x * b.y - a.y * b.x; }
long double dot(Point a, Point b) { return a.x * b.x + a.y * b.y; }
long double norm(Point a) { return sqrtl(dot(a, a)); }
long double distance(Point a, Point b) { return norm(a - b); }
Point rotate90(Point a) { return {-a.y, a.x}; }

vector<Point> convexHull(vector<Point> point) {
    sort(point.begin(), point.end(),
         [](Point a, Point b) { return a.x != b.x ? a.x < b.x : a.y < b.y; });
    point.erase(unique(point.begin(), point.end(),
                       [](Point a, Point b) { return a.x == b.x && a.y == b.y; }),
                point.end());
    if (point.size() < 2)
        return point;
    vector<Point> hull;
    for (Point p : point) {
        while (hull.size() >= 2 && cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    size_t lower = hull.size();
    for (int i = (int)point.size() - 2; i >= 0; i--) {
        Point p = point[i];
        while (hull.size() > lower &&
               cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    hull.pop_back();
    return hull;
}

vector<Point> minkowski(vector<Point> a, vector<Point> b) {
    if (a.size() < 3 || b.size() < 3) {
        vector<Point> sum;
        for (auto p : a)
            for (auto q : b)
                sum.push_back(p + q);
        return convexHull(sum);
    }
    auto prepare = [](vector<Point> &p) {
        auto it = min_element(p.begin(), p.end(),
                              [](auto a, auto b) { return a.y != b.y ? a.y < b.y : a.x < b.x; });
        rotate(p.begin(), it, p.end());
    };
    prepare(a);
    prepare(b);
    int n = a.size(), m = b.size(), i = 0, j = 0;
    vector<Point> result;
    Point current = a[0] + b[0];
    result.push_back(current);
    while (i < n || j < m) {
        Point x = i < n ? a[(i + 1) % n] - a[i] : Point{},
              y = j < m ? b[(j + 1) % m] - b[j] : Point{};
        long double c = cross(x, y);
        if (j == m || (i < n && c > EPS)) {
            current = current + x;
            i++;
        } else if (i == n || c < -EPS) {
            current = current + y;
            j++;
        } else {
            current = current + x + y;
            i++;
            j++;
        }
        result.push_back(current);
    }
    result.pop_back();
    return result;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    while (cin >> n >> m && (n || m)) {
        vector<Point> a(n), b(m);
        for (auto &p : a)
            cin >> p.x >> p.y;
        for (auto &p : b) {
            cin >> p.x >> p.y;
            p = p * (-1);
        }
        auto c = minkowski(convexHull(a), convexHull(b));
        long double answer = 1e100L;
        bool inside = c.size() >= 3;
        for (int i = 0; i < (int)c.size(); i++) {
            Point x = c[i], y = c[(i + 1) % c.size()], e = y - x;
            if (cross(e, x * (-1)) < -EPS)
                inside = false;
            long double t = dot(e, e) > 0 ? -dot(x, e) / dot(e, e) : 0;
            t = max(0.0L, min(1.0L, t));
            answer = min(answer, norm(x + e * t));
        }
        if (inside)
            answer = 0;
        cout << fixed << setprecision(12) << answer << '\n';
    }
    return 0;
}
```

### 复杂度

含规范化凸包时间 $O(n\log n+m\log m)$，和及距离扫描 $O(n+m)$，空间 $O(n+m)$。

### 易错点

岛屿顶点输入可能顺时针，先重新求凸包统一方向。点间最短距离不一定发生在两个顶点，必须考虑点到边。
