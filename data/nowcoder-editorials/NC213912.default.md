## 芭芭拉冲鸭~（续）

### 题意

树路径上收集字母，可重排部分字母，求最长回文长度。

### 思路

任意字母可成对放在回文两侧，最多留下一个奇数频次字母放中心。路径长len、奇数频次字母数odd时，答案为len−max(0,odd−1)。用26位掩码记录根路径上字母频次奇偶，路径掩码为mask[x] xor mask[y] xor (1<<字母[LCA])，popcount得到odd。长度由深度和LCA求出。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct EulerTree {
    int n;
    vector<vector<int>> edge;
    vector<int> parent, depth, in, out, order;
    EulerTree(int n) : n(n), edge(n + 1), parent(n + 1), depth(n + 1), in(n + 1), out(n + 1) {}
    void add(int u, int v) {
        edge[u].push_back(v);
        edge[v].push_back(u);
    }
    void root(int start) {
        vector<int> stack = {start};
        parent[start] = 0;
        order.clear();
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            in[u] = order.size() + 1;
            order.push_back(u);
            for (int v : edge[u])
                if (v != parent[u]) {
                    parent[v] = u;
                    depth[v] = depth[u] + 1;
                    stack.push_back(v);
                }
        }
        vector<int> size(n + 1, 1);
        for (int i = (int)order.size() - 1; i >= 0; i--) {
            int u = order[i];
            out[u] = in[u] + size[u] - 1;
            if (parent[u])
                size[parent[u]] += size[u];
        }
    }
};

struct Ancestors {
    const EulerTree &tree;
    vector<vector<int>> up;
    Ancestors(const EulerTree &tree) : tree(tree) {
        int levels = 1;
        while ((1LL << levels) <= tree.n)
            levels++;
        up.assign(levels, vector<int>(tree.n + 1));
        up[0] = tree.parent;
        for (int b = 1; b < levels; b++)
            for (int u = 1; u <= tree.n; u++)
                up[b][u] = up[b - 1][up[b - 1][u]];
    }
    int lca(int u, int v) const {
        if (tree.depth[u] < tree.depth[v])
            swap(u, v);
        int gap = tree.depth[u] - tree.depth[v];
        for (int b = 0; b < (int)up.size(); b++)
            if (gap >> b & 1)
                u = up[b][u];
        if (u == v)
            return u;
        for (int b = (int)up.size() - 1; b >= 0; b--)
            if (up[b][u] != up[b][v]) {
                u = up[b][u];
                v = up[b][v];
            }
        return up[0][u];
    }
    int distance(int u, int v) const {
        return tree.depth[u] + tree.depth[v] - 2 * tree.depth[lca(u, v)];
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    EulerTree tree(n);
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        tree.add(u, v);
    }
    string s;
    cin >> s;
    tree.root(1);
    Ancestors ancestor(tree);
    vector<unsigned> mask(n + 1);
    for (int u : tree.order)
        mask[u] = mask[tree.parent[u]] ^ (1U << (s[u - 1] - 'a'));
    int q;
    cin >> q;
    while (q--) {
        int x, y;
        cin >> x >> y;
        int a = ancestor.lca(x, y), length = tree.depth[x] + tree.depth[y] - 2 * tree.depth[a] + 1;
        int odd = __builtin_popcount(mask[x] ^ mask[y] ^ (1U << (s[a - 1] - 'a')));
        cout << length - max(0, odd - 1) << '\n';
    }
    return 0;
}
```

### 复杂度

预处理 $O(n\log n)$，每问 $O(\log n)$，空间 $O(n\log n)$。

### 易错点

LCA在两个前缀掩码中抵消，必须补回它的字母。题目允许重排，不是路径上的最长回文子串。
