## [网络流24题]方格取数问题

### 题意

网格选不相邻格最大权值和。

### 思路

棋盘黑白染色，邻接图是二分图。最大权独立集=总权−最小权顶点覆盖。源到黑格容量权值，白格到汇容量权值，黑白相邻格连无穷容量。有限割必须在每条邻接边至少舍弃一个端点，割成本恰是舍弃格的总权，因此答案总权−最大流。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Dinic {
    struct Edge {
        int to, reverse;
        long long capacity;
    };
    vector<vector<Edge>> graph;
    vector<int> level, current;
    explicit Dinic(int n) : graph(n), level(n), current(n) {}
    void resize(int n) {
        graph.resize(n);
        level.resize(n);
        current.resize(n);
    }
    int add(int u, int v, long long capacity) {
        int index = graph[u].size(), reverse = graph[v].size();
        graph[u].push_back({v, reverse, capacity});
        graph[v].push_back({u, index, 0});
        return index;
    }
    bool bfs(int source, int sink) {
        fill(level.begin(), level.end(), -1);
        vector<int> queue = {source};
        level[source] = 0;
        for (size_t i = 0; i < queue.size(); i++)
            for (auto e : graph[queue[i]])
                if (e.capacity > 0 && level[e.to] < 0) {
                    level[e.to] = level[queue[i]] + 1;
                    queue.push_back(e.to);
                }
        return level[sink] >= 0;
    }
    long long send(int u, int sink, long long amount) {
        if (u == sink)
            return amount;
        for (int &i = current[u]; i < (int)graph[u].size(); i++) {
            auto &e = graph[u][i];
            if (e.capacity > 0 && level[e.to] == level[u] + 1) {
                long long value = send(e.to, sink, min(amount, e.capacity));
                if (value) {
                    e.capacity -= value;
                    graph[e.to][e.reverse].capacity += value;
                    return value;
                }
            }
        }
        return 0;
    }
    long long maximum(int source, int sink, long long limit = LLONG_MAX / 4) {
        long long total = 0;
        while (total < limit && bfs(source, sink)) {
            fill(current.begin(), current.end(), 0);
            while (total < limit) {
                long long value = send(source, sink, limit - total);
                if (!value)
                    break;
                total += value;
            }
        }
        return total;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int rows, cols;
    cin >> rows >> cols;
    int cells = rows * cols, source = cells, sink = cells + 1;
    Dinic flow(cells + 2);
    vector<long long> weight(cells);
    long long total = 0;
    for (auto &x : weight) {
        cin >> x;
        total += x;
    }
    const int dr[] = {-1, 1, 0, 0}, dc[] = {0, 0, -1, 1};
    for (int r = 0; r < rows; r++)
        for (int c = 0; c < cols; c++) {
            int u = r * cols + c;
            if ((r + c) % 2 == 0) {
                flow.add(source, u, weight[u]);
                for (int d = 0; d < 4; d++) {
                    int rr = r + dr[d], cc = c + dc[d];
                    if (rr >= 0 && rr < rows && cc >= 0 && cc < cols)
                        flow.add(u, rr * cols + cc, total + 1);
                }
            } else
                flow.add(u, sink, weight[u]);
        }
    cout << total - flow.maximum(source, sink);
    return 0;
}
```

### 复杂度

一般Dinic最坏 $O(V²E)$，网格E=O(V)，空间 $O(V)$。

### 易错点

无穷容量取总权+1即可，不必取会溢出的极大值。答案是总权减割，不是割本身。
