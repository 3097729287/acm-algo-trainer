## 求和

### 题意

固定根树单点加，查询子树权值和，n至百万。

### 思路

深度优先序中，一个节点的整棵子树恰是连续区间[in[u],out[u]]。用显式栈生成DFS先序，逆序累加子树大小即可得到out。按in把点权放入树状数组，点修改为单点加，子树查询为两个前缀和之差。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct EulerTree {
    int n;
    vector<vector<int>> edge;
    vector<int> parent, depth, in, out, order;
    EulerTree(int n) : n(n), edge(n + 1), parent(n + 1), depth(n + 1), in(n + 1), out(n + 1) {}
    void add(int u, int v) {
        edge[u].push_back(v);
        edge[v].push_back(u);
    }
    void root(int start) {
        vector<int> stack = {start};
        parent[start] = 0;
        order.clear();
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            in[u] = order.size() + 1;
            order.push_back(u);
            for (int v : edge[u])
                if (v != parent[u]) {
                    parent[v] = u;
                    depth[v] = depth[u] + 1;
                    stack.push_back(v);
                }
        }
        vector<int> size(n + 1, 1);
        for (int i = (int)order.size() - 1; i >= 0; i--) {
            int u = order[i];
            out[u] = in[u] + size[u] - 1;
            if (parent[u])
                size[parent[u]] += size[u];
        }
    }
};

struct Fenwick {
    int n;
    vector<long long> tree;
    Fenwick(int n) : n(n), tree(n + 1) {}
    void add(int p, long long v) {
        for (; p <= n; p += p & -p)
            tree[p] += v;
    }
    long long sum(int p) {
        long long result = 0;
        for (; p; p -= p & -p)
            result += tree[p];
        return result;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m, k;
    cin >> n >> m >> k;
    vector<long long> value(n + 1);
    for (int i = 1; i <= n; i++)
        cin >> value[i];
    EulerTree tree(n);
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        tree.add(u, v);
    }
    tree.root(k);
    Fenwick bit(n);
    for (int u = 1; u <= n; u++)
        bit.add(tree.in[u], value[u]);
    while (m--) {
        int op, u;
        cin >> op >> u;
        if (op == 1) {
            long long x;
            cin >> x;
            bit.add(tree.in[u], x);
        } else
            cout << bit.sum(tree.out[u]) - bit.sum(tree.in[u] - 1) << '\n';
    }
    return 0;
}
```

### 复杂度

预处理 $O(n\log n)$，操作 $O(\log n)$，空间 $O(n)$。

### 易错点

根是输入k，不一定为1。点权与增量可能为负；所有和使用64位。
