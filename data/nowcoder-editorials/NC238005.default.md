## Master Spark

### 题意

在矩形0≤x≤X、0≤y≤Y内，判断是否存在不落入任何闭条带C≤Ax+By≤D的点。

### 思路

本地原题正文漏掉了区域的数学定义，上述题意依据[原比赛题面](https://codeforces.com/gym/103447/problem/F)补全。条带包含边界，目标是严格避开全部条带。

若存在安全点，安全区域作为有限个闭条带的补集，具有一小块二维开区域。该区域边界必有一段来自矩形边或某条带边界的线段。于是枚举4条矩形边（向内侧移动）和每个条带两条边（向条带外侧移动），寻找某段边界，沿指定侧无限小移动后仍安全。

固定候选直线Ax+By=C，用x作参数（竖直线改用y），先截取矩形内的参数区间。其他非平行条带在线上覆盖一个闭区间，把这些区间求并；有正长度空隙，就可从空隙中点向指定侧移动。平行条带单独处理：若整条线在其内部，或恰在边界且指定移动方向进入该条带，候选整条作废；零宽条带不阻止向任一侧离开。若候选线在矩形边上，还要保证指定方向进入矩形。

交点参数全部保存为整数分子、正分母，用128位交叉相乘比较，因此即使安全区域很窄，也不会因浮点面积消失。仅有单点空隙不算安全段，区间端点相接时没有间隙。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

using Wide = __int128_t;
struct Fraction {
    long long a, b;
    Fraction(long long x = 0, long long y = 1) : a(x), b(y) {
        if (b < 0) {
            a = -a;
            b = -b;
        }
    }
    bool operator<(const Fraction &v) const { return (Wide)a * v.b < (Wide)v.a * b; }
    bool operator==(const Fraction &v) const { return (Wide)a * v.b == (Wide)v.a * b; }
};
struct Strip {
    long long a, b, c, d;
};
struct Boundary {
    long long a, b, c, sign;
};
bool uncovered(const Boundary &line, const vector<Strip> &strips, long long X, long long Y) {
    auto [a, b, c, sign] = line;
    Fraction low, high;
    if (b) {
        low = Fraction(0);
        high = Fraction(X);
        if (a) {
            Fraction x(c, a), y(c - b * Y, a);
            if (y < x)
                swap(x, y);
            low = max(low, x);
            high = min(high, y);
        } else {
            Fraction y(c, b);
            if (y < Fraction(0) || Fraction(Y) < y)
                return false;
        }
    } else {
        Fraction x(c, a);
        if (x < Fraction(0) || Fraction(X) < x)
            return false;
        low = Fraction(0);
        high = Fraction(Y);
    }
    if (!(low < high))
        return false;
    if (!b) {
        Fraction x(c, a);
        if ((x == Fraction(0) && sign * a < 0) || (x == Fraction(X) && sign * a > 0))
            return false;
    }
    if (!a) {
        Fraction y(c, b);
        if ((y == Fraction(0) && sign * b < 0) || (y == Fraction(Y) && sign * b > 0))
            return false;
    }
    vector<pair<Fraction, Fraction>> interval;
    long long norm = a * a + b * b;
    for (auto s : strips) {
        long long parallel = a * s.b - b * s.a;
        if (!parallel) {
            long long value = c * (a * s.a + b * s.b), lower = s.c * norm, upper = s.d * norm,
                      move = sign * (a * s.a + b * s.b);
            if (lower < value && value < upper)
                return false;
            if (s.c != s.d && ((value == lower && move > 0) || (value == upper && move < 0)))
                return false;
            continue;
        }
        Fraction l, r;
        if (b) {
            long long coefficient = s.a * b - s.b * a;
            l = Fraction(s.c * b - s.b * c, coefficient);
            r = Fraction(s.d * b - s.b * c, coefficient);
        } else {
            l = Fraction(s.c * a - s.a * c, s.b * a);
            r = Fraction(s.d * a - s.a * c, s.b * a);
        }
        if (r < l)
            swap(l, r);
        l = max(l, low);
        r = min(r, high);
        if (l < r)
            interval.push_back({l, r});
    }
    sort(interval.begin(), interval.end(),
         [](const auto &x, const auto &y) { return x.first < y.first; });
    Fraction reach = low;
    for (auto [l, r] : interval) {
        if (reach < l)
            return true;
        reach = max(reach, r);
    }
    return reach < high;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int tests;
    cin >> tests;
    string answer;
    while (tests--) {
        long long X, Y;
        int n;
        cin >> X >> Y >> n;
        vector<Strip> strips(n);
        vector<Boundary> boundary = {{1, 0, 0, 1}, {1, 0, X, -1}, {0, 1, 0, 1}, {0, 1, Y, -1}};
        for (auto &s : strips) {
            cin >> s.a >> s.b >> s.c >> s.d;
            boundary.push_back({s.a, s.b, s.c, -1});
            boundary.push_back({s.a, s.b, s.d, 1});
        }
        bool found = false;
        for (auto l : boundary)
            if (uncovered(l, strips, X, Y)) {
                found = true;
                break;
            }
        answer.push_back(found ? '1' : '0');
    }
    cout << answer;
    return 0;
}
```

### 复杂度

每组时间 $O(n^2\log n)$，空间 $O(n)$；各组n总和≤2000。

### 易错点

“只看矩形四个角”会漏掉内部安全洞。只对条带并面积做浮点比较，也可能把极窄安全区误判为全覆盖。参数区间与平行侧别均需精确处理。
