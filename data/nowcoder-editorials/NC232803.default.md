## Ice Walls

### 题意

穿过每道互不相交的轴平行冰墙加k时间，求s到t的欧氏距离加穿墙代价最小值。

### 思路

最短路径的转折只需发生在冰墙端点：在其他位置弯折，可以局部拉直且不改变严格穿墙次数。建立s、t和全部墙端点为节点的完全图，直连边权=欧氏距离+k*严格穿过的墙数，再跑Dijkstra。关键是将逐边数墙的O(n³)降下来。

从一个源点出发，将射线绕一周扫描。每道不经过源点的墙占据一个小于π的开角区间，进入时加入、离开时删除。墙两两不相交，因此当前射线与各墙交点由近到远的顺序不会变化。用支持排名的平衡树维护交点距离；射线对准一个目标时，查询交点参数t<1的墙数。

端点角度相同时，先删除结束的墙，再查询目标，最后加入开始的墙，所以射线仅碰墙端点不算穿墙；与源点相接的墙也不计。交点参数用整数叉积组成的分数，比较时用128位相乘。每个Dijkstra选中的节点执行一次角度扫描，随即松弛其他节点，无需存整张完全图。代码使用GNU C++17的PBDS排名树。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

using Wide = __int128_t;
struct IPoint {
    long long x = 0, y = 0;
    IPoint operator+(IPoint b) const { return {x + b.x, y + b.y}; }
    IPoint operator-(IPoint b) const { return {x - b.x, y - b.y}; }
};
Wide cross(IPoint a, IPoint b) { return (Wide)a.x * b.y - (Wide)a.y * b.x; }
Wide dot(IPoint a, IPoint b) { return (Wide)a.x * b.x + (Wide)a.y * b.y; }
Wide absolute(Wide x) { return x < 0 ? -x : x; }
void printWide(Wide value) {
    if (value < 0) {
        cout << '-';
        value = -value;
    }
    string text;
    do {
        text.push_back('0' + value % 10);
        value /= 10;
    } while (value);
    reverse(text.begin(), text.end());
    cout << text;
}
bool angleLess(IPoint a, IPoint b) {
    auto half = [](IPoint p) { return p.y < 0 || (p.y == 0 && p.x < 0); };
    if (half(a) != half(b))
        return half(a) < half(b);
    return cross(a, b) > 0;
}
vector<IPoint> integerHull(vector<IPoint> point) {
    sort(point.begin(), point.end(),
         [](auto a, auto b) { return a.x != b.x ? a.x < b.x : a.y < b.y; });
    point.erase(
        unique(point.begin(), point.end(), [](auto a, auto b) { return a.x == b.x && a.y == b.y; }),
        point.end());
    if (point.size() < 2)
        return point;
    vector<IPoint> hull;
    for (auto p : point) {
        while (hull.size() >= 2 && cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    size_t lower = hull.size();
    for (int i = (int)point.size() - 2; i >= 0; i--) {
        auto p = point[i];
        while (hull.size() > lower &&
               cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    hull.pop_back();
    return hull;
}

#include <ext/pb_ds/assoc_container.hpp>
#include <ext/pb_ds/tree_policy.hpp>
using namespace __gnu_pbds;
struct Wall {
    IPoint a, b;
};
vector<Wall> wall;
IPoint origin, direction;
pair<Wide, Wide> rayTime(int id) {
    if (id < 0)
        return {1, 1};
    IPoint edge = wall[id].b - wall[id].a;
    Wide a = cross(wall[id].a - origin, edge), b = cross(direction, edge);
    if (b < 0) {
        a = -a;
        b = -b;
    }
    return {a, b};
}
struct RayOrder {
    bool operator()(int a, int b) const {
        auto x = rayTime(a), y = rayTime(b);
        Wide left = x.first * y.second, right = y.first * x.second;
        return left != right ? left < right : a < b;
    }
};
using Active = tree<int, null_type, RayOrder, rb_tree_tag, tree_order_statistics_node_update>;
vector<int> crossings(const vector<IPoint> &point, int source) {
    origin = point[source];
    direction = {1, 0};
    struct Event {
        IPoint d;
        int kind, id;
    };
    vector<Event> event;
    vector<int> initial;
    for (int i = 0; i < (int)wall.size(); i++) {
        IPoint a = wall[i].a - origin, b = wall[i].b - origin;
        if ((a.x == 0 && a.y == 0) || (b.x == 0 && b.y == 0))
            continue;
        Wide c = cross(a, b);
        if (!c)
            continue;
        if (c < 0)
            swap(a, b);
        event.push_back({a, 2, i});
        event.push_back({b, 0, i});
        if (angleLess(b, a))
            initial.push_back(i);
    }
    for (int i = 0; i < (int)point.size(); i++)
        if (i != source) {
            IPoint d = point[i] - origin;
            if (d.x || d.y)
                event.push_back({d, 1, i});
        }
    sort(event.begin(), event.end(), [](const Event &a, const Event &b) {
        if (angleLess(a.d, b.d))
            return true;
        if (angleLess(b.d, a.d))
            return false;
        return a.kind != b.kind ? a.kind < b.kind : a.id < b.id;
    });
    Active active;
    for (int id : initial)
        active.insert(id);
    vector<int> result(point.size());
    for (auto e : event) {
        direction = e.d;
        if (e.kind == 0)
            active.erase(e.id);
        else if (e.kind == 2)
            active.insert(e.id);
        else
            result[e.id] = active.order_of_key(-1);
    }
    return result;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int tests;
    cin >> tests;
    while (tests--) {
        int n;
        long long penalty;
        cin >> n >> penalty;
        wall.resize(n);
        vector<IPoint> point;
        for (auto &s : wall) {
            cin >> s.a.x >> s.a.y >> s.b.x >> s.b.y;
            point.push_back(s.a);
            point.push_back(s.b);
        }
        IPoint s, t;
        cin >> s.x >> s.y >> t.x >> t.y;
        point.push_back(s);
        point.push_back(t);
        int count = point.size(), source = count - 2, target = count - 1;
        vector<long double> dist(count, 1e100L);
        vector<bool> used(count);
        dist[source] = 0;
        for (int step = 0; step < count; step++) {
            int v = -1;
            for (int i = 0; i < count; i++)
                if (!used[i] && (v < 0 || dist[i] < dist[v]))
                    v = i;
            if (v == target)
                break;
            used[v] = true;
            auto crossed = crossings(point, v);
            for (int u = 0; u < count; u++)
                if (!used[u]) {
                    IPoint d = point[u] - point[v];
                    long double value =
                        dist[v] + sqrtl((long double)dot(d, d)) + (long double)penalty * crossed[u];
                    dist[u] = min(dist[u], value);
                }
        }
        cout << fixed << setprecision(12) << dist[target] << '\n';
    }
    return 0;
}
```

### 复杂度

节点数V=2n+2，总时间 $O(Vn\log n+V^2)$，空间 $O(V+n)$。

### 易错点

经过端点和沿墙移动均不收穿墙代价。把角区间设为闭区间会使经过端点的最短路多收k。不可用不支持排名的普通set再线性数前驱。
