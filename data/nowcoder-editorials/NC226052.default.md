## 智乃酱的双塔问题·极（带修改的DP，DDP）

### 题意

两塔的楼梯方向与长度可改，查询从一层一侧到更高层指定侧的最短路。

### 思路

每层到上一层只涉及左右两个状态，建二乘二转移矩阵：同塔上楼分别填左、右楼梯长度；斜楼梯按方向填入 (左,右) 或 (右,左)，不存在的转移记无穷。连续两段组合使用最小加法乘积 C[i][j]=min_k(A[i][k]+B[k][j])，对应枚举中间层所在侧，具有结合律。线段树维护从下向上的有序矩阵积，改单层只更新一个叶子，查询 hs 到 ht 就取 [hs,ht-1] 的积。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

const long long Empty = 4000000000000000000LL;
struct Matrix {
    long long value[2][2];
    Matrix() {
        for (auto &row : value)
            for (auto &x : row)
                x = Empty;
    }
};
Matrix identity() {
    Matrix a;
    a.value[0][0] = a.value[1][1] = 0;
    return a;
}
Matrix product(const Matrix &a, const Matrix &b) {
    Matrix result;
    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
            for (int k = 0; k < 2; k++) {
                if (a.value[i][k] < Empty && b.value[k][j] < Empty)
                    result.value[i][j] = min(result.value[i][j], a.value[i][k] + b.value[k][j]);
            }
    return result;
}
struct MatrixTree {
    int base = 1;
    vector<Matrix> tree;
    explicit MatrixTree(const vector<Matrix> &a) {
        while (base < (int)a.size())
            base *= 2;
        tree.assign(base * 2, identity());
        for (int i = 0; i < (int)a.size(); i++)
            tree[base + i] = a[i];
        for (int i = base - 1; i > 0; i--)
            tree[i] = product(tree[i * 2], tree[i * 2 + 1]);
    }
    void change(int p, Matrix a) {
        p += base;
        tree[p] = a;
        while (p > 1) {
            p /= 2;
            tree[p] = product(tree[p * 2], tree[p * 2 + 1]);
        }
    }
    Matrix query(int l, int r) {
        Matrix left = identity(), right = identity();
        for (l += base, r += base; l < r; l /= 2, r /= 2) {
            if (l & 1)
                left = product(left, tree[l++]);
            if (r & 1)
                right = product(tree[--r], right);
        }
        return product(left, right);
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    string direction;
    cin >> n >> m >> direction;
    vector<array<long long, 3>> cost(n);
    for (int h = 1; h < n; h++)
        cin >> cost[h][0] >> cost[h][1] >> cost[h][2];
    auto leaf = [&](int h) {
        Matrix a;
        a.value[0][0] = cost[h][0];
        a.value[1][1] = cost[h][1];
        if (direction[h - 1] == '/')
            a.value[0][1] = cost[h][2];
        else
            a.value[1][0] = cost[h][2];
        return a;
    };
    vector<Matrix> initial;
    initial.reserve(n - 1);
    for (int h = 1; h < n; h++)
        initial.push_back(leaf(h));
    MatrixTree tree(initial);
    while (m--) {
        int op;
        cin >> op;
        if (op == 0) {
            int h;
            char c;
            cin >> h >> c;
            direction[h - 1] = c;
            tree.change(h - 1, leaf(h));
        } else if (op == 1) {
            int h;
            array<long long, 3> values;
            cin >> h >> values[0] >> values[1] >> values[2];
            if (h < n) {
                cost[h] = values;
                tree.change(h - 1, leaf(h));
            }
        } else {
            int hs, ht, ps, pt;
            cin >> hs >> ht >> ps >> pt;
            long long answer = tree.query(hs - 1, ht - 1).value[ps][pt];
            cout << (answer == Empty ? -1 : answer) << '\n';
        }
    }
    return 0;
}
```

### 复杂度

每次操作 $O(2^3\log n)$，空间 $O(n)$。

### 易错点

矩阵积顺序不能颠倒；迭代查询用左右两个累积矩阵分别拼接。顶层没有向上楼梯，若输入修改顶层长度则读完参数后忽略。无穷项不参与加法，防止伪造可达路径。
