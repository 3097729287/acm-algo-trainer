## 拆路

### 题意

动态删边并查询所在连通块最大繁荣度。

### 思路

读完全部操作，先用从未被删除的边建最终图。逆序处理，原删除变成加边，原查询则直接查当前并查集根的最大繁荣度。合并时同步取块最大值。逆序得到的查询答案最后反转输出。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct DSU {
    vector<int> parent, size;
    DSU(int n) : parent(n + 1), size(n + 1, 1) { iota(parent.begin(), parent.end(), 0); }
    int find(int x) {
        while (parent[x] != x) {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        return x;
    }
    bool unite(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b)
            return false;
        if (size[a] < size[b])
            swap(a, b);
        parent[b] = a;
        size[a] += size[b];
        return true;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    vector<int> value(n);
    for (int &x : value)
        cin >> x;
    vector<pair<int, int>> edge(m);
    map<pair<int, int>, int> index;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        u--;
        v--;
        if (u > v)
            swap(u, v);
        edge[i] = {u, v};
        index[edge[i]] = i;
    }
    int q;
    cin >> q;
    struct Operation {
        char type;
        int x, y;
    };
    vector<Operation> operation(q);
    vector<bool> removed(m);
    for (auto &op : operation) {
        cin >> op.type >> op.x;
        op.x--;
        op.y = -1;
        if (op.type == 'D') {
            cin >> op.y;
            op.y--;
            if (op.x > op.y)
                swap(op.x, op.y);
            removed[index[{op.x, op.y}]] = true;
        }
    }
    DSU dsu(n);
    auto unite = [&](int u, int v) {
        int a = dsu.find(u), b = dsu.find(v), maximum = max(value[a], value[b]);
        dsu.unite(a, b);
        value[dsu.find(a)] = maximum;
    };
    for (int i = 0; i < m; i++)
        if (!removed[i])
            unite(edge[i].first, edge[i].second);
    vector<int> answer;
    for (int i = q - 1; i >= 0; i--) {
        auto op = operation[i];
        if (op.type == 'D')
            unite(op.x, op.y);
        else
            answer.push_back(value[dsu.find(op.x)]);
    }
    reverse(answer.begin(), answer.end());
    for (int x : answer)
        cout << x << '\n';
    return 0;
}
```

### 复杂度

时间 $O((n+m+q)\log n)$，空间 $O(n+m+q)$。

### 易错点

无向边统一较小端在前，避免删除时找不到反向输入的边。查询答案要恢复原时间顺序。
