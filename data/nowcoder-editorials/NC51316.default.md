## Going Home

### 题意

每人各进不同房屋，最小曼哈顿移动总费用。

### 思路

地图没有障碍，每个人到每间房的最短路为曼哈顿距离。建立源→人容量1、人→房容量1费用距离、房→汇容量1的费用流图。逐次最短增广得到n单位流的最小费用；反向边费用取负，允许撤回之前指派来改进组合。代码用SPFA在残量图找最短路。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct CostFlow {
    struct Edge {
        int to, reverse;
        long long capacity, cost;
    };
    vector<vector<Edge>> graph;
    explicit CostFlow(int n) : graph(n) {}
    void add(int u, int v, long long capacity, long long cost) {
        int a = graph[u].size(), b = graph[v].size();
        graph[u].push_back({v, b, capacity, cost});
        graph[v].push_back({u, a, 0, -cost});
    }
    pair<long long, long long> minimum(int source, int sink) {
        const long long inf = LLONG_MAX / 4;
        long long flow = 0, cost = 0;
        int n = graph.size();
        while (true) {
            vector<long long> distance(n, inf);
            vector<int> parent(n, -1), index(n);
            vector<bool> inQueue(n);
            queue<int> queue;
            queue.push(source);
            inQueue[source] = true;
            distance[source] = 0;
            while (!queue.empty()) {
                int u = queue.front();
                queue.pop();
                inQueue[u] = false;
                for (int i = 0; i < (int)graph[u].size(); i++) {
                    auto e = graph[u][i];
                    if (e.capacity > 0 && distance[e.to] > distance[u] + e.cost) {
                        distance[e.to] = distance[u] + e.cost;
                        parent[e.to] = u;
                        index[e.to] = i;
                        if (!inQueue[e.to]) {
                            inQueue[e.to] = true;
                            queue.push(e.to);
                        }
                    }
                }
            }
            if (parent[sink] < 0)
                break;
            long long amount = inf;
            for (int v = sink; v != source; v = parent[v])
                amount = min(amount, graph[parent[v]][index[v]].capacity);
            for (int v = sink; v != source; v = parent[v]) {
                auto &e = graph[parent[v]][index[v]];
                e.capacity -= amount;
                graph[v][e.reverse].capacity += amount;
            }
            flow += amount;
            cost += amount * distance[sink];
        }
        return {flow, cost};
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int rows, cols;
    while (cin >> rows >> cols && (rows || cols)) {
        vector<pair<int, int>> people, house;
        for (int r = 0; r < rows; r++) {
            string s;
            cin >> s;
            for (int c = 0; c < cols; c++) {
                if (s[c] == 'm')
                    people.push_back({r, c});
                if (s[c] == 'H')
                    house.push_back({r, c});
            }
        }
        int n = people.size(), source = 2 * n, sink = source + 1;
        CostFlow flow(sink + 1);
        for (int i = 0; i < n; i++) {
            flow.add(source, i, 1, 0);
            flow.add(n + i, sink, 1, 0);
            for (int j = 0; j < n; j++)
                flow.add(i, n + j, 1,
                         abs(people[i].first - house[j].first) +
                             abs(people[i].second - house[j].second));
        }
        cout << flow.minimum(source, sink).second << '\n';
    }
    return 0;
}
```

### 复杂度

V=2n+2、E=O(n²)，最坏SPFA增广 $O(nVE)$，空间 $O(E)$。

### 易错点

可以穿过其他房屋而不入住，也可多人经过同一格，不添加网格占用冲突。
