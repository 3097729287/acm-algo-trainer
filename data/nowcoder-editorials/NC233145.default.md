## Illuminations II

### 题意

外凸多边形边界随机放灯，内凸多边形被照亮周长期望。

### 思路

固定内多边形的一条逆时针边AB，外部光源在其右侧时这条边整条可见，在左侧时被内多边形挡住。因此该边期望贡献=边长*外边界位于其右半平面的长度/外周长，各边线性相加。

AB所在直线穿过外多边形内部，边界恰分为左右两条连续弧。用法向量支撑查询找到正侧、负侧极值顶点，再分别在两条顶点链上二分符号变化，得到两个交点所在边及线性插值位置。外周长前缀和将交点间右侧弧长化为常数求和。整数叉积判断侧别，只有交点比例与周长用long double。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

using Wide = __int128_t;
struct IPoint {
    long long x = 0, y = 0;
    IPoint operator+(IPoint b) const { return {x + b.x, y + b.y}; }
    IPoint operator-(IPoint b) const { return {x - b.x, y - b.y}; }
};
Wide cross(IPoint a, IPoint b) { return (Wide)a.x * b.y - (Wide)a.y * b.x; }
Wide dot(IPoint a, IPoint b) { return (Wide)a.x * b.x + (Wide)a.y * b.y; }
Wide absolute(Wide x) { return x < 0 ? -x : x; }
void printWide(Wide value) {
    if (value < 0) {
        cout << '-';
        value = -value;
    }
    string text;
    do {
        text.push_back('0' + value % 10);
        value /= 10;
    } while (value);
    reverse(text.begin(), text.end());
    cout << text;
}
bool angleLess(IPoint a, IPoint b) {
    auto half = [](IPoint p) { return p.y < 0 || (p.y == 0 && p.x < 0); };
    if (half(a) != half(b))
        return half(a) < half(b);
    return cross(a, b) > 0;
}
vector<IPoint> integerHull(vector<IPoint> point) {
    sort(point.begin(), point.end(),
         [](auto a, auto b) { return a.x != b.x ? a.x < b.x : a.y < b.y; });
    point.erase(
        unique(point.begin(), point.end(), [](auto a, auto b) { return a.x == b.x && a.y == b.y; }),
        point.end());
    if (point.size() < 2)
        return point;
    vector<IPoint> hull;
    for (auto p : point) {
        while (hull.size() >= 2 && cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    size_t lower = hull.size();
    for (int i = (int)point.size() - 2; i >= 0; i--) {
        auto p = point[i];
        while (hull.size() > lower &&
               cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    hull.pop_back();
    return hull;
}

struct Support {
    const vector<IPoint> &point;
    vector<pair<IPoint, int>> normal;
    explicit Support(const vector<IPoint> &p) : point(p) {
        if (p.size() < 3)
            return;
        for (int i = 0; i < (int)p.size(); i++) {
            IPoint e = p[(i + 1) % p.size()] - p[i];
            normal.push_back({{e.y, -e.x}, i});
        }
        sort(normal.begin(), normal.end(),
             [](auto a, auto b) { return angleLess(a.first, b.first); });
    }
    int index(IPoint direction) const {
        if (point.size() < 3) {
            int best = 0;
            for (int i = 1; i < (int)point.size(); i++)
                if (dot(point[i], direction) > dot(point[best], direction))
                    best = i;
            return best;
        }
        auto it = lower_bound(normal.begin(), normal.end(), direction,
                              [](const auto &a, IPoint b) { return angleLess(a.first, b); });
        if (it == normal.end())
            it = normal.begin();
        return it->second;
    }
};
int visibleEdge(const vector<IPoint> &p, IPoint q) {
    int n = p.size();
    if (cross(p[1] - p[0], q - p[0]) < 0)
        return 0;
    if (cross(p[n - 1] - p[0], q - p[0]) > 0)
        return n - 1;
    int l = 1, r = n - 1;
    while (r - l > 1) {
        int mid = (l + r) / 2;
        if (cross(p[mid] - p[0], q - p[0]) >= 0)
            l = mid;
        else
            r = mid;
    }
    return cross(p[r] - p[l], q - p[l]) < 0 ? l : -1;
}
bool insideConvex(const vector<IPoint> &p, IPoint q, bool strict) {
    int n = p.size();
    if (n == 1)
        return !strict && q.x == p[0].x && q.y == p[0].y;
    if (n == 2)
        return !strict && cross(p[1] - p[0], q - p[0]) == 0 && dot(q - p[0], q - p[1]) <= 0;
    Wide a = cross(p[1] - p[0], q - p[0]), b = cross(p[n - 1] - p[0], q - p[0]);
    if (a < 0 || b > 0)
        return false;
    if (a == 0)
        return !strict && dot(q - p[0], q - p[1]) <= 0;
    if (b == 0)
        return !strict && dot(q - p[0], q - p[n - 1]) <= 0;
    int l = 1, r = n - 1;
    while (r - l > 1) {
        int mid = (l + r) / 2;
        if (cross(p[mid] - p[0], q - p[0]) >= 0)
            l = mid;
        else
            r = mid;
    }
    Wide side = cross(p[r] - p[l], q - p[l]);
    return strict ? side > 0 : side >= 0;
}
vector<IPoint> minkowski(vector<IPoint> a, vector<IPoint> b) {
    if (a.size() < 3 || b.size() < 3) {
        vector<IPoint> sum;
        for (auto p : a)
            for (auto q : b)
                sum.push_back(p + q);
        return integerHull(sum);
    }
    auto prepare = [](vector<IPoint> &p) {
        auto it = min_element(p.begin(), p.end(),
                              [](auto a, auto b) { return a.y != b.y ? a.y < b.y : a.x < b.x; });
        rotate(p.begin(), it, p.end());
    };
    prepare(a);
    prepare(b);
    int n = a.size(), m = b.size(), i = 0, j = 0;
    vector<IPoint> result;
    IPoint current = a[0] + b[0];
    result.push_back(current);
    while (i < n || j < m) {
        IPoint x = i < n ? a[(i + 1) % n] - a[i] : IPoint{},
               y = j < m ? b[(j + 1) % m] - b[j] : IPoint{};
        Wide c = cross(x, y);
        if (j == m || (i < n && c > 0)) {
            current = current + x;
            i++;
        } else if (i == n || c < 0) {
            current = current + y;
            j++;
        } else {
            current = current + x + y;
            i++;
            j++;
        }
        result.push_back(current);
    }
    result.pop_back();
    return result;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    vector<IPoint> outer(n), inner(m);
    for (auto &p : outer)
        cin >> p.x >> p.y;
    for (auto &p : inner)
        cin >> p.x >> p.y;
    Support support(outer);
    vector<long double> length(n), prefix(n + 1);
    for (int i = 0; i < n; i++) {
        auto e = outer[(i + 1) % n] - outer[i];
        length[i] = sqrtl((long double)dot(e, e));
        prefix[i + 1] = prefix[i] + length[i];
    }
    long double perimeter = prefix[n], answer = 0;
    for (int i = 0; i < m; i++) {
        IPoint a = inner[i], e = inner[(i + 1) % m] - a, normal = {-e.y, e.x},
               negative = {e.y, -e.x};
        int positive = support.index(normal), minus = support.index(negative);
        auto side = [&](int vertex) { return cross(e, outer[vertex] - a); };
        auto crossing = [&](int start, int finish, bool sign) {
            int low = 0, high = (finish - start + n) % n;
            while (high - low > 1) {
                int mid = (low + high) / 2;
                if ((side((start + mid) % n) >= 0) == sign)
                    low = mid;
                else
                    high = mid;
            }
            int u = (start + low) % n, v = (u + 1) % n;
            Wide su = side(u), sv = side(v);
            long double ratio = (long double)su / (long double)(su - sv);
            return prefix[u] + ratio * length[u];
        };
        long double enter = crossing(positive, minus, true),
                    leave = crossing(minus, positive, false), arc = leave - enter;
        if (arc < 0)
            arc += perimeter;
        answer += sqrtl((long double)dot(e, e)) * arc / perimeter;
    }
    cout << fixed << setprecision(15) << answer;
    return 0;
}
```

### 复杂度

预处理 $O(n\log n)$，全部内边 $O(m\log n)$，空间 $O(n+m)$。

### 易错点

光源按外边界弧长均匀，不是外顶点均匀。内边整条可见与否由其支撑直线侧别决定，无需逐点数值积分。
