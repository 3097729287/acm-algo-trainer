## 三轮

### 题意

n种商品每种十万件，统计体积1到m的装载方案总数，模19260817。

### 思路

m≤50000，每种十万件的上限不会用完，等价于完全背包。设体积d有c_d种不同商品，生成函数$F(x)=\prod_d(1-x^d)^{-c_d}$。对数求导得到$xF^\prime(x)=F(x)H(x)$，其中$H_i=\sum_{d\mid i}d\,c_d$。故$f_0=1,\quad f_i=i^{-1}\sum_{j=0}^{i-1}f_jH_{i-j}$。答案是f1到fm之和。

用CDQ按下标求递推：先算完左半f，再把“左半f卷积H”的系数累加到右半待定和，最后递归右半。短区间直接转移。模数19260817不支持需要长度的NTT，分别在998244353、1004535809、469762049下卷积，再用三模CRT还原整数系数并对目标模数取模。三模乘积大于每次整数卷积可能的最大系数，128位足够承载CRT。

实现把递归范围补成2的幂N，卷积只做长度N的循环卷积：左半多项式长度N/2，溢出到N以上的项只回绕到下标小于N/2的左半，不会污染需要的右半系数。这样每次只做N点NTT。超过m的叶子不再计算。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

constexpr int MOD = 19260817;
int power(long long a, int e, int mod) {
    long long answer = 1;
    for (; e; e >>= 1, a = a * a % mod)
        if (e & 1)
            answer = answer * a % mod;
    return answer;
}
void transform(vector<int> &a, bool inverse, int mod) {
    int n = a.size();
    for (int i = 1, j = 0; i < n; i++) {
        int bit = n >> 1;
        for (; j & bit; bit >>= 1)
            j ^= bit;
        j ^= bit;
        if (i < j)
            swap(a[i], a[j]);
    }
    for (int len = 2; len <= n; len <<= 1) {
        int root = power(3, (mod - 1) / len, mod);
        if (inverse)
            root = power(root, mod - 2, mod);
        for (int i = 0; i < n; i += len) {
            long long w = 1;
            for (int j = 0; j < len / 2; j++, w = w * root % mod) {
                int u = a[i + j], v = w * a[i + j + len / 2] % mod;
                a[i + j] = u + v;
                if (a[i + j] >= mod)
                    a[i + j] -= mod;
                a[i + j + len / 2] = u - v;
                if (a[i + j + len / 2] < 0)
                    a[i + j + len / 2] += mod;
            }
        }
    }
    if (inverse) {
        int scale = power(n, mod - 2, mod);
        for (int &x : a)
            x = (long long)x * scale % mod;
    }
}
vector<int> cyclicProduct(const vector<int> &a, const vector<int> &b, int n) {
    const int p1 = 998244353, p2 = 1004535809, p3 = 469762049;
    vector<vector<int>> result;
    for (int mod : {p1, p2, p3}) {
        vector<int> x(n), y(n);
        copy(a.begin(), a.end(), x.begin());
        copy(b.begin(), b.end(), y.begin());
        transform(x, false, mod);
        transform(y, false, mod);
        for (int i = 0; i < n; i++)
            x[i] = (long long)x[i] * y[i] % mod;
        transform(x, true, mod);
        result.push_back(move(x));
    }
    static const int inverse12 = power(p1, p2 - 2, p2),
                     inverse123 = power((long long)p1 * p2 % p3, p3 - 2, p3);
    vector<int> value(n);
    for (int i = 0; i < n; i++) {
        long long a1 = result[0][i], a2 = (result[1][i] - a1 + p2) % p2 * inverse12 % p2,
                  x = a1 + (long long)p1 * a2;
        long long a3 = (result[2][i] - x % p3 + p3) % p3 * inverse123 % p3;
        __int128 full = x + (__int128)p1 * p2 * a3;
        value[i] = full % MOD;
    }
    return value;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    int size = 1;
    while (size <= m)
        size <<= 1;
    vector<int> count(m + 1), kernel(size), f(size), sum(size), inverse(size);
    for (int i = 0; i < n; i++) {
        int v;
        cin >> v;
        if (v <= m)
            count[v]++;
    }
    for (int d = 1; d <= m; d++)
        if (count[d])
            for (int j = d; j <= m; j += d)
                kernel[j] = (kernel[j] + (long long)d * count[d]) % MOD;
    inverse[1] = 1;
    for (int i = 2; i < size; i++)
        inverse[i] = MOD - (long long)(MOD / i) * inverse[MOD % i] % MOD;
    f[0] = 1;
    function<void(int, int)> solve = [&](int l, int r) {
        if (l > m)
            return;
        if (r - l <= 32) {
            for (int i = l; i < min(r, m + 1); i++) {
                if (i)
                    f[i] = (long long)sum[i] * inverse[i] % MOD;
                for (int j = i + 1; j < min(r, m + 1); j++)
                    sum[j] = (sum[j] + (long long)f[i] * kernel[j - i]) % MOD;
            }
            return;
        }
        int mid = (l + r) / 2;
        solve(l, mid);
        vector<int> a(f.begin() + l, f.begin() + mid), b(kernel.begin(), kernel.begin() + r - l);
        auto product = cyclicProduct(a, b, r - l);
        for (int j = mid; j < min(r, m + 1); j++) {
            sum[j] += product[j - l];
            if (sum[j] >= MOD)
                sum[j] -= MOD;
        }
        solve(mid, r);
    };
    solve(0, size);
    int answer = 0;
    for (int i = 1; i <= m; i++) {
        answer += f[i];
        if (answer >= MOD)
            answer -= MOD;
    }
    cout << answer;
    return 0;
}
```

### 复杂度

设N为大于m的最小2次幂，时间 $O(m\log m+N\log^2N)$，空间 $O(N+m)$。

### 易错点

同体积的不同商品仍是不同种类，必须用c_d指数计数。最终不含空方案f0。不能直接在19260817下套998244353的NTT根，也不能仅用两个NTT模数还原可能超过它们乘积的系数。
