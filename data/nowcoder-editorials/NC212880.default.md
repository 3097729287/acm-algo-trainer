## 线段树

### 题意

支持区间加乘，并查询区间中全部无序两两乘积之和，答案模素数 p。

### 思路

记区间和为 S、平方和为 Q，则两两乘积和为 (S²-Q)/2。区间仿射修改的维护公式为 S′=aS+b len、Q′=a²Q+2abS+b²len，可用懒线段树维护。

p 允许等于二，此时二没有模逆元。为统一处理所有 p，在线段树内部对 2p 取模：因为真实 S²-Q 总是偶数，知道它模 2p 的偶数余数，再用普通整数除二，就恰好得到真实答案模 p。2p 小于 2^32，两个余数相乘仍可放入 unsigned long long。输入可能为负，先规范成非负余数。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct AffineTree {
    using U = unsigned long long;
    struct Node {
        U sum = 0, squares = 0, multiply = 1, add = 0;
    };
    int n;
    U modulus;
    vector<Node> tree;
    AffineTree(const vector<U> &a, U mod) : n((int)a.size() - 1), modulus(mod), tree(4 * n + 4) {
        build(1, 1, n, a);
    }
    U plus(U a, U b) const { return modulus ? (a + b) % modulus : a + b; }
    U times(U a, U b) const { return modulus ? a * b % modulus : a * b; }
    U length(int len) const { return modulus ? (U)len % modulus : (U)len; }
    void pull(int p) {
        tree[p].sum = plus(tree[p * 2].sum, tree[p * 2 + 1].sum);
        tree[p].squares = plus(tree[p * 2].squares, tree[p * 2 + 1].squares);
    }
    void build(int p, int l, int r, const vector<U> &a) {
        if (l == r) {
            tree[p].sum = a[l];
            tree[p].squares = times(a[l], a[l]);
            return;
        }
        int mid = (l + r) / 2;
        build(p * 2, l, mid, a);
        build(p * 2 + 1, mid + 1, r, a);
        pull(p);
    }
    void apply(int p, int len, U a, U b) {
        U old = tree[p].sum;
        tree[p].squares =
            plus(plus(times(times(a, a), tree[p].squares), times(times(times(2, a), b), old)),
                 times(times(b, b), length(len)));
        tree[p].sum = plus(times(a, old), times(b, length(len)));
        tree[p].multiply = times(tree[p].multiply, a);
        tree[p].add = plus(times(tree[p].add, a), b);
    }
    void push(int p, int l, int r) {
        if (tree[p].multiply == 1 && tree[p].add == 0)
            return;
        int mid = (l + r) / 2;
        apply(p * 2, mid - l + 1, tree[p].multiply, tree[p].add);
        apply(p * 2 + 1, r - mid, tree[p].multiply, tree[p].add);
        tree[p].multiply = 1;
        tree[p].add = 0;
    }
    void change(int p, int l, int r, int ql, int qr, U a, U b) {
        if (ql <= l && r <= qr) {
            apply(p, r - l + 1, a, b);
            return;
        }
        push(p, l, r);
        int mid = (l + r) / 2;
        if (ql <= mid)
            change(p * 2, l, mid, ql, qr, a, b);
        if (qr > mid)
            change(p * 2 + 1, mid + 1, r, ql, qr, a, b);
        pull(p);
    }
    pair<U, U> query(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr)
            return {tree[p].sum, tree[p].squares};
        push(p, l, r);
        int mid = (l + r) / 2;
        pair<U, U> result{0, 0};
        if (ql <= mid)
            result = query(p * 2, l, mid, ql, qr);
        if (qr > mid) {
            auto other = query(p * 2 + 1, mid + 1, r, ql, qr);
            result.first = plus(result.first, other.first);
            result.second = plus(result.second, other.second);
        }
        return result;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int tests;
    cin >> tests;
    while (tests--) {
        int n, m;
        long long p;
        cin >> n >> m >> p;
        unsigned long long modulus = 2ULL * p;
        auto normalize = [&](long long x) {
            long long value = x % (long long)modulus;
            if (value < 0)
                value += modulus;
            return (unsigned long long)value;
        };
        vector<unsigned long long> a(n + 1);
        for (int i = 1; i <= n; i++) {
            long long x;
            cin >> x;
            a[i] = normalize(x);
        }
        AffineTree tree(a, modulus);
        while (m--) {
            int op, l, r;
            cin >> op >> l >> r;
            if (op == 3) {
                auto [sum, squares] = tree.query(1, 1, n, l, r);
                auto difference = (tree.times(sum, sum) + modulus - squares) % modulus;
                cout << difference / 2 << '\n';
            } else {
                long long x;
                cin >> x;
                if (op == 1)
                    tree.change(1, 1, n, l, r, 1, normalize(x));
                else
                    tree.change(1, 1, n, l, r, normalize(x), 0);
            }
        }
    }
    return 0;
}
```

### 复杂度

每组时间 $O((n+m)\log n)$，空间 $O(n)$。

### 易错点

不能无条件用 (p+1)/2 当逆元，p=2 会错。内部取模数是 2p，最终的普通除二只做在规范后的偶数余数上。
