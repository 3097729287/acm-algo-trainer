## 普通平衡树

### 题意

支持重复值的插删、排名、第k小、严格前驱后继。

### 思路

FHQ Treap按键保持BST、按随机优先级保持堆。split将树拆成≤key与>key，merge合并键值范围有序的两棵树。每个不同键存重复次数count，size为含重复的子树大小。插删时两次split隔离该键；排名按左子树size累加，第k小按左右数量定位，前驱后继沿BST下降。

重复插入只增count，删除只减一份，减至0才合并两个儿子替换该节点。随机优先级使树高期望对数。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Treap {
    struct Node {
        int left = 0, right = 0, count = 1, size = 1;
        long long key = 0;
        unsigned priority = 0;
    };
    vector<Node> tree;
    mt19937 random{20261108};
    int root = 0;
    Treap(int n) {
        tree.reserve(n + 1);
        tree.emplace_back();
        tree[0].count = tree[0].size = 0;
    }
    void pull(int u) {
        if (u)
            tree[u].size = tree[tree[u].left].size + tree[tree[u].right].size + tree[u].count;
    }
    void split(int u, long long key, int &a, int &b) {
        if (!u) {
            a = b = 0;
            return;
        }
        if (tree[u].key <= key) {
            a = u;
            split(tree[u].right, key, tree[u].right, b);
            pull(a);
        } else {
            b = u;
            split(tree[u].left, key, a, tree[u].left);
            pull(b);
        }
    }
    int merge(int a, int b) {
        if (!a || !b)
            return a ? a : b;
        if (tree[a].priority < tree[b].priority) {
            tree[a].right = merge(tree[a].right, b);
            pull(a);
            return a;
        }
        tree[b].left = merge(a, tree[b].left);
        pull(b);
        return b;
    }
    void change(long long key, bool insert) {
        int a, b, c;
        split(root, key - 1, a, b);
        split(b, key, b, c);
        if (insert) {
            if (b) {
                tree[b].count++;
                pull(b);
            } else {
                b = tree.size();
                tree.emplace_back();
                tree[b].key = key;
                tree[b].priority = random();
            }
        } else if (b) {
            if (--tree[b].count == 0)
                b = merge(tree[b].left, tree[b].right);
            else
                pull(b);
        }
        root = merge(merge(a, b), c);
    }
    int rank(long long key) {
        int u = root, answer = 1;
        while (u) {
            if (tree[u].key < key) {
                answer += tree[tree[u].left].size + tree[u].count;
                u = tree[u].right;
            } else
                u = tree[u].left;
        }
        return answer;
    }
    long long kth(int k) {
        int u = root;
        while (u) {
            int left = tree[tree[u].left].size;
            if (k <= left)
                u = tree[u].left;
            else if (k <= left + tree[u].count)
                return tree[u].key;
            else {
                k -= left + tree[u].count;
                u = tree[u].right;
            }
        }
        return 0;
    }
    long long neighbor(long long key, bool successor) {
        int u = root;
        long long answer = 0;
        while (u) {
            if (successor ? tree[u].key > key : tree[u].key < key) {
                answer = tree[u].key;
                u = successor ? tree[u].left : tree[u].right;
            } else
                u = successor ? tree[u].right : tree[u].left;
        }
        return answer;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    Treap tree(n);
    while (n--) {
        int op;
        long long x;
        cin >> op >> x;
        if (op == 1 || op == 2)
            tree.change(x, op == 1);
        else if (op == 3)
            cout << tree.rank(x) << '\n';
        else if (op == 4)
            cout << tree.kth(x) << '\n';
        else
            cout << tree.neighbor(x, op == 6) << '\n';
    }
    return 0;
}
```

### 复杂度

每次期望 $O(\log n)$，极端优先级下最坏 $O(n)$；空间 $O(n)$。

### 易错点

严格前驱/后继不包含相等值。key用64位，split(key−1)不会在原32位边界溢出。题目查询需要存在合法答案。
