## Closest Pair of Segments

### 题意

互不相交的n条线段之间，求最近距离。

### 思路

两条线段最近点至少有一个是端点，因此两线段距离是四个“端点到另一线段距离”的最小值。为避免枚举全部线段对，二分距离d，把每条线段膨胀半径d/2，得到线段加圆盘的胶囊形；存在两胶囊相交，当且仅当原线段最近距离≤d。

每个胶囊是凸集，竖直截面是区间。按最左x−d/2插入、最右x+d/2删除，用有序集合维护竖直次序，每次只检查新增邻居的两胶囊是否相交。尚未相交时相对上下次序稳定；首次接触前必成为邻居。比较时可取原线段在扫描x处的y，x超出原线段范围时夹到最近端点；这个点总在胶囊截面内，足以确定两个不相交胶囊的次序，竖直线段取中点y。

插入序列与删除序列分别只需按原线段最左、最右x排序一次，每轮二分线性归并这两个事件序列。原线段互不相交是题目保证，所以距离计算无需再做线段相交分支。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

constexpr long double EPS = 1e-12L;
struct Point {
    long double x = 0, y = 0;
    Point operator+(Point b) const { return {x + b.x, y + b.y}; }
    Point operator-(Point b) const { return {x - b.x, y - b.y}; }
    Point operator*(long double t) const { return {x * t, y * t}; }
    Point operator/(long double t) const { return {x / t, y / t}; }
};
long double cross(Point a, Point b) { return a.x * b.y - a.y * b.x; }
long double dot(Point a, Point b) { return a.x * b.x + a.y * b.y; }
long double norm(Point a) { return sqrtl(dot(a, a)); }
long double distance(Point a, Point b) { return norm(a - b); }
Point rotate90(Point a) { return {-a.y, a.x}; }

struct Segment {
    Point a, b;
};
vector<Segment> segment;
long double sweepX, threshold;
long double height(int i) {
    auto [a, b] = segment[i];
    if (a.x == b.x)
        return (a.y + b.y) / 2;
    long double x = clamp(sweepX, a.x, b.x);
    return a.y + (b.y - a.y) * (x - a.x) / (b.x - a.x);
}
struct Order {
    bool operator()(int a, int b) const {
        long double x = height(a), y = height(b);
        return x != y ? x < y : a < b;
    }
};
long double pointDistance(Point p, const Segment &s) {
    Point e = s.b - s.a;
    long double t = clamp(dot(p - s.a, e) / dot(e, e), 0.0L, 1.0L);
    return distance(p, s.a + e * t);
}
long double segmentDistance(int i, int j) {
    auto a = segment[i], b = segment[j];
    return min({pointDistance(a.a, b), pointDistance(a.b, b), pointDistance(b.a, a),
                pointDistance(b.b, a)});
}
bool within(long double gap, const vector<int> &start, const vector<int> &finish) {
    int n = segment.size();
    long double radius = gap / 2;
    set<int, Order> active;
    vector<set<int, Order>::iterator> position(n);
    auto close = [&](int a, int b) { return segmentDistance(a, b) <= gap; };
    int i = 0, j = 0;
    while (i < n || j < n) {
        bool enter =
            j == n || (i < n && segment[start[i]].a.x - radius <= segment[finish[j]].b.x + radius);
        int id = enter ? start[i++] : finish[j++];
        sweepX = enter ? segment[id].a.x - radius : segment[id].b.x + radius;
        if (enter) {
            auto it = active.insert(id).first;
            position[id] = it;
            if (it != active.begin() && close(*prev(it), *it))
                return true;
            if (next(it) != active.end() && close(*it, *next(it)))
                return true;
        } else {
            auto it = position[id], right = next(it);
            if (it != active.begin() && right != active.end() && close(*prev(it), *right))
                return true;
            active.erase(it);
        }
    }
    return false;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int tests;
    cin >> tests;
    while (tests--) {
        int n;
        cin >> n;
        segment.resize(n);
        vector<int> start(n), finish(n);
        for (int i = 0; i < n; i++) {
            auto &s = segment[i];
            cin >> s.a.x >> s.a.y >> s.b.x >> s.b.y;
            if (s.a.x > s.b.x)
                swap(s.a, s.b);
            start[i] = finish[i] = i;
        }
        sort(start.begin(), start.end(),
             [](int a, int b) { return segment[a].a.x < segment[b].a.x; });
        sort(finish.begin(), finish.end(),
             [](int a, int b) { return segment[a].b.x < segment[b].b.x; });
        long double l = 0, r = segmentDistance(0, 1);
        for (int i = 1; i < min(n, 100); i++)
            r = min(r, segmentDistance(i - 1, i));
        for (int i = 0; i < 55; i++) {
            long double mid = (l + r) / 2;
            if (within(mid, start, finish))
                r = mid;
            else
                l = mid;
        }
        cout << fixed << setprecision(15) << r << '\n';
    }
    return 0;
}
```

### 复杂度

时间 $O(n\log n\log(C/\varepsilon))$，空间 $O(n)$。

### 易错点

只比较原始端点的最近点对会漏掉“端点到长线段内部”的距离。胶囊的排序代表点必须落在当前竖直截面内，不能无限外推原直线。
