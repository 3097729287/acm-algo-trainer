## RC Kaboom Show

### 题意

各车可延迟启动但只沿固定射线前进，求最早能安排碰撞的时间。

### 思路

给定结束时间T，第i辆车能到达射线上从起点到p_i+v_i*T的整条线段；能安排碰撞当且仅当某两条这样的线段相交。若车提前到达交点，可把启动时刻延后，让两车同时到达。可行性对T单调，用二分。

相交判定用竖直扫描线：按线段左右端点排序，维护当前线段的上下顺序，插入时检查新出现的两对邻居，删除时检查被连成新邻居的一对。若真有相交，在首次相交前线段次序不会改变，相交双方一定会先成为邻居，因此不会漏判。比较器随扫描x变化，仅在尚未发现交点的前提下维护有序结构；检查的是整条线段是否相交，发现后立即结束。

两车检查直接由原始整数射线计算最早碰撞时间，避免用巨大T生成的端点做不稳定叉积。非平行时求唯一射线交点，并取两车到达时间最大值；同向共线时后车追到前车起点，反向相向时双方同时开动，背向则无解。上界10^13超过本题整数坐标、方向参数能产生的有限交会时间。

扫描线思路可对照[原比赛讲解](https://codeforces.com/blog/entry/78116)。下面代码独立实现射线到达时间与端点事件。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

constexpr long double EPS = 1e-12L;
struct Point {
    long double x = 0, y = 0;
    Point operator+(Point b) const { return {x + b.x, y + b.y}; }
    Point operator-(Point b) const { return {x - b.x, y - b.y}; }
    Point operator*(long double t) const { return {x * t, y * t}; }
    Point operator/(long double t) const { return {x / t, y / t}; }
};
long double cross(Point a, Point b) { return a.x * b.y - a.y * b.x; }
long double dot(Point a, Point b) { return a.x * b.x + a.y * b.y; }
long double norm(Point a) { return sqrtl(dot(a, a)); }
long double distance(Point a, Point b) { return norm(a - b); }
Point rotate90(Point a) { return {-a.y, a.x}; }

struct Car {
    Point p, d, v;
    long double speed;
    long long x, y, dx, dy;
};
vector<Car> car;
vector<pair<Point, Point>> segment;
long double sweepX, duration;
long double height(int i) {
    auto [a, b] = segment[i];
    if (a.x > b.x)
        swap(a, b);
    long double x = clamp(sweepX, a.x, b.x);
    return a.y + (b.y - a.y) * (x - a.x) / (b.x - a.x);
}
struct Order {
    bool operator()(int a, int b) const {
        long double x = height(a), y = height(b);
        return x != y ? x < y : a < b;
    }
};
long double collision(int i, int j) {
    const auto &a = car[i];
    const auto &b = car[j];
    long long x = b.x - a.x, y = b.y - a.y, den = a.dx * b.dy - a.dy * b.dx;
    if (den) {
        long double u = (long double)(x * b.dy - y * b.dx) / den,
                    v = (long double)(x * a.dy - y * a.dx) / den;
        if (u < 0 || v < 0)
            return 1e100L;
        return max(u * norm(a.d) / a.speed, v * norm(b.d) / b.speed);
    }
    if (x * a.dy - y * a.dx)
        return 1e100L;
    long double z = dot(b.p - a.p, a.d) / norm(a.d);
    if (dot(a.d, b.d) > 0)
        return z >= 0 ? z / a.speed : -z / b.speed;
    return z >= 0 ? z / (a.speed + b.speed) : 1e100L;
}
bool canCrash(long double time) {
    duration = time;
    int n = car.size();
    segment.resize(n);
    struct Event {
        long double x;
        int kind, id;
    };
    vector<Event> event;
    for (int i = 0; i < n; i++) {
        Point a = car[i].p, b = a + car[i].v * time;
        if (a.x > b.x)
            swap(a, b);
        segment[i] = {a, b};
        event.push_back({a.x, 0, i});
        event.push_back({b.x, 1, i});
    }
    sort(event.begin(), event.end(),
         [](auto a, auto b) { return a.x != b.x ? a.x < b.x : a.kind < b.kind; });
    set<int, Order> active;
    vector<set<int, Order>::iterator> position(n);
    auto intersect = [&](int a, int b) { return collision(a, b) <= time; };
    for (auto e : event) {
        sweepX = e.x;
        if (e.kind == 0) {
            auto it = active.insert(e.id).first;
            position[e.id] = it;
            if (it != active.begin() && intersect(*prev(it), *it))
                return true;
            if (next(it) != active.end() && intersect(*it, *next(it)))
                return true;
        } else {
            auto it = position[e.id], right = next(it);
            if (it != active.begin() && right != active.end() && intersect(*prev(it), *right))
                return true;
            active.erase(it);
        }
    }
    return false;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    car.resize(n);
    for (auto &a : car) {
        cin >> a.x >> a.y >> a.dx >> a.dy >> a.speed;
        a.p = {(long double)a.x, (long double)a.y};
        a.d = {(long double)a.dx, (long double)a.dy};
        a.v = a.d * (a.speed / norm(a.d));
    }
    long double l = 0, r = 1e13L;
    if (!canCrash(r)) {
        cout << "No show :(";
        return 0;
    }
    for (int i = 0; i < 75; i++) {
        long double mid = (l + r) / 2;
        if (canCrash(mid))
            r = mid;
        else
            l = mid;
    }
    cout << fixed << setprecision(15) << r;
    return 0;
}
```

### 复杂度

时间 $O(n\log n\log(C/\varepsilon))$，空间 $O(n)$。

### 易错点

速度si是长度/秒，应将方向向量归一化。不能要求两车从0时刻同时启动。起点互异，但共线射线仍需专门处理。
