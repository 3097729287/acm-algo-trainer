## Jerome's House

### 题意

三个半径r圆完全在凸房间内，使三个圆心行列式绝对值最大。

### 思路

圆心可行域是房间各边向内平移r后的半平面交。输入顶点为顺时针，先反转成逆时针，再沿左法向量内缩。目标|w1+w2+w3|正是三个圆心构成三角形的二倍面积。固定另两点时面积对第三点是线性的，故最大值一定可在可行域顶点上取得。

得到凸多边形后，枚举第一个顶点i，第二个顶点j沿序前进，用单调指针k找离ij最远的顶点；凸性保证k不回退。每个i花O(h)，总O(h²)。输出二倍面积即可，题目允许圆毯互相重叠。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

constexpr long double EPS = 1e-12L;
struct Point {
    long double x = 0, y = 0;
    Point operator+(Point b) const { return {x + b.x, y + b.y}; }
    Point operator-(Point b) const { return {x - b.x, y - b.y}; }
    Point operator*(long double t) const { return {x * t, y * t}; }
    Point operator/(long double t) const { return {x / t, y / t}; }
};
long double cross(Point a, Point b) { return a.x * b.y - a.y * b.x; }
long double dot(Point a, Point b) { return a.x * b.x + a.y * b.y; }
long double norm(Point a) { return sqrtl(dot(a, a)); }
long double distance(Point a, Point b) { return norm(a - b); }
Point rotate90(Point a) { return {-a.y, a.x}; }

struct Line {
    Point p, d;
    long double angle;
};
Line line(Point a, Point b) {
    Point d = (b - a) / distance(a, b);
    return {a, d, atan2l(d.y, d.x)};
}
bool left(const Line &l, Point p) { return cross(l.d, p - l.p) >= -1e-15L; }
Point intersection(const Line &a, const Line &b) {
    return a.p + a.d * (cross(b.p - a.p, b.d) / cross(a.d, b.d));
}
vector<Point> halfplanes(vector<Line> input) {
    const long double box = 1e9L;
    vector<Point> corners = {{-box, -box}, {box, -box}, {box, box}, {-box, box}};
    for (int i = 0; i < 4; i++)
        input.push_back(line(corners[i], corners[(i + 1) % 4]));
    sort(input.begin(), input.end(),
         [](const Line &a, const Line &b) { return a.angle < b.angle; });
    vector<Line> unique;
    for (auto l : input) {
        if (!unique.empty() && fabsl(cross(unique.back().d, l.d)) < 1e-18L &&
            dot(unique.back().d, l.d) > 0) {
            if (cross(unique.back().d, l.p - unique.back().p) > 0)
                unique.back() = l;
        } else
            unique.push_back(l);
    }
    deque<Line> q;
    for (auto l : unique) {
        while (q.size() > 1 && !left(l, intersection(q[q.size() - 2], q.back())))
            q.pop_back();
        while (q.size() > 1 && !left(l, intersection(q[0], q[1])))
            q.pop_front();
        if (!q.empty() && fabsl(cross(q.back().d, l.d)) < 1e-18L)
            return {};
        q.push_back(l);
    }
    while (q.size() > 2 && !left(q.front(), intersection(q[q.size() - 2], q.back())))
        q.pop_back();
    while (q.size() > 2 && !left(q.back(), intersection(q[0], q[1])))
        q.pop_front();
    if (q.size() < 3 || fabsl(cross(q.front().d, q.back().d)) < 1e-18L)
        return {};
    vector<Point> result;
    for (int i = 0; i < (int)q.size(); i++)
        result.push_back(intersection(q[i], q[(i + 1) % q.size()]));
    return result;
}
long double polygonArea(const vector<Point> &p) {
    long double sum = 0;
    for (int i = 0; i < (int)p.size(); i++)
        sum += cross(p[i], p[(i + 1) % p.size()]);
    return fabsl(sum) / 2;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int tests;
    cin >> tests;
    while (tests--) {
        int n;
        long double radius;
        cin >> n >> radius;
        vector<Point> p(n);
        for (auto &q : p)
            cin >> q.x >> q.y;
        reverse(p.begin(), p.end());
        vector<Line> lines;
        for (int i = 0; i < n; i++) {
            Point a = p[i], b = p[(i + 1) % n],
                  offset = rotate90((b - a) / distance(a, b)) * radius;
            lines.push_back(line(a + offset, b + offset));
        }
        auto q = halfplanes(lines);
        int h = q.size();
        long double answer = 0;
        for (int i = 0; i < h; i++) {
            int k = i + 2;
            for (int j = i + 1; j + 1 < h; j++) {
                k = max(k, j + 1);
                auto area = [&](int at) { return cross(q[j] - q[i], q[at] - q[i]); };
                while (k + 1 < h && area(k + 1) > area(k))
                    k++;
                answer = max(answer, area(k));
            }
        }
        cout << fixed << setprecision(4) << answer << '\n';
    }
    return 0;
}
```

### 复杂度

时间 $O(n\log n+h^2)$，空间 $O(n+h)$。

### 易错点

不需要给三个圆心互加距离≥2r的限制，题面并未禁止地毯重叠。行列式已经是题目价格，不能再除2。
