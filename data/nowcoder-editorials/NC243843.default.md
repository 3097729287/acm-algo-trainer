## 2D Geometry 110 in 1!

### 题意

处理六种圆相关构造命令，严格按列表与元组格式输出。

### 思路

把六个命令统一成“圆心在哪里”的交点问题。外接圆：解到三个顶点等距的两条线性方程；内切圆：三个顶点按对边长度加权，半径=二倍面积/周长。

过点切线：点在圆内无解，在圆上一条切线，在圆外两条，方向角为圆心到点方向±asin(r/d)，按无向直线归一到[0,180)。

过点且与直线相切、半径固定：圆心在过该点的半径r圆上，同时在给定直线两侧各平移r的直线上，求圆线交点。与两条不平行直线相切：分别平移±r，两两相交得到4个圆心。与两不相交圆外切：圆心到两已知圆心的距离分别为r+r1、r+r2，求两圆交点。

将几何计算与输出分开：点列表按x、y排序并去重，切线角排序并去重，统一4位小数，不添加空格，消除−0.0000。相切时只输出一个交点。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

constexpr long double EPS = 1e-12L;
struct Point {
    long double x = 0, y = 0;
    Point operator+(Point b) const { return {x + b.x, y + b.y}; }
    Point operator-(Point b) const { return {x - b.x, y - b.y}; }
    Point operator*(long double t) const { return {x * t, y * t}; }
    Point operator/(long double t) const { return {x / t, y / t}; }
};
long double cross(Point a, Point b) { return a.x * b.y - a.y * b.x; }
long double dot(Point a, Point b) { return a.x * b.x + a.y * b.y; }
long double norm(Point a) { return sqrtl(dot(a, a)); }
long double distance(Point a, Point b) { return norm(a - b); }
Point rotate90(Point a) { return {-a.y, a.x}; }

struct Circle {
    Point center;
    long double radius = -1;
};
bool contains(Circle c, Point p) { return distance(c.center, p) <= c.radius + 1e-12L; }
Circle diameter(Point a, Point b) {
    Point c = (a + b) / 2;
    return {c, distance(a, b) / 2};
}
Circle throughThree(Point a, Point b, Point c) {
    Point u = b - a, v = c - a;
    long double den = 2 * cross(u, v);
    if (fabsl(den) < 1e-24L) {
        Circle best = {{}, numeric_limits<long double>::infinity()};
        for (auto d : {diameter(a, b), diameter(a, c), diameter(b, c)})
            if (contains(d, a) && contains(d, b) && contains(d, c) && d.radius < best.radius)
                best = d;
        return best;
    }
    Point offset = {(dot(u, u) * v.y - dot(v, v) * u.y) / den,
                    (u.x * dot(v, v) - v.x * dot(u, u)) / den};
    Point center = a + offset;
    return {center, distance(center, a)};
}

vector<Point> circleIntersections(Point a, long double r, Point b, long double s) {
    Point e = b - a;
    long double d = norm(e);
    if (d < EPS || d > r + s + EPS || d < fabsl(r - s) - EPS)
        return {};
    long double x = (r * r - s * s + d * d) / (2 * d), h2 = r * r - x * x;
    if (h2 < -EPS)
        return {};
    Point center = a + e * (x / d), offset = rotate90(e) * (sqrtl(max(0.0L, h2)) / d);
    if (norm(offset) < EPS)
        return {center};
    return {center + offset, center - offset};
}

Point lineIntersection(Point a, Point u, Point b, Point v) {
    return a + u * (cross(b - a, v) / cross(u, v));
}
vector<Point> lineCircle(Point a, Point d, Point center, long double radius) {
    Point projection = a + d * (dot(center - a, d) / dot(d, d));
    long double h2 = radius * radius - dot(projection - center, projection - center);
    if (h2 < -1e-9L)
        return {};
    if (fabsl(h2) < 1e-14L * max(1.0L, radius * radius))
        h2 = 0;
    Point offset = d * (sqrtl(max(0.0L, h2)) / norm(d));
    if (norm(offset) < 1e-10L)
        return {projection};
    return {projection - offset, projection + offset};
}
long double clean(long double x) { return fabsl(x) < 0.000049999L ? 0 : x; }
void pointOutput(Point p) { cout << '(' << clean(p.x) << ',' << clean(p.y) << ')'; }
void pointsOutput(vector<Point> points) {
    sort(points.begin(), points.end(),
         [](Point a, Point b) { return fabsl(a.x - b.x) > 1e-9L ? a.x < b.x : a.y < b.y; });
    points.erase(unique(points.begin(), points.end(),
                        [](Point a, Point b) { return distance(a, b) < 1e-9L; }),
                 points.end());
    cout << '[';
    for (int i = 0; i < (int)points.size(); i++) {
        if (i)
            cout << ',';
        pointOutput(points[i]);
    }
    cout << "]\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    string command;
    cout << fixed << setprecision(4);
    const long double pi = acosl(-1.0L);
    while (cin >> command) {
        if (command == "CircumscribedCircle" || command == "InscribedCircle") {
            Point a, b, c;
            cin >> a.x >> a.y >> b.x >> b.y >> c.x >> c.y;
            Circle result;
            if (command == "CircumscribedCircle")
                result = throughThree(a, b, c);
            else {
                long double x = distance(b, c), y = distance(a, c), z = distance(a, b),
                            sum = x + y + z;
                result = {(a * x + b * y + c * z) / sum, fabsl(cross(b - a, c - a)) / sum};
            }
            cout << '(' << clean(result.center.x) << ',' << clean(result.center.y) << ','
                 << clean(result.radius) << ")\n";
        } else if (command == "TangentLineThroughPoint") {
            Point c, p;
            long double r;
            cin >> c.x >> c.y >> r >> p.x >> p.y;
            Point d = p - c;
            long double length = norm(d);
            vector<long double> angle;
            if (length >= r - 1e-10L) {
                long double base = atan2l(d.y, d.x), delta = asinl(clamp(r / length, -1.0L, 1.0L));
                for (long double a : {base - delta, base + delta}) {
                    a = fmodl(a, pi);
                    if (a < 0)
                        a += pi;
                    if (a > pi - 1e-12L)
                        a = 0;
                    angle.push_back(a * 180 / pi);
                }
                sort(angle.begin(), angle.end());
                angle.erase(unique(angle.begin(), angle.end(),
                                   [](auto a, auto b) { return fabsl(a - b) < 1e-8L; }),
                            angle.end());
            }
            cout << '[';
            for (int i = 0; i < (int)angle.size(); i++) {
                if (i)
                    cout << ',';
                cout << clean(angle[i]);
            }
            cout << "]\n";
        } else if (command == "CircleThroughAPointAndTangentToALineWithRadius") {
            Point p, a, b;
            long double r;
            cin >> p.x >> p.y >> a.x >> a.y >> b.x >> b.y >> r;
            Point d = b - a, n = rotate90(d) * (r / norm(d));
            vector<Point> answer;
            for (int sign : {-1, 1}) {
                auto points = lineCircle(a + n * sign, d, p, r);
                answer.insert(answer.end(), points.begin(), points.end());
            }
            pointsOutput(answer);
        } else if (command == "CircleTangentToTwoLinesWithRadius") {
            Point a, b, c, d;
            long double r;
            cin >> a.x >> a.y >> b.x >> b.y >> c.x >> c.y >> d.x >> d.y >> r;
            Point u = b - a, v = d - c, n = rotate90(u) * (r / norm(u)),
                  m = rotate90(v) * (r / norm(v));
            vector<Point> answer;
            for (int x : {-1, 1})
                for (int y : {-1, 1})
                    answer.push_back(lineIntersection(a + n * x, u, c + m * y, v));
            pointsOutput(answer);
        } else {
            Point a, b;
            long double r, s, t;
            cin >> a.x >> a.y >> r >> b.x >> b.y >> s >> t;
            pointsOutput(circleIntersections(a, r + t, b, s + t));
        }
    }
    return 0;
}
```

### 复杂度

每条命令时间、额外空间均为 $O(1)$。

### 易错点

这版要求4位小数，原UVA题的6位不能照搬。圆心列表只输出(x,y)，前两个三角形命令输出(x,y,r)。切线角是无向直线角，180度要归零。
