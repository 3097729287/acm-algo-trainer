## [HAOI2006]受欢迎的牛

### 题意

有向关系可传递，统计所有点都能到达的点数。

### 思路

缩点DAG中，若有两个不同出度0分量，它们无法彼此到达，无人能被全体到达。若恰有一个汇分量，DAG中任一点沿出边最终必到某个汇，故全部会到这个唯一汇。答案就是该SCC大小。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

pair<int, vector<int>> components(const vector<vector<int>> &graph,
                                  const vector<vector<int>> &reverse) {
    int n = graph.size();
    vector<bool> seen(n);
    vector<int> finish;
    for (int start = 0; start < n; start++)
        if (!seen[start]) {
            vector<pair<int, int>> stack{{start, 0}};
            seen[start] = true;
            while (!stack.empty()) {
                int u = stack.back().first, &position = stack.back().second;
                if (position == (int)graph[u].size()) {
                    finish.push_back(u);
                    stack.pop_back();
                } else {
                    int v = graph[u][position++];
                    if (!seen[v]) {
                        seen[v] = true;
                        stack.push_back({v, 0});
                    }
                }
            }
        }
    vector<int> component(n, -1);
    int count = 0;
    for (int i = n - 1; i >= 0; i--) {
        int start = finish[i];
        if (component[start] != -1)
            continue;
        vector<int> stack{start};
        component[start] = count;
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            for (int v : reverse[u])
                if (component[v] == -1) {
                    component[v] = count;
                    stack.push_back(v);
                }
        }
        count++;
    }
    return {count, component};
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    vector<vector<int>> graph(n), reverse(n);
    while (m--) {
        int u, v;
        cin >> u >> v;
        u--;
        v--;
        graph[u].push_back(v);
        reverse[v].push_back(u);
    }
    auto [count, component] = components(graph, reverse);
    vector<int> size(count), out(count);
    for (int u = 0; u < n; u++) {
        size[component[u]]++;
        for (int v : graph[u])
            if (component[u] != component[v])
                out[component[u]]++;
    }
    int target = -1;
    for (int c = 0; c < count; c++)
        if (!out[c]) {
            if (target >= 0) {
                cout << 0;
                return 0;
            }
            target = c;
        }
    cout << size[target];
    return 0;
}
```

### 复杂度

时间、空间均 $O(n+m)$。

### 易错点

所求是所有点都能到它，关注汇分量，而不是入度0源分量。重复边不影响是否为0。
