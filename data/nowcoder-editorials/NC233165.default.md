## Enclosure

### 题意

已有k个点，加入一个候选点后最大凸包面积。

### 思路

先求已有点逆时针凸包及各边叉积前缀和。内点不改变面积；外点q会替换一段连续可见边，其中可见条件cross(edge,q−start)<0。用扇形二分找到一条可见边e。

以e的内法向量求最远支撑顶点，它的两条相邻边至少一条不可见：若q在两边外侧，沿此支撑方向的投影必超过最远顶点，和q位于e外侧矛盾。于是得到一个可见、一个不可见边。可见边在环上连续，分别沿两条弧二分找可见段首尾，得到两切点A、B。

新二倍面积=旧二倍面积−可见边叉积和+cross(A,q)+cross(q,B)。前缀和使面积替换常数完成，每个候选只做对数次支撑与侧别判断。全部面积用128位，精确输出.0或.5。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

using Wide = __int128_t;
struct IPoint {
    long long x = 0, y = 0;
    IPoint operator+(IPoint b) const { return {x + b.x, y + b.y}; }
    IPoint operator-(IPoint b) const { return {x - b.x, y - b.y}; }
};
Wide cross(IPoint a, IPoint b) { return (Wide)a.x * b.y - (Wide)a.y * b.x; }
Wide dot(IPoint a, IPoint b) { return (Wide)a.x * b.x + (Wide)a.y * b.y; }
Wide absolute(Wide x) { return x < 0 ? -x : x; }
void printWide(Wide value) {
    if (value < 0) {
        cout << '-';
        value = -value;
    }
    string text;
    do {
        text.push_back('0' + value % 10);
        value /= 10;
    } while (value);
    reverse(text.begin(), text.end());
    cout << text;
}
bool angleLess(IPoint a, IPoint b) {
    auto half = [](IPoint p) { return p.y < 0 || (p.y == 0 && p.x < 0); };
    if (half(a) != half(b))
        return half(a) < half(b);
    return cross(a, b) > 0;
}
vector<IPoint> integerHull(vector<IPoint> point) {
    sort(point.begin(), point.end(),
         [](auto a, auto b) { return a.x != b.x ? a.x < b.x : a.y < b.y; });
    point.erase(
        unique(point.begin(), point.end(), [](auto a, auto b) { return a.x == b.x && a.y == b.y; }),
        point.end());
    if (point.size() < 2)
        return point;
    vector<IPoint> hull;
    for (auto p : point) {
        while (hull.size() >= 2 && cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    size_t lower = hull.size();
    for (int i = (int)point.size() - 2; i >= 0; i--) {
        auto p = point[i];
        while (hull.size() > lower &&
               cross(hull.back() - hull[hull.size() - 2], p - hull.back()) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    hull.pop_back();
    return hull;
}

struct Support {
    const vector<IPoint> &point;
    vector<pair<IPoint, int>> normal;
    explicit Support(const vector<IPoint> &p) : point(p) {
        if (p.size() < 3)
            return;
        for (int i = 0; i < (int)p.size(); i++) {
            IPoint e = p[(i + 1) % p.size()] - p[i];
            normal.push_back({{e.y, -e.x}, i});
        }
        sort(normal.begin(), normal.end(),
             [](auto a, auto b) { return angleLess(a.first, b.first); });
    }
    int index(IPoint direction) const {
        if (point.size() < 3) {
            int best = 0;
            for (int i = 1; i < (int)point.size(); i++)
                if (dot(point[i], direction) > dot(point[best], direction))
                    best = i;
            return best;
        }
        auto it = lower_bound(normal.begin(), normal.end(), direction,
                              [](const auto &a, IPoint b) { return angleLess(a.first, b); });
        if (it == normal.end())
            it = normal.begin();
        return it->second;
    }
};
int visibleEdge(const vector<IPoint> &p, IPoint q) {
    int n = p.size();
    if (cross(p[1] - p[0], q - p[0]) < 0)
        return 0;
    if (cross(p[n - 1] - p[0], q - p[0]) > 0)
        return n - 1;
    int l = 1, r = n - 1;
    while (r - l > 1) {
        int mid = (l + r) / 2;
        if (cross(p[mid] - p[0], q - p[0]) >= 0)
            l = mid;
        else
            r = mid;
    }
    return cross(p[r] - p[l], q - p[l]) < 0 ? l : -1;
}
bool insideConvex(const vector<IPoint> &p, IPoint q, bool strict) {
    int n = p.size();
    if (n == 1)
        return !strict && q.x == p[0].x && q.y == p[0].y;
    if (n == 2)
        return !strict && cross(p[1] - p[0], q - p[0]) == 0 && dot(q - p[0], q - p[1]) <= 0;
    Wide a = cross(p[1] - p[0], q - p[0]), b = cross(p[n - 1] - p[0], q - p[0]);
    if (a < 0 || b > 0)
        return false;
    if (a == 0)
        return !strict && dot(q - p[0], q - p[1]) <= 0;
    if (b == 0)
        return !strict && dot(q - p[0], q - p[n - 1]) <= 0;
    int l = 1, r = n - 1;
    while (r - l > 1) {
        int mid = (l + r) / 2;
        if (cross(p[mid] - p[0], q - p[0]) >= 0)
            l = mid;
        else
            r = mid;
    }
    Wide side = cross(p[r] - p[l], q - p[l]);
    return strict ? side > 0 : side >= 0;
}
vector<IPoint> minkowski(vector<IPoint> a, vector<IPoint> b) {
    if (a.size() < 3 || b.size() < 3) {
        vector<IPoint> sum;
        for (auto p : a)
            for (auto q : b)
                sum.push_back(p + q);
        return integerHull(sum);
    }
    auto prepare = [](vector<IPoint> &p) {
        auto it = min_element(p.begin(), p.end(),
                              [](auto a, auto b) { return a.y != b.y ? a.y < b.y : a.x < b.x; });
        rotate(p.begin(), it, p.end());
    };
    prepare(a);
    prepare(b);
    int n = a.size(), m = b.size(), i = 0, j = 0;
    vector<IPoint> result;
    IPoint current = a[0] + b[0];
    result.push_back(current);
    while (i < n || j < m) {
        IPoint x = i < n ? a[(i + 1) % n] - a[i] : IPoint{},
               y = j < m ? b[(j + 1) % m] - b[j] : IPoint{};
        Wide c = cross(x, y);
        if (j == m || (i < n && c > 0)) {
            current = current + x;
            i++;
        } else if (i == n || c < 0) {
            current = current + y;
            j++;
        } else {
            current = current + x + y;
            i++;
            j++;
        }
        result.push_back(current);
    }
    result.pop_back();
    return result;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, k;
    cin >> n >> k;
    vector<IPoint> p(n);
    for (auto &a : p)
        cin >> a.x >> a.y;
    auto hull = integerHull(vector<IPoint>(p.begin(), p.begin() + k));
    int h = hull.size();
    Support support(hull);
    vector<Wide> prefix(2 * h + 1);
    for (int i = 0; i < 2 * h; i++)
        prefix[i + 1] = prefix[i] + cross(hull[i % h], hull[(i + 1) % h]);
    Wide base = prefix[h], answer = base;
    for (int i = k; i < n; i++) {
        IPoint q = p[i];
        int visible = visibleEdge(hull, q);
        if (visible < 0)
            continue;
        auto side = [&](int edge) {
            return cross(hull[(edge + 1) % h] - hull[edge], q - hull[edge]);
        };
        IPoint e = hull[(visible + 1) % h] - hull[visible], normal = {-e.y, e.x};
        int vertex = support.index(normal),
            hidden = side(vertex) > 0 ? vertex : (vertex + h - 1) % h;
        int distance = (hidden - visible + h) % h, low = 0, high = distance;
        while (high - low > 1) {
            int mid = (low + high) / 2;
            if (side((visible + mid) % h) < 0)
                low = mid;
            else
                high = mid;
        }
        int end = (visible + low) % h;
        distance = (visible - hidden + h) % h;
        low = 0;
        high = distance;
        while (high - low > 1) {
            int mid = (low + high) / 2;
            if (side((hidden + mid) % h) > 0)
                low = mid;
            else
                high = mid;
        }
        int start = (hidden + high) % h, count = (end - start + h) % h + 1;
        Wide removed = prefix[start + count] - prefix[start];
        Wide area = base - removed + cross(hull[start], q) + cross(q, hull[(end + 1) % h]);
        answer = max(answer, area);
    }
    printWide(answer / 2);
    cout << (answer % 2 ? ".5" : ".0");
    return 0;
}
```

### 复杂度

时间 $O(k\log k+(n-k)\log k)$，空间 $O(n+k)$。

### 易错点

无三点共线保证候选不会恰好落在某凸包边延长线上，侧别二分无零值歧义。不能只取离某一个点最远的候选。
