## [USACO 2017 Jan P]Promotion Counting

### 题意

每个节点统计子树中能力严格更高的后代数。

### 思路

先用DFS序得到每个子树区间，然后按能力从高到低插入节点。处理u时，已插入的恰是能力更高者，查询其子树区间计数就是答案，随后把u插入。原题能力互不相同，因此无需同值分组。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

struct EulerTree {
    int n;
    vector<vector<int>> edge;
    vector<int> parent, depth, in, out, order;
    EulerTree(int n) : n(n), edge(n + 1), parent(n + 1), depth(n + 1), in(n + 1), out(n + 1) {}
    void add(int u, int v) {
        edge[u].push_back(v);
        edge[v].push_back(u);
    }
    void root(int start) {
        vector<int> stack = {start};
        parent[start] = 0;
        order.clear();
        while (!stack.empty()) {
            int u = stack.back();
            stack.pop_back();
            in[u] = order.size() + 1;
            order.push_back(u);
            for (int v : edge[u])
                if (v != parent[u]) {
                    parent[v] = u;
                    depth[v] = depth[u] + 1;
                    stack.push_back(v);
                }
        }
        vector<int> size(n + 1, 1);
        for (int i = (int)order.size() - 1; i >= 0; i--) {
            int u = order[i];
            out[u] = in[u] + size[u] - 1;
            if (parent[u])
                size[parent[u]] += size[u];
        }
    }
};

struct Fenwick {
    int n;
    vector<long long> tree;
    Fenwick(int n) : n(n), tree(n + 1) {}
    void add(int p, long long v) {
        for (; p <= n; p += p & -p)
            tree[p] += v;
    }
    long long sum(int p) {
        long long result = 0;
        for (; p; p -= p & -p)
            result += tree[p];
        return result;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<long long> power(n + 1);
    vector<int> order(n);
    for (int i = 1; i <= n; i++) {
        cin >> power[i];
        order[i - 1] = i;
    }
    EulerTree tree(n);
    for (int v = 2; v <= n; v++) {
        int p;
        cin >> p;
        tree.add(p, v);
    }
    tree.root(1);
    sort(order.begin(), order.end(), [&](int a, int b) { return power[a] > power[b]; });
    Fenwick bit(n);
    vector<int> answer(n + 1);
    for (int u : order) {
        answer[u] = bit.sum(tree.out[u]) - bit.sum(tree.in[u] - 1);
        bit.add(tree.in[u], 1);
    }
    for (int u = 1; u <= n; u++)
        cout << answer[u] << '\n';
    return 0;
}
```

### 复杂度

时间 $O(n\log n)$，空间 $O(n)$。

### 易错点

先查询后插入，避免把自身算入后代。统计严格高于而非不低于。
