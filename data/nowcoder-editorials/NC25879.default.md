## 外挂

### 题意

区间加并询问所有下标对i<j的a[i]*a[j]之和。

### 思路

节点维护总和S和下标对乘积和P。左右合并时，跨左右的对贡献S左*S右，所以P=P左+P右+S左*S右。长len的一段统一加x，每一对新增x*(a[i]+a[j])+x²；每个原值被计算len−1次，因此P新增x*(len−1)*S+C(len,2)*x²，S新增len*x。以区间加懒标记维护即可。

中间乘积增长比区间和更快，代码用128位保存S、P与乘法，并提供十进制输出。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, q;
    cin >> n >> q;
    vector<__int128> sum(4 * n + 4), pairs(sum.size()), tag(sum.size());
    auto pull = [&](int u) {
        sum[u] = sum[u * 2] + sum[u * 2 + 1];
        pairs[u] = pairs[u * 2] + pairs[u * 2 + 1] + sum[u * 2] * sum[u * 2 + 1];
    };
    auto apply = [&](int u, int l, int r, __int128 x) {
        long long len = r - l + 1;
        pairs[u] += x * (len - 1) * sum[u] + (__int128)len * (len - 1) / 2 * x * x;
        sum[u] += len * x;
        tag[u] += x;
    };
    auto build = [&](auto &&self, int u, int l, int r) -> void {
        if (l == r) {
            long long x;
            cin >> x;
            sum[u] = x;
            return;
        }
        int mid = (l + r) / 2;
        self(self, u * 2, l, mid);
        self(self, u * 2 + 1, mid + 1, r);
        pull(u);
    };
    build(build, 1, 1, n);
    auto push = [&](int u, int l, int r) {
        if (!tag[u] || l == r)
            return;
        int mid = (l + r) / 2;
        apply(u * 2, l, mid, tag[u]);
        apply(u * 2 + 1, mid + 1, r, tag[u]);
        tag[u] = 0;
    };
    auto change = [&](auto &&self, int u, int l, int r, int ql, int qr, long long x) -> void {
        if (ql <= l && r <= qr) {
            apply(u, l, r, x);
            return;
        }
        push(u, l, r);
        int mid = (l + r) / 2;
        if (ql <= mid)
            self(self, u * 2, l, mid, ql, qr, x);
        if (qr > mid)
            self(self, u * 2 + 1, mid + 1, r, ql, qr, x);
        pull(u);
    };
    auto query = [&](auto &&self, int u, int l, int r, int ql, int qr) -> pair<__int128, __int128> {
        if (ql <= l && r <= qr)
            return {sum[u], pairs[u]};
        push(u, l, r);
        int mid = (l + r) / 2;
        pair<__int128, __int128> a = {0, 0}, b = {0, 0};
        if (ql <= mid)
            a = self(self, u * 2, l, mid, ql, qr);
        if (qr > mid)
            b = self(self, u * 2 + 1, mid + 1, r, ql, qr);
        return {a.first + b.first, a.second + b.second + a.first * b.first};
    };
    auto print = [](__int128 value) {
        if (value < 0) {
            cout << '-';
            value = -value;
        }
        string text;
        do {
            text.push_back('0' + value % 10);
            value /= 10;
        } while (value);
        reverse(text.begin(), text.end());
        cout << text << '\n';
    };
    while (q--) {
        int op, l, r;
        cin >> op >> l >> r;
        if (op == 1) {
            long long x;
            cin >> x;
            change(change, 1, 1, n, l, r, x);
        } else
            print(query(query, 1, 1, n, l, r).second);
    }
    return 0;
}
```

### 复杂度

构建 $O(n)$，每次操作 $O(\log n)$，空间 $O(n)$。

### 易错点

先用旧S更新P，再修改S。单元素区间没有数对，P为0；两个查询结果的合并也要加入跨段乘积。
