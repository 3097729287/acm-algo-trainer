## 大吉大利，晚上吃鸡！

### 题意

统计无序点对，使每条S到T最短路恰好经过其中一个点。

### 思路

先从S、T各做一次Dijkstra，只保留满足 distS[u]+w+distT[v]=distS[T] 的方向u→v，得到全部最短路组成的DAG；边权为正，distS递增就是拓扑序。答案只依赖这个DAG的路径结构。

**第一步：找所有必经点。** 在DAG中，点v的直接支配点是其全部前驱在支配树中的LCA：一条路可以由任意前驱进入v，所以v之前的共同必经点，恰是这些前驱的共同支配祖先。按拓扑序增量建树，用倍增求LCA。T在支配树上的全部祖先就是S到T的必经点。

若A是必经点，B必须完全不在最短路DAG上，否则存在同时经过两者的最短路。因此这类贡献是“必经点数×DAG外点数”。两个DAG内部的有效候选均非必经点，且必定位于相邻两个必经点之间的同一段，否则可以同时经过。

**第二步：固定两条参考路径。** 将DAG结点拆成入、出两点，非必经点的通过容量为1，必经点为2，原边容量为2，送两单位S到T流。任一小于2的点割只能是单个必经点，所以两单位流一定可行。将流分解为路径P、Q，它们只在必经点重合。在每个相邻必经点之间，两路内部不相交。任何覆盖全部路径的两个非必经点，必须分别在P、Q上选一个；否则至少一条参考路径完全没被挡住。

**第三步：判断一对切点。** 考虑这一段的P[i]、Q[j]。把路径外结点组成的连续段压成跳转：从某个参考路径结点u出发，记录不经过其他参考路径结点之前，能首次接触P、Q的最大下标，记jumpP[u]、jumpQ[u]。在反拓扑序中，路径结点立即停止，其他结点取后继最大值，即可线性求出这些跳转。

P[i]之前所有P结点的跳转取前缀最大值，记PP[i]、PQ[i]；Q[j]之前同理记QQ[j]、QP[j]。删去P[i]、Q[j]后仍能到终点，当且仅当某次跳转越过了两路构成的分界。因此二者为合法割点，恰需：

`PP[i] <= i，QQ[j] <= j，PQ[i] <= j，QP[j] <= i`。

前两项排除本路绕过切点，后两项排除从一路切点之前跳到另一路切点之后。若存在避开切点的完整路，观察它第一次从分界前走到分界后的相邻两次参考路径接触，就会违反其中一项；反之任何违反的跳转，都可接上参考路径的前缀和后缀，构造一条漏网路径。

**第四步：排除同一路上的两点。** 组成一个二点割还不够，可能存在一条路先经过P[i]再经过Q[j]。另外在反拓扑序中求reachQ[u]：从u出发能到达的Q的最小下标；沿途经过参考路径结点也继续传播，不像上一步在首次接触处停止。则P[i]能到Q[j]恰当且仅当reachQ[P[i]]<=j，因为到达更早的Q位置后可沿Q走到j。同理定义reachP。还需 j<reachQ[P[i]] 且 i<reachP[Q[j]]，精确保证两点无法先后出现在同一路上。

固定i时，PQ[i]给出j的下界，单调的QP[j]给出上界；reachQ给出另一上界，沿Q单调不减的reachP给出另一下界。对QQ[j]<=j的点预做前缀计数，两次二分交出合法下标区间，再常数时间取数量。各段相加，计数完整且不重复。若S、T不可达，没有可能路径，两条全称条件均成立，答案为全部点对；S=T时仅含S的零边路径，答案n−1。

### 参考代码

```cpp
#include <bits/stdc++.h>
using namespace std;

#include <cassert>

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m, s, t;
    cin >> n >> m >> s >> t;
    --s;
    --t;
    struct Road {
        int u, v;
        long long w;
    };
    vector<Road> roads(m);
    vector<vector<pair<int, long long>>> graph(n);
    for (auto &e : roads) {
        cin >> e.u >> e.v >> e.w;
        --e.u;
        --e.v;
        graph[e.u].push_back({e.v, e.w});
        graph[e.v].push_back({e.u, e.w});
    }
    if (s == t) {
        cout << n - 1;
        return 0;
    }
    const long long inf = 1LL << 60;
    auto dijkstra = [&](int start) {
        vector<long long> d(n, inf);
        priority_queue<pair<long long, int>, vector<pair<long long, int>>,
                       greater<pair<long long, int>>>
            heap;
        d[start] = 0;
        heap.push({0, start});
        while (!heap.empty()) {
            auto [cost, u] = heap.top();
            heap.pop();
            if (cost != d[u])
                continue;
            for (auto [v, w] : graph[u])
                if (d[v] > cost + w) {
                    d[v] = cost + w;
                    heap.push({d[v], v});
                }
        }
        return d;
    };
    auto ds = dijkstra(s), dt = dijkstra(t);
    if (ds[t] == inf) {
        cout << 1LL * n * (n - 1) / 2;
        return 0;
    }
    vector<vector<int>> dag(n), pred(n);
    vector<int> order;
    vector<bool> onPath(n);
    for (int u = 0; u < n; u++)
        if (ds[u] != inf && dt[u] != inf && ds[u] + dt[u] == ds[t]) {
            onPath[u] = true;
            order.push_back(u);
        }
    auto addDag = [&](int u, int v, long long w) {
        if (ds[u] + w + dt[v] == ds[t]) {
            dag[u].push_back(v);
            pred[v].push_back(u);
        }
    };
    for (auto e : roads) {
        addDag(e.u, e.v, e.w);
        addDag(e.v, e.u, e.w);
    }
    sort(order.begin(), order.end(), [&](int a, int b) { return ds[a] < ds[b]; });
    constexpr int log = 18;
    vector<array<int, log>> up(n);
    vector<int> depth(n), dominator(n, -1);
    up[s].fill(s);
    dominator[s] = s;
    auto lca = [&](int a, int b) {
        if (depth[a] < depth[b])
            swap(a, b);
        int difference = depth[a] - depth[b];
        for (int j = 0; j < log; j++)
            if (difference >> j & 1)
                a = up[a][j];
        if (a == b)
            return a;
        for (int j = log - 1; j >= 0; j--)
            if (up[a][j] != up[b][j]) {
                a = up[a][j];
                b = up[b][j];
            }
        return up[a][0];
    };
    for (int u : order)
        if (u != s) {
            int p = pred[u][0];
            for (int v : pred[u])
                p = lca(p, v);
            dominator[u] = p;
            depth[u] = depth[p] + 1;
            up[u][0] = p;
            for (int j = 1; j < log; j++)
                up[u][j] = up[up[u][j - 1]][j - 1];
        }
    vector<bool> mandatory(n);
    vector<int> gates;
    for (int u = t;; u = dominator[u]) {
        gates.push_back(u);
        mandatory[u] = true;
        if (u == s)
            break;
    }
    reverse(gates.begin(), gates.end());
    struct FlowEdge {
        int to, capacity, initial;
    };
    vector<FlowEdge> flow;
    vector<vector<int>> network(2 * n);
    auto addFlow = [&](int u, int v, int capacity) {
        int index = flow.size();
        flow.push_back({v, capacity, capacity});
        flow.push_back({u, 0, 0});
        network[u].push_back(index);
        network[v].push_back(index + 1);
    };
    for (int u : order) {
        addFlow(2 * u, 2 * u + 1, mandatory[u] ? 2 : 1);
        for (int v : dag[u])
            addFlow(2 * u + 1, 2 * v, 2);
    }
    int source = 2 * s + 1, sink = 2 * t;
    for (int unit = 0; unit < 2; unit++) {
        vector<int> previous(2 * n, -1), queue{source};
        previous[source] = -2;
        for (size_t h = 0; h < queue.size() && previous[sink] == -1; h++)
            for (int e : network[queue[h]])
                if (flow[e].capacity > 0 && previous[flow[e].to] == -1) {
                    previous[flow[e].to] = e;
                    queue.push_back(flow[e].to);
                }
        assert(previous[sink] != -1);
        for (int v = sink; v != source;) {
            int e = previous[v];
            flow[e].capacity--;
            flow[e ^ 1].capacity++;
            v = flow[e ^ 1].to;
        }
    }
    auto extract = [&]() {
        vector<int> path{s};
        int u = source;
        while (u != sink) {
            int chosen = -1;
            for (int e : network[u])
                if (flow[e].initial > 0 && flow[e].capacity < flow[e].initial) {
                    chosen = e;
                    break;
                }
            assert(chosen != -1);
            flow[chosen].capacity++;
            flow[chosen ^ 1].capacity--;
            u = flow[chosen].to;
            if (u % 2 == 0)
                path.push_back(u / 2);
        }
        return path;
    };
    auto p = extract(), q = extract();
    vector<int> pIndex(n, -1), qIndex(n, -1);
    for (int i = 0; i < (int)p.size(); i++)
        pIndex[p[i]] = i;
    for (int i = 0; i < (int)q.size(); i++)
        qIndex[q[i]] = i;
    vector<int> firstP(n, -1), firstQ(n, -1), jumpP(n, -1), jumpQ(n, -1);
    for (int index = (int)order.size() - 1; index >= 0; index--) {
        int u = order[index];
        for (int v : dag[u]) {
            jumpP[u] = max(jumpP[u], firstP[v]);
            jumpQ[u] = max(jumpQ[u], firstQ[v]);
        }
        if (pIndex[u] != -1 || qIndex[u] != -1) {
            firstP[u] = pIndex[u];
            firstQ[u] = qIndex[u];
        } else {
            firstP[u] = jumpP[u];
            firstQ[u] = jumpQ[u];
        }
    }
    vector<int> reachP(n, n), reachQ(n, n);
    for (int index = (int)order.size() - 1; index >= 0; index--) {
        int u = order[index];
        if (pIndex[u] >= 0)
            reachP[u] = pIndex[u];
        if (qIndex[u] >= 0)
            reachQ[u] = qIndex[u];
        for (int v : dag[u]) {
            reachP[u] = min(reachP[u], reachP[v]);
            reachQ[u] = min(reachQ[u], reachQ[v]);
        }
    }
    long long answer = 1LL * gates.size() * (n - (int)order.size());
    for (int segment = 0; segment + 1 < (int)gates.size(); segment++) {
        int leftP = pIndex[gates[segment]], rightP = pIndex[gates[segment + 1]],
            leftQ = qIndex[gates[segment]], rightQ = qIndex[gates[segment + 1]];
        vector<int> cross, earliestP, prefix{0};
        int ownQ = -1, crossQ = -1;
        for (int j = leftQ + 1; j < rightQ; j++) {
            int u = q[j - 1];
            ownQ = max(ownQ, jumpQ[u]);
            crossQ = max(crossQ, jumpP[u]);
            cross.push_back(crossQ);
            earliestP.push_back(reachP[q[j]]);
            prefix.push_back(prefix.back() + (ownQ <= j));
        }
        int ownP = -1, crossP = -1;
        for (int i = leftP + 1; i < rightP; i++) {
            int u = p[i - 1];
            ownP = max(ownP, jumpP[u]);
            crossP = max(crossP, jumpQ[u]);
            if (ownP > i)
                continue;
            int lower = clamp(max(leftQ + 1, crossP) - leftQ - 1, 0, (int)cross.size());
            lower = max(lower, (int)(upper_bound(earliestP.begin(), earliestP.end(), i) -
                                     earliestP.begin()));
            int upper = upper_bound(cross.begin(), cross.end(), i) - cross.begin();
            upper = min(upper, clamp(reachQ[p[i]] - leftQ - 1, 0, (int)cross.size()));
            if (lower < upper)
                answer += prefix[upper] - prefix[lower];
        }
    }
    cout << answer;
    return 0;
}
```

### 复杂度

两次最短路、支配树LCA及分界二分合计 $O((n+m)\log n)$；两单位增广与路径压缩均为 $O(n+m)$。空间 $O(n\log n+m)$。

### 易错点

只比较经过两点的最短路数量之和不足以保证两点不会在同一路上。这里用路径分界的结构条件精确检查，不对路径数量取模。DAG外点仍可能与必经点组成有效方案，不能直接删掉而不计数。原样例解释列出的(2,4)同处一条最短路，应改为(2,5)；正确六对是{2,4,6}与{3,5}的全部交叉组合。
